section ‹Semi_Thue Systems›

theory Semi_Thue_Systems
  imports
    String_Rewriting
    ShortLex
    Jacobson_Basic_Algebra.Group_Theory
    Jacobson_Basic_Algebra.Set_Theory
begin

hide_const inverse

no_notation divide (infixl "'/" 70)
no_notation inverse_divide (infixl "'/" 70)
no_notation subst_apply_term (infixl "⋅" 67)

inductive_set Kleene_star :: "char set ⇒ string set"  ("(_⇧⋆)" [1000] 999)
  for S :: "char set"
  where
    Kleen_star_basis: "[] ∈ S⇧⋆"
  | Kleene_star_step: "a ∈ S⇧⋆ ⟹ b ∈ S ⟹ a @ [b] ∈ S⇧⋆"

lemmas  Kleene_starI =  Kleene_star.intros [intro]
lemmas  Kleene_starE =  Kleene_star.cases [elim]

notation Nil ("ε")

locale semi_Thue = 
  fixes R :: sts (* A semi-Thue system *)
    and T:: sts
    and S:: "char set" (* alphabet *)
  assumes cong:"T = (ststep R)⇧↔⇧*" (* Thue relation *)
    and "finite S"
    and "S ≠ {}"
    and "finite R"
    and R_sig:"(ststep R)⇧* ⊆ S⇧⋆ × S⇧⋆"
begin

sublocale mo: monoid "S⇧⋆" append ε 
proof(unfold_locales)
  show "⋀s t. s ∈ S⇧⋆ ⟹ t ∈ S⇧⋆ ⟹ s @ t ∈ S⇧⋆"
    using R_sig local.cong by blast
  show "ε ∈ S⇧⋆" by (simp add: Kleene_star.Kleen_star_basis)
  show "⋀s t u. s ∈ S⇧⋆ ⟹ t ∈ S⇧⋆ ⟹ u ∈ S⇧⋆ ⟹ (s @ t) @ u = s @ t @ u" by simp
  show "⋀s. s ∈ S⇧⋆ ⟹ ε @ s = s" by simp
  show "⋀s. s ∈ S⇧⋆ ⟹ s @ ε = s" by simp
qed

sublocale eq: equivalence "S⇧⋆" "T"
proof(unfold_locales)
  show "T ⊆ S⇧⋆ × S⇧⋆" using R_sig by auto
  show "⋀s. s ∈ S⇧⋆ ⟹ (s, s) ∈ T" 
    by (simp add: local.cong)
  show "⋀s t. (s, t) ∈ T ⟹ (t, s) ∈ T"
    by (simp add: conversion_inv local.cong)
  show "⋀s t u. (s, t) ∈ T ⟹ (t, u) ∈ T ⟹ (s, u) ∈ T" 
    by (metis conversion_def local.cong relto_pair.trans_NS_point)
qed

sublocale pt: partition "S⇧⋆" "(S⇧⋆/T)"
  by (simp add: eq.partition_axioms)

sublocale mc: monoid_congruence "S⇧⋆" "append" ε T 
proof(unfold_locales)
  show "⋀s s' t t'. (s, s') ∈ T ⟹ (t, t') ∈ T ⟹ (s @ t, s' @ t') ∈ T"
  proof -
    fix s s' t t'
    assume ss':"(s, s') ∈ T" and tt':"(t, t') ∈ T"
    then show "(s @ t, s' @ t') ∈ T"
    proof -
      have  "(s @ t, s' @ t) ∈ T" using cong sclosed_trancl ss' by blast
      moreover have "(s' @ t, s' @ t') ∈ T" using tt' 
        by (metis append.right_neutral local.cong sclosed_trancl ststepI)
      ultimately show ?thesis using eq.transitive by blast
    qed
  qed
qed

abbreviation quotient_composition (infixl "[⋅]" 70) where "s [⋅] t ≡ mc.quotient_composition s t"

(* Lemma 1 *)
sublocale quot: monoid "S⇧⋆/T" "([⋅])" "eq.Class ε"
  using mc.quotient.monoid_axioms by blast

definition monoid_presentation
  where "monoid_presentation M compos unit ⟷ (∃η. monoid_isomorphism η M compos unit (S⇧⋆/T) ([⋅]) (equivalence.Class (S⇧⋆) T ε))"

definition monoid_finite_presented_by
  where "monoid_finite_presented_by M compos unit ⟷ (∃η. monoid_isomorphism η M compos unit (S⇧⋆/T) ([⋅]) (equivalence.Class (S⇧⋆) T ε) ∧ finite S ∧ finite R)"

definition monoid_presented_by
  where "monoid_presented_by M compos unit = (if (∃η. monoid_isomorphism η M compos unit (S⇧⋆/T) ([⋅]) (equivalence.Class (S⇧⋆) T ε)) then Some (S, R) else None)"

definition "invertible_S ⟷ (∀u ∈ S. mc.quotient.invertible (eq.Class [u]))"

lemma assumes inv_S_imp_inv_S_star:"∀u. u ∈ S ⟶ (∃v. ([u] @ v, ε) ∈ T ∧ (v @ [u], ε) ∈ T ∧ v ∈ S⇧⋆)"
  shows "∀u. u ∈ S⇧⋆ ⟶ (∃v. (u @ v, ε) ∈ T ∧ (v @ u, ε) ∈ T ∧ v ∈ S⇧⋆)"
proof(intro impI allI)
  fix u
  assume "u ∈ S⇧⋆"
  then show "(∃v. (u @ v, ε) ∈ T ∧ (v @ u, ε) ∈ T ∧ v ∈ S⇧⋆)"
  proof(induct)
    case Kleen_star_basis
    then show ?case 
      by auto
  next
    case (Kleene_star_step a b)
    then obtain v where bv:"([b] @ v, ε) ∈ T" and vg:"(v @ [b], ε) ∈ T" and v:"v ∈ S⇧⋆" using assms by auto
    from Kleene_star_step obtain w where aw:"(a @ w, ε) ∈ T" and wa:"(w @ a, ε) ∈ T" and w:"w ∈ S⇧⋆" by auto
    from bv aw have *:"(a @ [b] @ v @ w, ε) ∈ T" 
      by (metis (mono_tags, opaque_lifting) Kleene_star_step.hyps(1) append_Nil append_eq_Cons_conv eq.reflexive eq.transitive mc.cong w)
    moreover have **:"(v @ w @ a @ [b], ε) ∈ T"
    proof -
      have "(w @ a @ [b], ε @ [b]) ∈ T" using wa by (smt (verit, del_insts) Kleene_star.Kleene_star_step 
            Kleene_star_step.hyps(1) Kleene_star_step.hyps(3) append.assoc append.right_neutral 
            empty_append eq.Class_equivalence mc.Class_cong mo.composition_closed mo.unit_closed w)
      moreover have "(v @ w @ a @ [b], v @ ε @ [b]) ∈ T" using calculation mc.cong v by auto
      ultimately show ?thesis using vg eq.transitive by (metis self_append_conv2)
    qed
    show ?case using * ** exI[of _ "v @ w"] by (simp add: v w)
  qed
qed

lemma inv_S_imp_invertible_S: assumes "∀u. u ∈ S ⟶ (∃v. ([u] @ v, ε) ∈ T ∧ (v @ [u], ε) ∈ T ∧ v ∈ S⇧⋆)"
  shows "invertible_S"
proof -
  {
    fix u
    assume uS:"u ∈ S"
    from assms uS obtain v where "([u] @ v, ε) ∈ T" and "(v @ [u], ε) ∈ T" and "v ∈ S⇧⋆" by auto
    hence "invertible_S" unfolding invertible_S_def 
      by auto (smt (verit, ccfv_SIG) Kleene_star.Kleen_star_basis Kleene_star.Kleene_star_step append_eq_Cons_conv 
        eq.Block_self eq.Class_eq eq.block_exists assms mc.natural.commutes_with_composition mc.quotient.invertibleI)
  }
  then show ?thesis using invertible_S_def by blast
qed

lemma inv_compose: assumes "invertible_S"
  shows "⟦a ∈ S⇧⋆; mc.quotient.invertible (eq.Class a); b ∈ S⟧ ⟹ mc.quotient.invertible (eq.Class (a @ [b]))"
proof -
  assume aS:"a ∈ S⇧⋆" and invertible_a:"mc.quotient.invertible (eq.Class a)" and bS:"b ∈ S"
  then show ?thesis
  proof -
    have "mc.quotient.invertible (eq.Class [b])" using assms[unfolded invertible_S_def] bS by auto
    then show ?thesis by (smt (verit, best) Kleene_star.Kleene_star_step aS append_eq_append_conv2 bS 
          eq.natural.map_closed invertible_a mc.Class_commutes_with_composition mc.quotient.composition_invertible 
          mo.right_unit mo.unit_closed same_append_eq)
  qed
qed

lemma invertible_S_imp_invertible_S_star: assumes "invertible_S"
  shows "∀u. u ∈ S⇧⋆  ⟶ mc.quotient.invertible (eq.Class u)" 
proof (intro impI allI)
  fix u
  assume asm:"u ∈ S⇧⋆" 
  then show "mc.quotient.invertible (eq.Class u)"
  proof(induction rule:Kleene_star.induct)
    case Kleen_star_basis
    then show ?case by blast
  next
    case (Kleene_star_step a b)
    then show ?case unfolding mc.quotient.invertible_def using inv_compose[OF assms] by auto
  qed
qed

lemma invertible_S_imp_quotient_invertible: assumes "invertible_S"
  shows "∀u. u ∈ S⇧⋆/T ⟶ mc.quotient.invertible u"
proof (intro impI allI)
  fix u
  assume asm:"u ∈ S⇧⋆/T"
  then show "mc.quotient.invertible u" using invertible_S_imp_invertible_S_star[OF assms] by auto
qed


lemma quotient_monoid_group: assumes "∀u. u ∈ S⇧⋆/T ⟶ mc.quotient.invertible u"
  shows "group (S⇧⋆/T) ([⋅]) (eq.Class ε)" using assms
  by (meson Group_Theory.group.intro Group_Theory.group_axioms_def mc.quotient.monoid_axioms)

(* Lemma 3 *)
lemma quotient_monoid_group_invertible: assumes "∀u. u ∈ S ⟶ (∃v. ([u] @ v, ε) ∈ T ∧ (v @ [u], ε) ∈ T ∧ v ∈ S⇧⋆)"
  shows "group (S⇧⋆/T) ([⋅]) (eq.Class ε)"
proof -
  from assms have "invertible_S" using inv_S_imp_invertible_S by fastforce
  hence "∀u. u ∈ S⇧⋆/T ⟶ mc.quotient.invertible u" using invertible_S_imp_quotient_invertible by auto
  then show ?thesis using quotient_monoid_group by auto
qed

corollary Thue_word_problem_pre: assumes "(u, v) ∈ T"
  shows "eq.Class u = eq.Class v" using assms eq.Class_eq by blast

corollary Thue_word_problem_pre2: assumes "(u, v) ∈ T"
  shows "∀w. w ∈ S⇧⋆/T ⟶ ((u ∈ w ∧ v ∈ w) ∨ (u ∉ w ∧ v ∉ w))" using assms Thue_word_problem_pre
  by (metis eq.Block_self eq.ClassI eq.Class_revI)

(* Theorem 10 *)
theorem Semi_Thue_monoid_word_problem: assumes cr:"CR (ststep R)" and sn:"SN (ststep R)"
  shows "monoid (S⇧⋆/T) ([⋅]) (eq.Class ε)" 
    "(the_NF (ststep R) s = the_NF (ststep R) t) ⟹ eq.Class s = eq.Class t"
    "(the_NF (ststep R) s ≠ the_NF (ststep R) t) ⟹ eq.Class s ≠ eq.Class t"
proof -
  show "monoid (S⇧⋆/T) ([⋅]) (eq.Class ε)" using mc.quotient.monoid_axioms by blast
  assume theNF:"(the_NF (ststep R) s = the_NF (ststep R) t)"
  from cr sn theNF obtain u where su:"(s, u) ∈ (ststep R)⇧*" and tu:"(t, u) ∈ (ststep R)⇧*" and "u ∈ NF(ststep R)"
    unfolding the_NF_def by (metis normalizability_E theNF the_NF)
  hence "(s, t) ∈ (ststep R)⇧* O ((ststep R)¯)⇧*" using su tu by (metis joinI join_def)
  hence "(s, t) ∈ T" using cong 
    by (simp add: CR_imp_conversionIff_join cr join_def)
  then show "eq.Class s = eq.Class t"
    using eq.equivalence_axioms equivalence.Class_eq by metis
next
  assume "(the_NF (ststep R) s ≠ the_NF (ststep R) t)"
  then show "eq.Class s ≠ eq.Class t" 
    using eq.equivalence_axioms by (metis conversion_refl cr eq.Class_equivalence 
        equivalence.left_closed local.cong sn the_NF_conv)
qed

end
