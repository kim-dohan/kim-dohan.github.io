section ‹ShortLex: Length-Lexicographic Order›

theory ShortLex
  imports
   "Abstract-Rewriting.Abstract_Rewriting"

begin

datatype sctxt =
  Hole ("○") |
  More "string" "sctxt" "string"

fun sctxt_apply_string :: "sctxt ⇒ string ⇒ string" ("_⟪_⟫" [900, 0] 900)
  where
    hole: "○⟪s⟫ = s" |
    more: "(More ss1 C ss2)⟪s⟫  = (ss1 @ (C⟪s⟫) @ ss2)"

fun sctxt_compose :: "sctxt ⇒ sctxt ⇒ sctxt" (infixl "∘⇩l" 75)
  where
    "○ ∘⇩l D = D" |
    "(More ss1 C ss2) ∘⇩l D = More ss1 (C ∘⇩l D) ss2"

lemma sctxt_compose[simp]: "(C ∘⇩l D)⟪t⟫ = C⟪D⟪t⟫⟫" by (induct C) simp_all

lemma empty_append [simp]: "l = [] @ l @ []" by simp

definition cancellation_property :: "string rel ⇒ bool" where
  "cancellation_property R ⟷ (∀u v w. (u @ w, v @ w) ∈ R  ⟶ (u, v) ∈ R)"

interpretation sctxt: rel_closure "sctxt_apply_string" "○" "(∘⇩l)" 
  using sctxt_compose by (simp add: rel_closure.intro)

lemma sctxt_closed_strings: assumes sc:"sctxt.closed R"
  and st:"(s, t) ∈ R"
shows "∀u v. (u @ s @ v, u @ t @ v) ∈ R" using assms
proof -
  from sc st obtain C where "(C⟪s⟫, C⟪t⟫) ∈ R" by blast
  then show ?thesis 
    by (metis ShortLex.more sc sctxt.closure.intros sctxt.closure_id sctxt.cop_nil st)
qed

lemma sctxt_app [simp]: "∃bef aft. C⟪l⟫ = bef @ l @ aft" 
  using  sctxt_apply_string.simps
proof (induct "C")
  case Hole
  then show ?case using empty_append by metis
next
  case (More x1 D x2)
  then show ?case 
    by (metis append.assoc)
qed

lemma sctxt_app_step [simp]: "∃bef aft. C⟪l⟫ = (bef @  l @ aft) ∧ C⟪r⟫ = (bef @ r @ aft)"
  using  sctxt_apply_string.simps  sctxt_app 
proof (induct "C")
  case Hole
  then show ?case using empty_append by metis
next
  case (More x1 D x3)
  then show ?case by (metis append_eq_appendI)
qed

lemma sctxt_strings[simp]: assumes C:"(C⟪s⟫ , C⟪t⟫) ∈ R"
  shows "∃u v. (u @ s @ v , u @ t @ v) ∈ R" using assms
  by (metis sctxt_app_step)

lemma sublist_sctxt[simp]: "sublist m l ⟹ ∃C. l = C⟪m⟫"
proof -
  assume "sublist m l"
  hence "∃bef aft. l = bef @ m @ aft" 
    by (simp add: sublist_def)
  then show "∃C. l = C⟪m⟫" 
    by (metis sctxt.cop_nil more)
qed

lemma sublist_sctxt2[simp]: "l = C⟪m⟫ ⟹ sublist m l"
proof -
  assume "l = C⟪m⟫"
  hence "∃bef aft. l = bef @ m @ aft" 
    by (simp add: sublist_def)
  then show "sublist m l"
    by force
qed

locale lex =
  fixes prc :: "char ⇒ char ⇒ bool"
  assumes prc_irrefl: "¬ prc a a"
    and prc_trans: "prc a b ⟹ prc b c ⟹ prc a c" 
begin

abbreviation prc_s (infix "≻⇩p" 50) where "s ≻⇩p t ≡ prc s t"
abbreviation prc_ns (infix "≽⇩p" 50) where "s ≽⇩p t ≡ prc s t ∨ s = t"
abbreviation prc_inv_s (infix "≺⇩p" 50) where "s ≺⇩p t ≡ t ≻⇩p s"
abbreviation prc_inv_ns (infix "≼⇩p" 50) where "s ≼⇩p t ≡ t ≽⇩p s"

fun lexorder :: "string ⇒ string ⇒ bool" where 
  "lexorder s t = (∃u1 u2 u3 ch1 ch2. s = u1 @ [ch1] @ u2 ∧ t = u1 @ [ch2] @ u3 ∧ ch1 ≻⇩p ch2)"

abbreviation lex_s (infix "≻⇩l⇩e⇩x" 50) where "s ≻⇩l⇩e⇩x t ≡ lexorder s t"
abbreviation lex_ns (infix "≽⇩l⇩e⇩x" 50) where "s ≽⇩l⇩e⇩x t ≡ lexorder s t ∨ s = t"

fun co_lexorder :: "string ⇒ string ⇒ bool" where 
  "co_lexorder s t = (¬ (t ≽⇩l⇩e⇩x s))"

abbreviation co_lex_s (infix "≻⇩c⇩o⇩l⇩e⇩x" 50) where "s ≻⇩c⇩o⇩l⇩e⇩x t ≡ co_lexorder s t"

abbreviation "lex_S ≡ {(s,t). s ≻⇩l⇩e⇩x t}"
abbreviation "lex_NS ≡ {(s,t). s ≽⇩l⇩e⇩x t}"
abbreviation "co_lex_S ≡ {(s,t). s ≻⇩c⇩o⇩l⇩e⇩x t}"

(* In the literature, if a string t is a strict prefix of s, then it is often the case that s ≻⇩l⇩e⇩x t. 
  We exclude this case because in that case, ≻⇩l⇩e⇩x is not closed under contexts. 
  For example, given the precedence a > b > c, aaac ≻⇩l⇩e⇩x aaa does not imply baaacb ≻⇩l⇩e⇩x baaab.
  In fact, baaab ≻⇩l⇩e⇩x baaacb. In this example, we have aaac ≻⇩c⇩o⇩l⇩e⇩x aaa instead of aaac ≻⇩l⇩e⇩x aaa *)

lemma lexorder_trans: "lexorder s t ⟹ lexorder t u ⟹ lexorder s u"
proof -
  assume lst:"lexorder s t" and ltu:"lexorder t u"
  from lst have "∃u1 u2 u3 v w. s = u1 @ [v] @ u2 ∧ t = u1 @ [w] @ u3 ∧ v ≻⇩p w"
    unfolding lexorder.simps by metis
  then obtain u1 u2 u3 v w where s:"s = u1 @ [v] @ u2" and t1:"t = u1 @ [w] @ u3"
    and vw:"v ≻⇩p w" by blast
  from ltu have "∃l1 l2 l3 p q. t = l1 @ [p] @ l2 ∧ u = l1 @ [q] @ l3 ∧ p ≻⇩p q" 
    unfolding lexorder.simps by metis
  then obtain l1 l2 l3 p q where t2:"t = l1 @ [p] @ l2" and u:"u = l1 @ [q] @ l3" 
    and pq:"p ≻⇩p q" by blast
  then show ?thesis
  proof(cases "length u1 = length l1")
    case True
    hence "w = p" using t1 t2 by simp
    hence wq:"w ≻⇩p q" using pq by simp
    then show ?thesis unfolding lexorder.simps using True prc_trans 
        s u wq vw t1 t2 append_eq_append_conv by metis
  next
    case False note F1 = this
    then show ?thesis
    proof(cases "length u1 < length l1")
      case True
      hence "take (length u1) l1 = u1" 
        by (metis append_eq_append_conv_if leD t1 t2)
      hence "∃r. l1 = u1 @ [w] @ r" using t1 t2 True
        by (metis Cons_eq_appendI Cons_nth_drop_Suc append_eq_conv_conj append_self_conv2 nth_append_length nth_take)
      then obtain r where l1:"l1 = u1 @ [w] @ r" by auto
      from l1 u have "u = u1 @ [w] @ r @ [q] @ l3" by simp
      with vw show ?thesis
        by (simp add: lexorder.elims vw s u; blast) 
    next
      case False note F2 = this
      from F1 F2 have lu1l1:"length u1 > length l1" by simp
      hence "take (length l1) u1 = l1" using F2
        by (metis append_eq_append_conv_if linorder_le_less_linear t1 t2)
      hence "∃r. u1 = l1 @ [p] @ r" using t2 t1 F1 F2 lu1l1 
      proof -
        have "u1 ! length l1 = p"
          by (metis (no_types) append_Cons append_eq_conv_conj lu1l1 nth_append_length nth_take t1 t2)
        then show ?thesis
          by (metis (no_types) ‹take (length l1) u1 = l1› append_assoc append_eq_conv_conj append_one_prefix lu1l1 prefix_def)
      qed
      then obtain r where "u1 = l1 @ [p] @ r" by auto
      then show ?thesis 
        by (metis append_assoc lexorder.elims(3) pq s u)
    qed
  qed
qed

lemma sctxt_closed_lex_S_pre[simp]: "s ≻⇩l⇩e⇩x t ⟹ (bef @ s @ aft) ≻⇩l⇩e⇩x (bef @ t @ aft)"
proof -
  assume asm:"s ≻⇩l⇩e⇩x t"
  then show ?thesis
  proof - 
    have "∃u1 u2 u3 v w. s = u1 @ [v] @ u2 ∧ t = u1 @ [w] @ u3 ∧ v ≻⇩p w" unfolding lexorder.simps using asm by auto
    then obtain u1 u2 u3 v w where s:"s = u1 @ [v] @ u2" and t:"t = u1 @ [w] @ u3" and vw:"v ≻⇩p w" by auto
    have "∃u1 u2 u3 v w. bef @ s @ aft = u1 @ [v] @ u2 ∧ bef @ t @ aft = u1 @ [w] @ u3 ∧ v ≻⇩p w" 
      by (rule exI[of _ "bef @ u1"], rule exI[of _ "u2 @ aft"], rule exI[of _ "u3 @ aft"], 
          rule exI[of _ "v"], rule exI[of _ "w"], insert s t vw, auto)
    hence "bef @ s @ aft ≻⇩l⇩e⇩x bef @ t @ aft" unfolding lexorder.simps using asm by auto
    then show ?thesis using asm unfolding lexorder.simps by auto
  qed
qed

lemma sctxt_closed_lex_NS_pre[simp]: "s ≽⇩l⇩e⇩x t ⟹ (bef @ s @ aft) ≽⇩l⇩e⇩x (bef @ t @ aft)"
proof -
  assume asm:"s ≽⇩l⇩e⇩x t"
  then show ?thesis
  proof (cases "s = t")
    case True
    hence "bef @ s @ aft = bef @ t @ aft" by simp
    then show ?thesis unfolding lexorder.simps by simp
  next
    case False
    then show ?thesis using sctxt_closed_lex_S_pre asm by auto
  qed
qed

lemma sctxt_closed_lex_S[simp]: "s ≻⇩l⇩e⇩x t ⟹ C⟪s⟫ ≻⇩l⇩e⇩x C⟪t⟫" using sctxt_closed_lex_S_pre 
  by (metis sctxt_app_step)

lemma sctxt_closed_lex_NS[simp]: "s ≽⇩l⇩e⇩x t ⟹ C⟪s⟫ ≽⇩l⇩e⇩x C⟪t⟫" using sctxt_closed_lex_NS_pre 
  by (metis sctxt_app_step)

lemma compat_lex_NS_S_point: "(s, t) ∈ lex_NS ⟹ (t, u) ∈ lex_S ⟹ (s, u) ∈ lex_S"
  using lexorder_trans by blast

lemma compat_lex_S_NS_point: "(s, t) ∈ lex_S ⟹ (t, u) ∈ lex_NS ⟹ (s, u) ∈ lex_S"
  using lexorder_trans by blast

lemma refl_lex_NS:"refl lex_NS" by (simp add: reflI)

lemma irrefl_lex_S:"irrefl lex_S" by (simp add: irreflI prc_irrefl)

lemma co_compat_main:"lex_NS ∩ (co_lex_S)^-1 = {}" by auto

lemma co_compat:"lex_NS ∩ (lex_S)^-1 = {}"
proof(rule ccontr)
  assume asm:"¬ ?thesis"
  hence ctr: "lex_NS ∩ (lex_S)^-1 ≠ {}" by simp
  then obtain a b where ab:"(a, b) ∈ lex_NS" and "(a, b) ∈ (lex_S)^-1" by blast
  hence "(b, a) ∈ lex_S" by auto
  hence "(a, a) ∈ lex_S" using ab compat_lex_NS_S_point by blast
  then show False using irrefl_lex_S
    by (meson irreflD)
qed

theorem co_rewrite_pair_lex_co_lex: "sts_co_rewrite_pair co_lex_S lex_NS"
proof
  show "refl lex_NS" using refl_lex_NS by auto
  show "trans lex_NS" using lexorder_trans 
    by (smt (verit, best) case_prodD case_prodI mem_Collect_eq transI)
  show "sctxt.closed lex_NS" 
    using sctxt.closedI sctxt_closed_lex_NS by force
  show "lex_NS ∩ co_lex_S¯ = {}" using co_compat_main by auto
qed

proposition co_rewrite_pair_lex: "sts_co_rewrite_pair lex_S lex_NS"
proof
  show "refl lex_NS" using refl_lex_NS by auto
  show "trans lex_NS" using lexorder_trans 
    by (smt (verit, best) case_prodD case_prodI mem_Collect_eq transI)
  show "sctxt.closed lex_NS" 
    using sctxt.closedI sctxt_closed_lex_NS by force
  show "lex_NS ∩ lex_S¯ = {}" using co_compat by auto
qed

proposition co_rewrite_pair_extended_lex: "sts_co_rewrite_pair_extended lex_S lex_NS"
proof
  show "refl lex_NS" using refl_lex_NS by auto
  show "trans lex_NS" using lexorder_trans 
    by (smt (verit, best) case_prodD case_prodI mem_Collect_eq transI)
  show "sctxt.closed lex_NS" 
    using sctxt.closedI sctxt_closed_lex_NS by force
  show "lex_NS ∩ lex_S¯ = {}" using co_compat by auto
  show "sctxt.closed (lex_S¯)" using sctxt.closedI sctxt_closed_lex_S by force
qed

end

fun lenorder :: "string ⇒ string ⇒ bool" where 
  "lenorder s t = (length s > length t)" 

abbreviation len_s (infix "≻⇩l⇩e⇩n" 50) where "s ≻⇩l⇩e⇩n t ≡ lenorder s t"
abbreviation len_ns (infix "≽⇩l⇩e⇩n" 50) where "s ≽⇩l⇩e⇩n t ≡ lenorder s t ∨ s = t"

abbreviation "len_S ≡ {(s,t). s ≻⇩l⇩e⇩n t}"
abbreviation "len_NS ≡ {(s,t). s ≽⇩l⇩e⇩n t}"

lemma lenorder_trans: "lenorder s t ⟹ lenorder t u ⟹ lenorder s u"
proof -
  assume lst:"lenorder s t" and ltu:"lenorder t u"
  then show ?thesis by auto
qed

lemma sctxt_len_closed[simp]: "s ≽⇩l⇩e⇩n t ⟹ (bef @ s @ aft) ≽⇩l⇩e⇩n (bef @ t @ aft)"
proof -
  assume asm:"s ≽⇩l⇩e⇩n t"
  then show ?thesis by (cases "s = t", auto)
qed

lemma sctxt_len_closed_S[simp]: "s ≽⇩l⇩e⇩n t ⟹ C⟪s⟫ ≽⇩l⇩e⇩n C⟪t⟫" using sctxt_len_closed 
  sctxt_app_step[of C s t] by force

lemma compat_len_NS_S_point: "(s, t) ∈ len_NS ⟹ (t, u) ∈ len_S ⟹ (s, u) ∈ len_S"
  using lenorder_trans by blast

lemma compat_len_S_NS_point: "(s, t) ∈ len_S ⟹ (t, u) ∈ len_NS ⟹ (s, u) ∈ len_S"
  using lenorder_trans by blast

lemma refl_len_NS:"refl len_NS" by (simp add: reflI)

lemma irrefl_len_S:"irrefl len_S" by (simp add: irreflI prc_irrefl)

  then obtain a b where ab:"(a, b) ∈ len_NS" and "(a, b) ∈ (len_S)^-1" by blast

  then show False using irrefl_len_S

(* Compare lengths first and then use lexicographic order *)

fun shortlex :: "string ⇒ string ⇒ bool"
  where "shortlex str1 str2 = (if lenorder str1 str2 then True
          else if lenorder str2 str1 then False else lexorder str1 str2)"

abbreviation shortlex_s (infix "≻⇩s⇩l" 50) where "s ≻⇩s⇩l t ≡ shortlex s t"
abbreviation shortlex_ns (infix "≽⇩s⇩l" 50) where "s ≽⇩s⇩l t ≡ shortlex s t ∨ s = t"

abbreviation shortlex_inv_s (infix "≺⇩s⇩l" 50) where "s ≺⇩s⇩l t ≡ t ≻⇩s⇩l s"

abbreviation "shortlex_S ≡ {(s,t). s ≻⇩s⇩l t}"
abbreviation "shortlex_NS ≡ {(s,t). s ≽⇩s⇩l t}"

fun co_shortlex :: "string ⇒ string ⇒ bool" where 
  "co_shortlex s t = (¬ (t ≽⇩s⇩l s))" 

abbreviation co_shortlex_s (infix "≻⇩c⇩o⇩s⇩l" 50) where "s ≻⇩c⇩o⇩s⇩l t ≡ co_shortlex s t"
abbreviation "co_shortlex_S ≡ {(s,t). s ≻⇩c⇩o⇩s⇩l t}"

lemma shortlex_trans[simp]: "s ≻⇩s⇩l t ⟹ t ≻⇩s⇩l u ⟹ s ≻⇩s⇩l u"
proof -
  assume st:"s ≻⇩s⇩l t" and tu:"t ≻⇩s⇩l u"
  then show "s ≻⇩s⇩l u"
  proof(cases "length s = length t")
    case False
    hence "length s > length u" using st tu  
        using linorder_neqE_nat by fastforce        
    then show ?thesis unfolding shortlex.simps by simp
  next
    case True note T1 = this
    then show ?thesis
    proof(cases "length t = length u")
      case False
      hence "length s > length u" using st tu  
        using linorder_neqE_nat by fastforce
      then show ?thesis unfolding shortlex.simps by simp
    next
      case True note T2 = this
      hence lenst:"length s = length t" and lentu:"length t = length u" 
        using T1 T2 by auto
      from st lenst have lst:"s ≻⇩l⇩e⇩x t" unfolding shortlex.simps by auto 
      from tu lentu have ltu:"t ≻⇩l⇩e⇩x u" unfolding shortlex.simps by auto
      from lexorder_trans[OF lst ltu] have lsu:"s ≻⇩l⇩e⇩x u" by auto
      from lenst lentu have lensu:"length s = length u" by auto
      from lsu lensu show ?thesis unfolding shortlex.simps by simp
    qed
  qed
qed

lemma sctxt_shortlex_closed_S_pre[simp]: "s ≻⇩s⇩l t ⟹ (bef @ s @ aft)≻⇩s⇩l (bef @ t @ aft)"
proof -
  assume asm:"s ≻⇩s⇩l t"
  then show ?thesis
  proof(cases "lenorder s t")
    case True
    then show ?thesis by simp
  next
    case False
    hence "lexorder s t"
      by (metis asm shortlex.elims(2))
    then show ?thesis using asm sctxt_closed_lex_S_pre by auto 
  qed
qed

lemma sctxt_shortlex_closed_NS_pre[simp]: "s ≽⇩s⇩l t ⟹ (bef @ s @ aft) ≽⇩s⇩l (bef @ t @ aft)"
proof -
  assume st:"s ≽⇩s⇩l t"
  then show ?thesis
  proof (cases "s = t")
    case True
    then show ?thesis by fastforce
  next
    case False
    hence "s ≻⇩s⇩l t" using st by auto
    hence "(bef @ s @ aft) ≻⇩s⇩l (bef @ t @ aft)" using sctxt_shortlex_closed_S_pre by metis
    then show ?thesis by auto
  qed
qed

lemma sctxt_shortlex_closed_S[simp]: "s ≻⇩s⇩l t ⟹ C⟪s⟫ ≻⇩s⇩l C⟪t⟫" using sctxt_shortlex_closed_S_pre
  sctxt_app_step[of C s t] by metis

lemma sctxt_shortlex_closed_inv_S[simp]: "s ≺⇩s⇩l t ⟹ C⟪s⟫ ≺⇩s⇩l C⟪t⟫" using sctxt_shortlex_closed_S_pre
  sctxt_app_step[of C s t] by metis 

lemma sctxt_shortlex_closed_NS[simp]: "s ≽⇩s⇩l t ⟹ C⟪s⟫ ≽⇩s⇩l C⟪t⟫" using sctxt_shortlex_closed_NS_pre
  sctxt_app_step[of C s t] by metis 

lemma compat_shortlex_NS_S_point[simp]: "(s, t) ∈ shortlex_NS ⟹ (t, u) ∈ shortlex_S ⟹ (s, u) ∈ shortlex_S"
  using shortlex_trans by blast

lemma compat_shortlex_S_NS_point[simp]: "(s, t) ∈ shortlex_S ⟹ (t, u) ∈ shortlex_NS ⟹ (s, u) ∈ shortlex_S"
  using shortlex_trans by blast

lemma compat_shortlex_NS_S[simp]:"shortlex_NS O shortlex_S ⊆ shortlex_S"
proof
  fix x y
  assume "(x, y) ∈ shortlex_NS O shortlex_S"
  then obtain z where "(x, z) ∈ shortlex_NS" and "(z, y) ∈ shortlex_S" by auto
  then show "(x, y) ∈ shortlex_S" using compat_shortlex_NS_S_point[of x z y] by auto+
qed

lemma compat_shortlex_S_NS[simp]:"shortlex_S O shortlex_NS ⊆ shortlex_S"
proof
  fix x y
  assume "(x, y) ∈ shortlex_S O shortlex_NS"
  then obtain z where "(x, z) ∈ shortlex_S" and "(z, y) ∈ shortlex_NS" by auto
  then show "(x, y) ∈ shortlex_S" using compat_shortlex_S_NS_point[of x z y] by auto+

lemma sl_left_append: assumes "∀(u, v) ∈ S. t ≻⇩s⇩l u @ w  ∧ t ≻⇩s⇩l v @ w "
  shows "∀(u, v) ∈ S. bef @ t ≻⇩s⇩l u @ w  ∧ bef @ t ≻⇩s⇩l v @ w "
proof -
  {
    fix u v

    hence "bef @ t ≻⇩s⇩l u @ w  ∧ bef @ t ≻⇩s⇩l v @ w" using sl_left_append_pre[of t "u @ w" bef]

lemma sl_cancellation_property: assumes "(u @ w) ≽⇩s⇩l (v @ w)"
  shows "u ≽⇩s⇩l v" using assms 
proof(cases "length u = length v")
  case True
  note T1 = this
  hence lg:"(u @ w) ≽⇩l⇩e⇩x (v @ w)" using assms by auto
  then show ?thesis
  proof(cases "u = v")
    case True
    then show ?thesis by auto
  next
    case False
    hence "u ≠ [] ∧ v ≠ []" using T1 by auto
    hence "(u @ w) ≻⇩l⇩e⇩x (v @ w)" using lg False by auto
    hence "(∃u1 u2 u3 q r. u @ w = u1 @ [q] @ u2 ∧ v @ w = u1 @ [r] @ u3 ∧ q ≻⇩p r)" by auto
    then obtain u1 u2 u3 q r where uw:"u @ w = u1 @ [q] @ u2" and vw:"v @ w = u1 @ [r] @ u3" and qr:"q ≻⇩p r" by auto
    hence nqr:"q ≠ r" using prc_irrefl by blast
    hence "length (u1 @ [q]) = length (u1 @ [r])" by auto
    have *:"length u ≥ length (u1 @ [q])"
    proof(rule ccontr)
      assume "¬ ?thesis"
      hence "length (u1 @ [q]) > length u" by auto
      then show False using nqr T1 
        by (auto, metis False append_eq_append_conv_if less_Suc_eq_le uw vw)
    qed
    have **:"length v ≥ length (u1 @ [r])" 
    proof(rule ccontr)
      assume "¬ ?thesis"
      hence "length (u1 @ [r]) > length v" by auto
      then show False using nqr T1
        by (auto, metis False append_eq_append_conv_if less_Suc_eq_le uw vw)
    qed
    have "∃z1. u = u1 @ [q] @ z1" using * uw
      by (auto, metis Suc_le_eq append_Cons_nth_middle append_eq_append_conv_if id_take_nth_drop not_less_eq_eq nth_append)
    moreover have "∃z2.  v = u1 @ [r] @ z2" using ** vw
      by (auto, metis Suc_le_eq append_Cons_nth_middle append_eq_append_conv_if id_take_nth_drop not_less_eq_eq nth_append)
    moreover have "length (u @ w) = length (v @ w)" using T1 by auto
    ultimately have "∃u4 u5. u2 = u4 @ w ∧ u3 = u5 @ w" using uw vw by force
    hence "u ≻⇩l⇩e⇩x v" using T1 False qr uw vw by fastforce
    then show ?thesis using T1 by auto 
  qed
next
  case False
  then show ?thesis 
    using assms by auto
qed

locale shortlex_total = lex prc
  for prc :: "char ⇒ char ⇒ bool" +
  fixes prc_w :: "char ⇒ nat"
   and  prc_max :: nat
  assumes prc_asym: "prc a b ⟹ ¬ prc b a"
    and prc_total: "a ≠ b  ⟹ prc a b ∨ prc b a"
    and prc_weight: "prc a b ⟹ prc_w a > prc_w b"
    and prc_max_weight: "prc_max > prc_w a"
    and prc_finite: "finite {(a, b). prc a b}"
    and prc_SN: "SN {(a, b). prc a b}"
begin

abbreviation "length_preserving_lex_S ≡ {(s,t). s ≻⇩l⇩e⇩x t ∧ length s = length t}"

lemma minimal_shortlex[simp]:"∀t. ¬ [] ≻⇩s⇩l t" 
  unfolding shortlex.simps by simp

lemma lex_tot: assumes snt:"s ≠ t"
  and len:"length s = length t"
shows "lexorder s t ∨ lexorder t s"
proof -
  from snt len
  have "∃u1 u2 u3 v w. s = u1 @ [v] @ u2 ∧ t = u1 @ [w] @ u3 ∧ v ≠ w" 
    using same_length_different by fastforce
  then obtain u1 u2 u3 v w where s:"s = u1 @ [v] @ u2" and t:"t = u1 @ [w] @ u3" and vnw:"v ≠ w" by auto
  from vnw have "v ≻⇩p w ∨ w ≻⇩p v" using prc_total by auto
  then show ?thesis
  proof
    assume "v ≻⇩p w"
    hence "lexorder s t"  using lexorder.simps s t vnw len by force
    then show ?thesis by blast
  next
    assume "w ≻⇩p v"
    hence "lexorder t s" using lexorder.simps len s t vnw by force 
    then show ?thesis by blast
  qed
qed

lemma shortlex_total: assumes "s ≠ t"
  shows "s ≻⇩s⇩l t ∨ t≻⇩s⇩l s"
proof -
  from assms have "(length s > length t ∨ length t > length s) ∨ (length s = length t ∧ 
    s ≠ t)" using nat_neq_iff by blast
  then show ?thesis
  proof
    assume "length s > length t ∨ length t > length s"
    then show ?thesis
    proof
      assume "length s > length t"
      hence "s≻⇩s⇩l t" using shortlex.simps by simp
      then show ?thesis by blast
    next
      assume "length t > length s"
      hence "t ≻⇩s⇩l s" using shortlex.simps by simp
      then show ?thesis by blast
    qed
  next
    assume "length s = length t ∧ s ≠ t"
    hence len:"length s = length t" and snt:"s ≠ t" by auto
    from lex_tot[OF snt len]
    have "lexorder s t ∨ lexorder t s" by simp
    then show ?thesis
    proof
      assume "lexorder s t"
      then show ?thesis using len by fastforce
    next
      assume "lexorder t s"
      then show ?thesis using len by fastforce
    qed 
  qed
qed

lemma shortlex_comb[simp]: "shortlex_S = length_preserving_lex_S ∪ len_S"

proof(safe, goal_cases)

  case (2 a b)

    by (meson lenorder.simps linorder_neqE_nat)
next
  case (3 a b u1 u2 u3 v w)
  then show ?case

qed auto 

lemma SN_len_S: "SN len_S"
proof -
  let ?A = "{(a::nat, b). a > b}"
  from SN_nat_gt have SNA:"SN ?A" by simp
  then show ?thesis
  proof -
    {
      fix s :: string
      let ?S = "λx. SN_on len_S {x}"
      have "?S s" unfolding SN_on_def
      proof (rule notI)
        assume "∃f. f 0 ∈ {s} ∧ (∀i. (f i, f (Suc i)) ∈ {(s, t). lenorder s t})" 
        then obtain S where s:"S 0 = s" and chainl:"chain len_S S" by auto
        let ?T = "λi. length (S i)"
        from chainl have init:"?T 0 = length s" and chainT:"chain ?A ?T"  
          using s case_prod_conv chainl by auto
        from init chainT have "¬ SN_on ?A {length s}"
          by (auto simp add:SN_def SN_on_def)
        hence *:"¬ SN ?A" unfolding SN_def by auto
        then show False using SNA using * by blast
      qed
    } 
    from SN_I[OF this] show ?thesis by blast
  qed
qed

fun lex_ext_string :: "('a ⇒ 'a ⇒ bool) ⇒ 'a list ⇒ 'a list ⇒ bool"
  where "lex_ext_string f [] [] = False" |
    "lex_ext_string f (_ # _) [] = False" |
    "lex_ext_string f [] (_ # _) = False" |
    "lex_ext_string f (a # as) (b # bs) =
      (if length (a # as) ≠ length (b # bs) then False else if f a b then True
      else if a = b then lex_ext_string f as bs
      else False)"

lemma lex_ext_string_irrefl[simp]:"¬ lex_ext_string prc s s"

fun string_prc_weight :: "string ⇒ nat"
  where "string_prc_weight [] = 0" |
    "string_prc_weight (x # xs) = (prc_w x * (prc_max + 2)^(length xs + 1)) + string_prc_weight xs"

lemma prc_max_pow: "(prc_max + 2)^(length xs + 1) > string_prc_weight xs" using prc_max_weight
proof(induct xs)
  case Nil
  then show ?case using bot_nat_0.not_eq_extremum by force
next
  case (Cons c ss)
  hence "(prc_max + 2) ^ (length ss + 1) > string_prc_weight ss" by auto
  moreover have "string_prc_weight (c # ss) = (prc_w c * (prc_max + 2)^(length ss + 1)) + string_prc_weight ss" by auto
  moreover have "... < prc_w c * (prc_max + 2)^(length ss + 1) + (prc_max + 2)^(length ss + 1)" using calculation(1) by force
  moreover have "... < (prc_w c + 2) * (prc_max + 2)^(length ss + 1)" using prc_max_weight by fastforce
  moreover have "... < (prc_max + 2) * (prc_max + 2)^(length ss + 1)" using prc_max_weight by auto
  ultimately show ?case using prc_max_weight by (simp add: add.commute add_mono_thms_linordered_field(5) trans_less_add2)
qed

lemma lex_ext_string_prc_weight: assumes "lex_ext_string prc s t"
  shows "string_prc_weight s > string_prc_weight t" using assms
proof(induct s arbitrary: t)
  case Nil
  then show ?case using lex_ext_string.elims(2) by blast
next
  case (Cons a ss)
  then obtain b ts where t:"t = b # ts" using lex_ext_string.elims(2) by blast
  from Cons(2) have "a ≻⇩p b ∨ (a = b ∧ lex_ext_string (≻⇩p) ss ts)" using t 
    by (auto, metis, argo)
  then show ?case
  proof
    assume asm:"a ≻⇩p b"
    have "string_prc_weight (a # ss) = (prc_w a * (prc_max + 2)^(length ss + 1)) + string_prc_weight ss" by auto
    moreover have "string_prc_weight (b # ts) = (prc_w b * (prc_max + 2)^(length ts + 1)) + string_prc_weight ts" by auto
    moreover have "prc_w a > prc_w b" using asm by (simp add: prc_weight)
    moreover have "length ss = length ts" by (metis Cons.prems add_right_imp_eq list.size(4) 
          shortlex_total.len_same_lex_ext_string shortlex_total_axioms t)
    moreover have "(prc_w a * (prc_max + 2)^(length ss + 1)) - (prc_w b * (prc_max +2 )^(length ts + 1)) ≥ (prc_max + 2)^(length ts + 1)"
      by (metis One_nat_def calculation(3) calculation(4) diff_is_0_eq diff_mult_distrib less_eq_Suc_le mult.commute mult.right_neutral mult_eq_0_iff zero_less_diff)
    ultimately show ?thesis using prc_max_weight t prc_max_pow[of ts] asm by auto
  next
    assume asm:"a = b ∧ lex_ext_string (≻⇩p) ss ts"
    hence "string_prc_weight ss > string_prc_weight ts" using Cons(1) by auto
    then show ?thesis using t asm len_same_lex_ext_string by auto
  qed
qed
