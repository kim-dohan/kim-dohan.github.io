(* Definition 11 *)
inductive sts_compl_step :: "sts × sts ⇒ sts × sts ⇒ bool" (infix "⊢⇩S⇩R" 55)
where
  deduce:"(E, R) ⊢⇩S⇩R (E ∪ {(s @ u3, u1 @ t)}, R)"
    if "(u1 @ u2, s) ∈ R" and "(u2 @ u3, t) ∈ R" and "u2 ≠ []"
| simplifyl:"(E ∪ {(u1 @ u2 @ u3, s)}, R) ⊢⇩S⇩R (E ∪ {(u1 @ t @ u3, s)}, R)"
    if "(u2, t) ∈ R"
| simplifyr:"(E ∪ {(s, u1 @ u2 @ u3)}, R) ⊢⇩S⇩R (E ∪ {(s, u1 @ t @ u3)}, R)"
    if "(u2, t) ∈ R"
| orientl: "(E ∪ {(s, t)}, R) ⊢⇩S⇩R (E, R ∪ {(s, t)})"
    if "s ≻⇩s⇩l t"
| orientr: "(E ∪ {(s, t)}, R) ⊢⇩S⇩R (E, R ∪ {(t, s)})"
    if "t ≻⇩s⇩l s"
| collapse: "(E, R ∪ {(u1 @ u2 @ u3, s)}) ⊢⇩S⇩R (E ∪ {(u1 @ t @ u3, s)}, R)"
    if "(u2, t) ∈ R"
| compose: "(E, R ∪ {(s, u1 @ u2 @ u3)}) ⊢⇩S⇩R (E, R ∪ {(s, u1 @ t @ u3)})"
    if "(u2, t) ∈ R"
| delete: "(E ∪ {(s, s)}, R) ⊢⇩S⇩R (E, R)"

lemmas stscompI = sts_compl_step.intros [intro]
lemmas stscompE = sts_compl_step.cases [elim]

declare shortlex.simps [simp del]

lemma deduce_cr:
  assumes u1u2s:"(u1 @ u2, s) ∈ R" and u2u3t:"(u2 @ u3, t) ∈ R" and "u2 ≠ []"
    and "(E, R) ⊢⇩S⇩R (E ∪ {(s @ u3, u1 @ t)}, R)"
  shows "(s @ u3, u1 @ t) ∈ sts_critical_pairs R"
proof -
  let ?x = "u1"
  let ?u = "u2 @ u3"
  let ?v = "t"
  let ?u' = "u1 @ u2"
  let ?y = "u3"
  let ?v' = "s"
  have leq:"?x @ ?u = ?u' @ ?y" by auto
  have len:"length ?x < length ?u'" using assms by auto
  have "(?v'@ ?y, ?x @ ?u, ?x @ ?v) ∈ sts_critical_peaks R" 
    using len u1u2s u2u3t by auto
  hence "(s @ u3, u1 @ u2 @ u3, u1 @ t) ∈ sts_critical_peaks R" by auto
  with sts_critical_peak_pairs[of "s @ u3" "u1 @ u2 @ u3" "u1 @ t"]
  show ?thesis by auto
qed

lemma SR_subset_pre:
  assumes "(E, R) ⊢⇩S⇩R (E', R')"
  shows "ststep (E ∪ R) ⊆ (ststep R')⇧= O (ststep (E' ∪ R'))⇧= O ((ststep R')¯)⇧="
proof -
  have "E ∪ R ⊆ (ststep R')⇧= O (ststep (E' ∪ R'))⇧= O ((ststep R')¯)⇧=" (is "?L ⊆ ?R")
  proof
    fix s t
    assume asm: "(s, t) ∈ ?L"
    from assms show "(s, t) ∈ ?R"
    proof (cases)
      case (deduce u1 u2 s u3 t)
      with sts_deduce_sub[of E' E s u3 u1 t R' R u2]
      have "E ∪ R ⊆ E' ∪ R'" by auto
      then show ?thesis using asm by auto  
    next
      case (simplifyl u2 t T u1 u3 s)
      with sts_simplifyl_sub[of E' E u1 u2 u3 s t R' R]
      have "E ∪ R ⊆ E' ∪ R' ∪ ststep R' O ststep E' ∪ ststep E' O (ststep R')¯" by auto
      then show ?thesis using asm by auto
    next
      case (simplifyr u2 t T s u1 u3)
      with sts_simplifyr_sub[of E' E s u1 u2 u3 t R' R]
      have "E ∪ R ⊆ E' ∪ R' ∪ ststep R' O ststep E' ∪ ststep E' O (ststep R')¯" by auto
      then show ?thesis using asm by auto
    next
      case (orientl s t)
      then show ?thesis using sts_orientl_sub asm by auto
    next
      case (orientr t s)
      then show ?thesis using sts_orientr_sub asm by auto
    next
      case (collapse u2 t u1 u3 s)
      with sts_collapse_sub[of E' E u1 t u3 s R' R u2]
      have "E ∪ R ⊆ E' ∪ R' ∪ (ststep R' O ststep E')" by auto
      then show ?thesis using asm by auto 
    next
      case (compose u2 t T s u1 u3)
      with sts_compose_sub[of E' E R' R s u1 u2 u3 t]
      have "E ∪ R ⊆ E' ∪ R' ∪ ststep R' O (ststep R')¯" by auto
      then show ?thesis using asm by auto
    next
      case (delete s)
      then show ?thesis using sts_delete_sub asm by auto
    qed
  qed
  from ststep_mono [OF this]
  have "ststep (E ∪ R) ⊆ ststep ((ststep R')⇧= O (ststep (E' ∪ R'))⇧= O ((ststep R')¯)⇧=)" by auto
  moreover have "ststep (E ∪ R) ⊆ ststep ((ststep R')⇧= O (ststep (E' ∪ R'))⇧= O ((ststep (R'¯))⇧=))"
    using ststep_converse[of R'] calculation by auto
  moreover have "... ⊆ ststep ((ststep R')⇧=) O ststep ((ststep (E' ∪ R'))⇧=) O ststep(((ststep (R'¯))⇧=))"
    using  sctxt.closed_comp sctxt.closed_converse sctxt_closed_ststep sctxt_closed_strings ststepE
    by (auto, metis+)
  ultimately show ?thesis by auto
qed
  
lemma SR_subset:
  assumes "(E, R) ⊢⇩S⇩R (E', R')"
  shows "ststep (E ∪ R) ⊆ ((ststep (E' ∪ R'))⇧↔)⇧*"
proof -
  have "E ∪ R ⊆ (ststep (E' ∪ R'))⇧= O (ststep (E' ∪ (R')¯))⇧=" (is "?L ⊆ ?R")
  proof
    fix s t
    assume asm: "(s, t) ∈ ?L"
    from assms show "(s, t) ∈ ?R"
      by (cases, insert asm, blast+)
  qed
  moreover have "... ⊆ ((ststep (E' ∪ R'))⇧↔)⇧*"
    by (auto, goal_cases, simp add: converse_rtrancl_into_rtrancl in_rtrancl_UnI,meson UnCI r_into_rtrancl 
        rtrancl.rtrancl_into_rtrancl,meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI, 
        meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI)
  from ststep_mono [OF this] show ?thesis 
    by (metis (no_types, lifting) calculation conversion_def order_subst2 sclosed_trancl ststep_mono)
qed

lemma SR_subset':
  assumes "(E, R) ⊢⇩S⇩R (E', R')"
  shows "ststep (E' ∪ R') ⊆ ((ststep (E ∪ R))⇧↔)⇧*" 
proof -
  have "E' ∪ R' ⊆ ((ststep (E ∪ R))⇧↔)⇧= O (ststep (E ∪ R))⇧=" (is "?L ⊆ ?R") 
  proof
    fix s t
    assume asm: "(s, t) ∈ ?L"
    from assms show "(s, t) ∈ ?R"
    proof(cases)
      case (deduce u1 u2 v u3 w)
      then show ?thesis using asm sts_deduce[of E' E v u3 u1 w R' R u2] by auto
    next
      case (simplifyl u2 t T u1 u3 s)
      with asm sts_simplifyl[of E' E u1 u2 u3 s t R' R] 
      have "E' ∪ R' ⊆ E ∪ R ∪ (ststep R)¯ O ststep E ∪ ststep E O ststep R" by auto
      then show ?thesis using asm by auto
    next
      case (simplifyr u2 t T s u1 u3)
      with asm sts_simplifyr[of E' E s u1 u2 u3 t R' R]
      have "E' ∪ R' ⊆ E ∪ R ∪ ((ststep R)¯ O ststep E) ∪ (ststep E O ststep R)" by auto
      then show ?thesis using asm by auto
    next
      case (orientl s t)
      then show ?thesis using sts_orientl_sub[of E' E s t R' R] using asm by force
    next
      case (orientr t s)
      then show ?thesis using sts_orientr_sub[of E' E s t R' R] using asm by force
    next
      case (collapse u2 t u1 u3 s)
      with asm sts_collapse[of E' E R' R u1 u2 u3 s t]
      have "E' ∪ R' ⊆ E ∪ R ∪ (ststep R)¯ O ststep R ∪ ststep R O ststep R" by auto
      then show ?thesis using asm by auto
    next
      case (compose u2 t T s u1 u3)
      with asm sts_compose[of E' E R' R s u1 u2 u3 t]
      have "E' ∪ R' ⊆ E ∪ R ∪ (ststep R)¯ O ststep R ∪ ststep R O ststep R" by auto
      then show ?thesis using asm by auto
    next
      case (delete s)
      with asm sts_delete[of E' E s R' R]
      have "E' ∪ R' ⊆ E ∪ R" by auto
      then show ?thesis using asm by auto
    qed
  qed
  moreover have "... ⊆ ((ststep (E ∪ R))⇧↔)⇧*"
  proof(auto, goal_cases)
    case (1 x y z)
    then show ?case by (simp add: converse_rtrancl_into_rtrancl in_rtrancl_UnI)
  next
    case (2 x y z)
    then show ?case by (meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point)
  next
    case (3 x y z)
    then show ?case by (meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI)
  next
    case (4 x y z)
    then show ?case by (meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI)
  next
    case (5 x y z)
    then show ?case by (meson UnCI converse_rtrancl_into_rtrancl r_into_rtrancl)
  next
    case (6 x y z)
    then show ?case by (simp add: converse_rtrancl_into_rtrancl in_rtrancl_UnI)
  next
    case (7 x y z)
    then show ?case by (meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI)
  next
    case (8 x y z)
    then show ?case by (meson in_rtrancl_UnI r_into_rtrancl relto_pair.trans_NS_point rtrancl_converseI)
  qed   
  from ststep_mono [OF this] show ?thesis 
    by (metis (no_types, lifting) calculation conversion_def order_subst2 sclosed_trancl ststep_mono)
qed

lemma sts_compl_conversion:
  assumes "(E, R) ⊢⇩S⇩R (E', R')"
  shows "(ststep (E ∪ R))⇧↔⇧* = (ststep (E' ∪ R'))⇧↔⇧*" (is "?lhs = ?rhs")
  using conversion_rtrancl[of "E ∪ R"] SR_subset[of E R E' R'] SR_subset'[of E R E' R']
    by (metis assms conversion_conversion_idemp conversion_def conversion_mono dual_order.eq_iff)

lemma sts_rtrancl_conversion:
  assumes "(⊢⇩S⇩R)⇧*⇧* (E, R) (E', R')"
  shows "(ststep (E ∪ R))⇧↔⇧* = (ststep (E' ∪ R'))⇧↔⇧*"
  using assms
  by (induct "(E, R)" "(E', R')" arbitrary: E' R')
    (force dest: sts_compl_conversion)+

lemma sts_rtrancl_subset_shortlex:
  assumes "(⊢⇩S⇩R)⇧*⇧* (E, R) (E', R')" and R:"R ⊆ shortlex_S"
  shows "R' ⊆ shortlex_S" using assms
proof (induction "(E, R)" "(E', R')" arbitrary: E' R') 
  case (rtrancl_into_rtrancl pair) 
  note IH = this
  then obtain Ex and Rx where pair: "pair = (Ex, Rx)" by force
  hence "(⊢⇩S⇩R)⇧*⇧* (E, R) (Ex, Rx)" using rtrancl_into_rtrancl.hyps(1) by auto
  have *:"(Ex, Rx) ⊢⇩S⇩R (E', R')" using rtrancl_into_rtrancl.hyps(3) pair by auto 
  have Rx:"Rx ⊆ shortlex_S" 
    by (rule IH(2)[of Ex Rx], simp add:pair, insert R, blast)
  from * show ?case
  proof(cases, insert Rx)
    case (compose u2 t T s u1 u3)
    hence "s ≻⇩s⇩l u1 @ u2 @ u3" using Rx by blast
    moreover have "u2 ≻⇩s⇩l t" using compose using Rx by blast
    ultimately have "s ≻⇩s⇩l u1 @ t @ u3" using sctxt_shortlex_closed_S_pre shortlex_trans by blast
    then show ?thesis using Rx compose by blast
  qed (blast+)
qed

lemma ststep_subset_S: "R ⊆ shortlex_S ⟹ ststep R ⊆ shortlex_S"
proof (rule subrelI)
  fix s t
  assume R:"R ⊆ shortlex_S"
    and st:"(s, t) ∈ ststep R"
  then show "(s, t) ∈ shortlex_S"
  proof -
    from st have "∃l r bef aft. (l, r) ∈ R ∧ s = bef @ l @ aft ∧ t = bef @ r @ aft" by blast
    then obtain l r bef aft where lr:"(l, r) ∈ R" and s:"s = bef @ l @ aft" and t:"t = bef @ r @ aft" by blast
    from sctxt_shortlex_closed_S_pre[of l r bef aft] 
    show ?thesis using lr s t R by blast
  qed
qed

lemma sts_finite_run_conv:
  assumes R0:"R 0 = {}" and En:"E n = {}"
    and run: "∀i < n. (E i, R i) ⊢⇩S⇩R (E (Suc i), R (Suc i))"
  shows "(ststep (E 0))⇧↔⇧* = (ststep (R n))⇧↔⇧*"
proof -
  have *: "⋀i. i ≤ n ⟹ (⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)"
  proof -
    fix i
    assume "i ≤ n"
    then show "(⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)" using run 
      by (induct i, insert Suc_le_eq, auto) 
  qed
  from sts_rtrancl_conversion [OF * [of n]]
  show "(ststep (E 0))⇧↔⇧* = (ststep (R n))⇧↔⇧*" using R0 En by simp
qed

lemma sts_finite_run_SN:
  assumes R0:"R 0 = {}" and En:"E n = {}"
    and run: "∀i < n. (E i, R i) ⊢⇩S⇩R (E (Suc i), R (Suc i))"
  shows "SN (ststep (R n))"
proof -
  have *: "⋀i. i ≤ n ⟹ (⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)"
  proof -
    fix i
    assume "i ≤ n"
    then show "(⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)" using run 
      by (induct i, insert Suc_le_eq, auto) 
  qed
  from * [THEN sts_rtrancl_subset_shortlex] and R0
  have "⋀i. i ≤ n ⟹ R i ⊆ shortlex_S" by simp
  then show "SN (ststep (R n))" 
    by (meson SN_subset le_refl shortlex_SN ststep_subset_S)
qed

text ‹A (non-empty) peak that is joinable or a critical pair.›
definition jcp_peak :: "sts ⇒ string ⇒ sts"
  where
    "jcp_peak R s = {(t, u).
      (s, t) ∈ (ststep R)⇧+ ∧ (s, u) ∈ (ststep R)⇧+ ∧
      ((t, u) ∈ (ststep R)⇧↓ ∨ (t, u) ∈ (ststep (sts_critical_pairs R))⇧↔)}"

lemma scp_subset_imp:
  assumes fair:"sts_critical_pairs R ⊆ (⋃ i ≤ n. (ststep (E i))⇧↔)"
  shows "(ststep (sts_critical_pairs R))⇧↔ ⊆ (⋃ i ≤ n. (ststep (E i))⇧↔)" (is "_ ⊆ ?rhs")
proof
  fix s t
  assume "(s, t) ∈ (ststep (sts_critical_pairs R))⇧↔"
  hence "(s, t) ∈ ststep (sts_critical_pairs R) ∨ (t, s) ∈ ststep (sts_critical_pairs R)" by auto
  then show "(s, t) ∈ ?rhs"
  proof 
    assume asm:"(s, t) ∈ ststep (sts_critical_pairs R)"
    then show ?thesis
    proof(rule ststepE, safe)
      fix l r bef aft sa ta
      assume lr:"(l, r) ∈ sts_critical_pairs R"
        and s:"s = bef @ l @ aft" 
        and t:"t = bef @ r @ aft" 
        and "(bef @ l @ aft, bef @ r @ aft) ∈ ststep (sts_critical_pairs R)"
      with assms [THEN subsetD, OF lr] 
      show "(bef @ l @ aft, bef @ r @ aft) ∈ (⋃ i ≤ n. (ststep (E i))⇧↔)" 
        unfolding ststep.simps using append.assoc ststepE ststepI 
        by (smt (verit, del_insts) UN_iff Un_iff converse_iff)
    qed
  next
    assume "(t, s) ∈ ststep (sts_critical_pairs R)"
    then show ?thesis
     proof(rule ststepE, safe)
      fix l r bef aft sa ta
      assume lr:"(l, r) ∈ sts_critical_pairs R"
        and t:"t = bef @ l @ aft" 
        and s:"s = bef @ r @ aft" 
        and "(bef @ l @ aft, bef @ r @ aft) ∈ ststep (sts_critical_pairs R)"
      with assms [THEN subsetD, OF lr] 
      show "(bef @ r @ aft, bef @ l @ aft) ∈ (⋃i≤n. (ststep (E i))⇧↔)" 
        unfolding ststep.simps using append.assoc ststepE ststepI 
        by (smt (verit, del_insts) UN_iff Un_iff converse_iff)
    qed
  qed
qed

lemma fair_imp_jcp_peak:
  assumes fair:"sts_critical_pairs R ⊆ (⋃ i ≤ n. (ststep (E i))⇧↔)"
  shows "⋀s t u. (t, u) ∈ jcp_peak R s ⟹ (t, u) ∈ (ststep R)⇧↓ ∪ (⋃ i ≤ n. (ststep (E i))⇧↔)"
  using scp_subset_imp [OF assms]
  by (auto simp add: jcp_peak_def)

(* Definition 14 *)
definition mststep :: "string multiset ⇒ sts ⇒ string rel"
  where
    "mststep M R = {(s, t). (s, t) ∈ ststep R ∧ (∃s' t'. s' ∈# M ∧ t' ∈# M ∧ s' ≽⇩s⇩l s ∧ t' ≽⇩s⇩l t)}"

lemma mststep_iff:
  "(x, y) ∈ mststep M R ⟷ (x, y) ∈ ststep R ∧ (∃s' t'. s' ∈# M ∧ t' ∈# M ∧ s' ≽⇩s⇩l x ∧ t' ≽⇩s⇩l y)"
  by (auto simp: mststep_def)

lemma UN_mststep:
  "(⋃x∈R. mststep M {x}) = mststep M R"
  by (auto simp add: mststep_iff) blast+

lemma mststep_Un [simp]:
  "mststep M (R ∪ R') = mststep M R ∪ mststep M R'"
  by (auto iff: mststep_iff)

lemma mststep_mono [simp]:
  "R ⊆ R' ⟹ mststep M R ⊆ mststep M R'"
  unfolding mststep_def by fast

lemma mststep_subset':
  "(mststep M R')⇧= O (mststep M (E' ∪ R'))⇧= O ((mststep M R')¯)⇧= ⊆ ((mststep M (E' ∪ R'))⇧↔)⇧*"
  (is "?L ⊆ ?R")
proof
  fix s t assume "(s, t) ∈ ?L"
  then obtain u and v where "(s, u) ∈ (mststep M R')⇧="
    and "(u, v) ∈ (mststep M (E' ∪ R'))⇧="
    and "(v, t) ∈ ((mststep M R')¯)⇧=" by auto
  then have "(s, u) ∈ ?R" and "(u, v) ∈ ?R" and "(v, t) ∈ ?R" by auto
  then show "(s, t) ∈ ?R" by auto
qed

lemma mststep_converse:"(mststep M R)⇧↔ = mststep M (R⇧↔)"  (is "?L = ?R")
proof 
  show lr:"?L ⊆ ?R"
  proof
    fix t
    assume "t ∈ ?L"
    then show "t ∈ ?R" using mststep_iff 
      by (smt (verit, ccfv_threshold) Un_iff converseE converseI mststep_Un ststep_converse)    
  qed
  show "?R ⊆ ?L"
  proof
    fix t
    assume "t ∈ ?R"
    then show "t ∈ ?L"  
      by (smt (verit, ccfv_threshold) UnCI UnE lr converse_iff equalityI mststep_Un mststep_iff ststep_converse subrelI)
  qed
qed

lemma ststeps_subset_shortlex_s:
  assumes "R ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "(ststep R)⇧+ ⊆ {(x, y). x ≻⇩s⇩l y}"
proof
  fix s t
  assume "(s, t) ∈ (ststep R)⇧+"
  then show "(s, t) ∈ {(x, y). x ≻⇩s⇩l y}"
  proof (induct)
    case (base u)
    with ststep_subset_S [OF assms] show ?case by auto
  next
    case (step t u)
    then show ?case
      using ststep_subset_S [OF assms]
      by (meson shorlex_pair.trans_S_point subsetD)
  qed
qed

lemma mststep_subset:
  assumes "(E, R) ⊢⇩S⇩R (E', R')" and "R' ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "mststep M (E ∪ R) ⊆ (mststep M (E' ∪ R'))⇧↔⇧*"
proof
  fix s t
  assume "(s, t) ∈ mststep M (E ∪ R)"
  then obtain s' and t' where "s' ∈# M" and "t' ∈# M"
    and "s' ≽⇩s⇩l s" and "t' ≽⇩s⇩l t" and "(s, t) ∈ ststep (E ∪ R)"
    by (auto simp: mststep_def)
  with SR_subset_pre [OF assms(1)] obtain u and v
    where "(s, u) ∈ (ststep R')⇧=" and "(u, v) ∈ (ststep (E' ∪ R'))⇧="
      and "(v, t) ∈ ((ststep R')¯)⇧=" by blast
  moreover have tv:"(t, v) ∈ (ststep R')⇧=" using calculation(3) by fastforce
  moreover have sr_slex:"ststep R' ⊆ shortlex_S" using ststep_subset_S [OF assms(2)]  by auto
  ultimately have "s ≽⇩s⇩l u" and "t ≽⇩s⇩l v"
    by (blast, metis Un_iff sr_slex case_prodD mem_Collect_eq pair_in_Id_conv subset_Un_eq tv)
  with ‹s' ≽⇩s⇩l s› and ‹t' ≽⇩s⇩l t› have "s' ≽⇩s⇩l u" and "t' ≽⇩s⇩l v" using shortlex_trans by blast+
  then have "(s, u) ∈ (mststep M R')⇧="
    and "(u, v) ∈ (mststep M (E' ∪ R'))⇧="
    and "(v, t) ∈ ((mststep M R')¯)⇧="
    using ‹s' ∈# M› and ‹t' ∈# M›
      and ‹s' ≽⇩s⇩l s› and ‹t' ≽⇩s⇩l t›
      and ‹(s, u) ∈ (ststep R')⇧=›
      and ‹(u, v) ∈ (ststep (E' ∪ R'))⇧=›
      and ‹(v, t) ∈ ((ststep R')¯)⇧=›
    unfolding mststep_def by (blast)+
  then have "(s, t) ∈ (mststep M R')⇧= O ((mststep M (E' ∪ R')))⇧= O ((mststep M R')¯)⇧=" by blast
  with mststep_subset' [of M R' E'] show "(s, t) ∈ (mststep M (E' ∪ R'))⇧↔⇧*"
    unfolding conversion_def by blast
qed

lemma ststeps_imp_mststeps:
  assumes "t ∈# M" and "(t, u) ∈ (ststep R)⇧*" and "R ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "(t, u) ∈ (mststep M R)⇧*"
  using assms(2, 1)
proof (induct)
  case base show ?case by simp
next
  note sl = ststeps_subset_shortlex_s [OF assms(3)]
  case (step u v)
  have "(u, v) ∈ ststep R" by fact
  moreover have "t ∈# M" by fact
  moreover
  from ‹(t, u) ∈ (ststep R)⇧*› and sl
  have "t ≽⇩s⇩l u" by (metis case_prodD mem_Collect_eq rtranclD subset_eq)
  moreover have "u ≻⇩s⇩l v" using sl step by blast
  ultimately have "(u, v) ∈ mststep M R" 
    by (meson mststep_iff shortlex_trans)
  with step show ?case by auto
qed

lemma mststep_subset'':
  assumes "(E, R) ⊢⇩S⇩R (E', R')" and "R' ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "(mststep M (E ∪ R))¯ ⊆ (mststep M (E' ∪ R'))⇧↔⇧*"
  using converse_mono [THEN iffD2, OF mststep_subset [OF assms, of M]]
  by auto

lemma mststep_symcl_subset:
  assumes "(E, R) ⊢⇩S⇩R (E', R')" and "R' ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "(mststep M (E ∪ R))⇧↔ ⊆ (mststep M (E' ∪ R'))⇧↔⇧*"
  using mststep_subset [OF assms] and mststep_subset'' [OF assms] by blast

(* Lemma 15 *)
lemma mststeps_subset:
  assumes "(E, R) ⊢⇩S⇩R (E', R')" and "R' ⊆ {(x, y). x ≻⇩s⇩l y}"
  shows "(mststep M (E ∪ R))⇧↔⇧* ⊆ (mststep M (E' ∪ R'))⇧↔⇧*"
  using mststep_symcl_subset [OF assms, THEN rtrancl_mono] by (simp add: conversion_def)

lemma UNIV_mstep_rstep_iff:
  "(⋃M∈UNIV. mststep M R) = ststep R"
proof -
  have *: "⋀s t. (s, t) ∈ ststep R ⟹
    s ∈# {#s, t#} ∧ t ∈# {#s, t#} ∧ s ≽⇩s⇩l s ∧ t ≽⇩s⇩l t ∧
  (s, t) ∈ mststep {#s, t#} R" by (auto iff: mststep_iff)
  show ?thesis by (auto iff: mststep_iff) (insert *, blast)
qed

lemma sts_finite_fair_run:
  assumes R0:"R 0 = {}" and En:"E n = {}"
    and run:"∀i < n. (E i, R i) ⊢⇩S⇩R (E (Suc i), R (Suc i))"
    and fair: "sts_critical_pairs (R n) ⊆ (⋃ i ≤ n. (ststep (E i))⇧↔)"
  shows "CR (ststep (R n))"
proof (rule Newman[OF sts_finite_run_SN[OF R0 En run]], unfold sts_critical_pair_lemma, safe)
  fix s t
  assume scp:"(s, t) ∈ sts_critical_pairs (R n)"
  define mshortlex_s where "mshortlex_s x y = mulex (≻⇩s⇩l)¯¯ y x" for x y
  have "CR (ststep (R n))"
  proof -
    interpret lab: ars_labeled_sn "λM. mststep M (R n)" UNIV mshortlex_s
    proof
      show "SN {(x, y). mshortlex_s x y}" unfolding mshortlex_s_def using shortlex_SN SN_mulex by blast
    qed
    interpret pd:ars_peak_decreasing "λM. mststep M (R n)" UNIV mshortlex_s
    proof
      fix s t u :: "string" and M⇩1 M⇩2 :: "string multiset"
      assume stM1:"(s, t) ∈ mststep M⇩1 (R n)" and suM2:"(s, u) ∈ mststep M⇩2 (R n)"
      from stM1 suM2 have st: "(s, t) ∈ (ststep (R n))" and su:"(s, u) ∈ ststep (R n)" 
        by (auto iff: mststep_iff)
      from fair have scp:"sts_critical_pairs (R n) ⊆ (⋃i≤n. (ststep (E i))⇧↔)" by fastforce
      from sts_critical_pairs_main[of s t "R n" u]
      have "((t, u) ∈ (ststep (R n))⇧↓ ∨ (t, u) ∈ (ststep (sts_critical_pairs (R n)))⇧↔)"
        using st su by blast
      hence tus:"(t, u) ∈ jcp_peak (R n) s" using st su unfolding jcp_peak_def by auto
      from fair_imp_jcp_peak[OF scp]
      have "(t, u) ∈ (ststep (R n))⇧↓ ∪ (⋃i≤n. (ststep (E i))⇧↔)" using tus by blast
      hence *:"(t, u) ∈ (ststep (R n))⇧↓ ∨ (t, u) ∈ (⋃i≤n. (ststep (E i))⇧↔)" using tus by blast
      let ?D = "(⋃c∈lab.downset2 M⇩1 M⇩2. mststep c (R n))⇧↔⇧*"
      let ?M = "{#t, u#}"
      have **: "⋀i. i ≤ n ⟹ (⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)"
      proof -
        fix i assume "i ≤ n"
        with run show "(⊢⇩S⇩R)⇧*⇧* (E 0, R 0) (E i, R i)" by (induct i, auto, meson less_eq_Suc_le rtranclp.simps)
      qed
      from ** [THEN sts_rtrancl_subset_shortlex] and ‹R 0 = {}›
      have R_shortlex:"⋀i. i ≤ n ⟹ R i ⊆ {(x, y). x ≻⇩s⇩l y}" by auto
      have msM1:"mshortlex_s M⇩1 ?M"
      proof -
        have sgeu:"s ≻⇩s⇩l u" using su R_shortlex
          by (metis (no_types, lifting) R_shortlex case_prodD less_or_eq_imp_le mem_Collect_eq ststep_subset_S subsetD)
        from st have ne1: "M⇩1 ≠ {#}" by (metis empty_iff mststep_iff set_mset_empty stM1)
        have MM1:"∀u. u ∈# ?M ⟶ (∃z. z ∈# M⇩1 ∧ z ≻⇩s⇩l u)" using sgeu mststep_iff shortlex_trans stM1
          by (smt (verit, best) R_shortlex add_eq_conv_ex add_mset_eq_single case_prodD less_or_eq_imp_le 
              mem_Collect_eq multi_member_split ststep_subset_S subsetD) 
        from mulex_on_all_strict [OF ne1 _ _ this, of UNIV]
        have "mulex (≻⇩s⇩l)¯¯ ?M M⇩1" 
          by (simp add:MM1 mulex_on_all_strict ne1)
        then show ?thesis using mshortlex_s_def by blast 
      qed
      have msM2:"mshortlex_s M⇩2 ?M"
      proof -
        have steu:"s ≻⇩s⇩l t" using st R_shortlex
          by (metis (no_types, lifting) R_shortlex case_prodD less_or_eq_imp_le mem_Collect_eq ststep_subset_S subsetD)
        from st have ne2: "M⇩2 ≠ {#}" by (metis empty_iff mststep_iff set_mset_empty suM2)
        have MM2:"∀u. u ∈# ?M ⟶ (∃z. z ∈# M⇩2 ∧ z ≻⇩s⇩l u)" using steu mststep_iff shortlex_trans suM2
          by (smt (verit, ccfv_SIG) R_shortlex add_eq_conv_ex add_mset_eq_single case_prodD less_or_eq_imp_le mem_Collect_eq multi_member_split ststep_subset_S subsetD)       
        from mulex_on_all_strict [OF ne2 _ _ this, of UNIV]
        have "mulex (≻⇩s⇩l)¯¯ ?M M⇩2" 
          by (simp add:MM2 mulex_on_all_strict ne2)
        then show ?thesis using mshortlex_s_def by blast 
      qed
      show "(t, u) ∈ ?D" using *
      proof
        assume "(t, u) ∈ (ststep (R n))⇧↓"
        then obtain w where tw:"(t, w) ∈ (ststep (R n))⇧*"
            and uw:"(u, w) ∈ (ststep (R n))⇧*" by auto
        from ststeps_imp_mststeps [of _ ?M, OF _ tw R_shortlex]
        have twM:"(t, w) ∈ (mststep ?M (R n))⇧↔⇧*" by auto
        from lab.conversion_in_downset2 [OF twM] msM1
        have twD: "(t, w) ∈ ?D" by auto
        from ststeps_imp_mststeps [of _ ?M, OF _ uw R_shortlex]
        have uwM:"(u, w) ∈ (mststep ?M (R n))⇧↔⇧*" by auto
        from lab.conversion_in_downset2 [OF uwM] msM2
        have uwD: "(u, w) ∈ ?D" by auto
        hence "(t, u) ∈ ?D" using twD uwD ars_labeled.conversion_R_conv_iff ars_labeled.conv_trans
          by (meson ars_labeled.conv_commute ars_labeled.conv_in_I)
        then show ?thesis .
      next
        assume "(t, u) ∈ (⋃i≤n. (ststep (E i))⇧↔)"
        then obtain i where iln:"i ≤ n" and "(t, u) ∈ (ststep (E i))⇧↔" by auto
        hence "(t, u) ∈ (mststep ?M (E i))⇧↔" by (auto iff: mststep_iff)
        hence "(t, u) ∈ (mststep ?M (E i ∪ R i))⇧↔⇧*" by (auto simp: conversion_def)
        then show ?thesis using iln
        proof(induct "n - i" arbitrary: i)
          case 0
          hence "i = n" by arith
          with 0 have tuM:"(t, u) ∈ (mststep ?M (R n))⇧↔⇧*" by (simp add: assms)
          from lab.conversion_in_downset2 [OF tuM] msM1 
          show ?case by auto
        next
          case (Suc m)
          from ‹Suc m = n - i› have *: "m = n - Suc i" and leq: "Suc i ≤ n" by auto
          from run have **: "(E i, R i) ⊢⇩S⇩R (E (Suc i), R (Suc i))" using leq by auto
          from ‹(t, u) ∈ (mststep ?M (E i ∪ R i))⇧↔⇧*›
          have "(t, u) ∈ (mststep ?M (E (Suc i) ∪ R (Suc i)))⇧↔⇧*"
            using mststeps_subset [OF **  R_shortlex [OF leq]] by blast
          from Suc.hyps(1) [OF * this leq]
          show ?case by auto
        qed
      qed
    qed
    from pd.CR show ?thesis using UNIV_mstep_rstep_iff by auto
  qed
  then show "(s, t) ∈ (ststep (R n))⇧↓"
    using En R0 run scp sts_critical_pair_CR sts_finite_run_SN by force
qed
