(* Definition 19 *)
inductive cond_eq_lr_theory :: "csts ⇒ (string × string) ⇒ bool" (infix "⊢⇩c⇩e⇩l⇩r" 55) where
  refl:"R ⊢⇩c⇩e⇩l⇩r (s, s)"
| symmetry:"R ⊢⇩c⇩e⇩l⇩r (s, t)" 
    if "R ⊢⇩c⇩e⇩l⇩r (t, s)"
| transitive:"R ⊢⇩c⇩e⇩l⇩r (s, u)" 
    if "R ⊢⇩c⇩e⇩l⇩r (s, t) ∧ R ⊢⇩c⇩e⇩l⇩r (t, u)"
| congruence:"R ⊢⇩c⇩e⇩l⇩r (u @ s @ v, u @ t @ v)" 
    if "R ⊢⇩c⇩e⇩l⇩r (s, t)"
| replacement:"R ⊢⇩c⇩e⇩l⇩r (u @ l @ w, u @ r @ w)" 
    if "((l, r), cs) ∈ R ∧ (∀(s⇩i, t⇩i) ∈ set cs. R ⊢⇩c⇩e⇩l⇩r (u @ s⇩i @ w, u @ t⇩i @ w))"

lemma cond_eq_theory_p_refl_sym_tran_closed: assumes "(s, t) ∈ (cond_eq_p_theory_rel R)⇧↔⇧*"
  shows "(s, t) ∈ (cond_eq_p_theory_rel R)"
proof -
  from assms obtain n where "(s, t) ∈ (cond_eq_p_theory_rel R)⇧↔^^n" by auto
  then show ?thesis
  proof(induct n arbitrary: t)
    case 0
    then show ?case 
      by (simp add: cond_eq_p_theory.refl cond_eq_p_theory_rel_def)
  next
    case (Suc n)
    from Suc(2) obtain u where su:"(s, u) ∈ (cond_eq_p_theory_rel R)⇧↔ ^^ n" and ut:"(u, t) ∈ (cond_eq_p_theory_rel R)⇧↔" by auto
    from su Suc(1) have "(s, u) ∈ (cond_eq_p_theory_rel R)" by auto
    moreover from ut have "(u, t) ∈ (cond_eq_p_theory_rel R)"
      by (auto, metis case_prod_conv cond_eq_p_theory.symmetry cond_eq_p_theory_rel_def mem_Collect_eq)
    ultimately show ?case
      by (metis case_prod_conv cond_eq_p_theory.transitive cond_eq_p_theory_rel_def mem_Collect_eq)
  qed
qed

lemma csr_semi_equational_cond_eq_p_theory_single:
  assumes "(s, t) ∈ csr_semi_equational_step R"
  shows "R ⊢⇩c⇩e⇩p (s, t)"
proof -
  from assms obtain n where "(s, t) ∈ (csr_semi_equational_step_n R (Suc n))" 
    using csr_semi_equational_step_iff csr_semi_equational_step_n_E by metis
  then obtain l r cs C where lr:"((l, r), cs) ∈ R" and cond:"∀(s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_semi_equational_step_n R n)⇧↔⇧*"
    and s:"s = C⟪l⟫" and t:"t = C⟪r⟫" by auto
  then show ?thesis
  proof(induct n arbitrary: l r cs C s t)
    case 0
    then show ?case by (auto, metis (mono_tags, lifting) case_prodI2 cond_eq_p_theory.congruence 
          cond_eq_p_theory.refl cond_eq_p_theory.replacement old.prod.case sctxt_app_step)
  next
    case (Suc n)
    let ?S = "{(C⟪l⟫, C⟪r⟫) | C l r cs.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_semi_equational_step_n R n)⇧↔⇧*)}"
    have "?S = csr_semi_equational_step_n R (Suc n)" by auto
    moreover have *:"∀a b. (a, b) ∈ ?S ⟶ R ⊢⇩c⇩e⇩p (a, b)" using Suc(1) by auto
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (?S)⇧↔⇧*" using Suc by auto
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (cond_eq_p_theory_rel R)⇧↔⇧*" unfolding cond_eq_p_theory_rel_def 
      using cond_membership_cond_eq_p_theory_rel_refl_sym_tran_closed[OF *] Suc
      by (auto, simp add: cond_eq_p_theory_rel_def)
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. R ⊢⇩c⇩e⇩p (s⇩i, t⇩i)" using cond_eq_theory_p_refl_sym_tran_closed
      using calculation(4) cond_eq_p_theory_rel_def by auto
    ultimately show ?case
    proof -
      have "R ⊢⇩c⇩e⇩p (l, r)"
        using Suc.prems(1) ‹∀(s⇩i, t⇩i)∈set cs. R ⊢⇩c⇩e⇩p (s⇩i, t⇩i)› by blast
      then show ?thesis
        using Suc.prems(3) Suc.prems(4) cond_eq_p_theory.congruence sctxt_app_step by metis
    qed
  qed
qed

lemma csr_r_semi_equational_cond_eq_r_theory_single:
  assumes "(s, t) ∈ csr_r_semi_equational_step R"
  shows "R ⊢⇩c⇩e⇩r (s, t)"
proof -
  from assms obtain n where "(s, t) ∈ (csr_r_semi_equational_step_n R (Suc n))" 
    using csr_r_semi_equational_step_iff csr_r_semi_equational_step_n_E by metis
  then obtain l r cs C w where lr:"((l, r), cs) ∈ R" and cond:"∀(s⇩i, t⇩i) ∈ set cs. (s⇩i @ w, t⇩i @ w) ∈ (csr_r_semi_equational_step_n R n)⇧↔⇧*"
    and s:"s = C⟪l @ w⟫" and t:"t = C⟪r @ w⟫" by auto
  then show ?thesis
  proof(induct n arbitrary: l r cs C w s t)
    case 0
    hence "R ⊢⇩c⇩e⇩r (l @ w, r @ w)" by fastforce
    then show ?case by (metis "0.prems"(3) "0.prems"(4) cond_eq_r_theory.congruence sctxt_app_step)
  next
    case (Suc n)
    let ?S = "{(C⟪l @ w⟫, C⟪r @ w⟫) | C l r cs w.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i @ w, t⇩i @ w) ∈ (csr_r_semi_equational_step_n R n)⇧↔⇧*)}"
    have "?S = csr_r_semi_equational_step_n R (Suc n)" by auto
    moreover have *:"∀a b. (a, b) ∈ ?S ⟶ R ⊢⇩c⇩e⇩r (a, b)" using Suc(1) by (auto split:prod.splits) 
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. (s⇩i @ w, t⇩i @ w) ∈ (?S)⇧↔⇧*" using Suc by (auto split:prod.splits)
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. (s⇩i @ w, t⇩i @ w) ∈ (cond_eq_r_theory_rel R)⇧↔⇧*" unfolding cond_eq_r_theory_rel_def 
      using cond_membership_cond_eq_r_theory_rel_refl_sym_tran_closed[OF *] Suc cond_eq_r_theory_rel_def by fastforce
    moreover have "∀(s⇩i, t⇩i) ∈ set cs. R ⊢⇩c⇩e⇩r (s⇩i @ w, t⇩i @ w)" using cond_eq_theory_r_refl_sym_tran_closed
      using calculation(4) cond_eq_r_theory_rel_def by (auto split:prod.splits)
    ultimately show ?case
    proof -
      have "R ⊢⇩c⇩e⇩r (l @ w, r @ w)"
        using Suc.prems(1) ‹∀(s⇩i, t⇩i)∈set cs. R ⊢⇩c⇩e⇩r (s⇩i @ w, t⇩i @ w)› by blast
      then show ?thesis
        using Suc.prems(3) Suc.prems(4) cond_eq_r_theory.congruence sctxt_app_step by metis
    qed
  qed
qed

(* Lemma 25(iii) *)
lemma csr_semi_equational_cond_eq_p_theory_equiv:
  "(s, t) ∈ (csr_semi_equational_step R)⇧↔⇧* ⟷ R ⊢⇩c⇩e⇩p (s, t)" (is "?LHS ⟷ ?RHS")
proof
  assume asm:?LHS
  then obtain n where "(s, t) ∈ (csr_semi_equational_step R)⇧↔^^n" by auto
  then show ?RHS
  proof (induct n arbitrary:t)
    case 0
    then show ?case by auto
  next
    case (Suc n)
    from Suc(2) obtain u where su:"(s, u) ∈ (csr_semi_equational_step R)⇧↔ ^^ n" and ut:"(u, t) ∈ (csr_semi_equational_step R)⇧↔" by auto
    with Suc(1) have "R ⊢⇩c⇩e⇩p (s, u)" by auto
    moreover from ut have "R ⊢⇩c⇩e⇩p (u, t)" using csr_semi_equational_cond_eq_p_theory_single by auto
    ultimately show ?case by auto 
  qed
next
  assume asm:?RHS
  then show ?LHS
  proof induct
    case (refl R s)
    then show ?case by auto
  next
    case (symmetry R t s)
    then show ?case by (simp add: conversion_inv)
  next
    case (transitive R s t u)
    then show ?case by (meson conversion_trans transD)
  next
    case (congruence R s t u v)
    then show ?case by (simp add: csr_semi_equational_step_ctxt_closed sctxt.closed_conversion sctxt_closed_strings)
  next
    case (replacement l r cs R)
    hence "(∀(s⇩i, t⇩i)∈set cs.  (s⇩i, t⇩i) ∈ (csr_semi_equational_step R)⇧↔⇧*)" by auto
    hence "∀C. (C⟪l⟫, C⟪r⟫) ∈ (csr_semi_equational_step R)" using csr_semi_equational_stepI replacement by auto
    then show ?case by (metis ShortLex.hole conversionI' r_into_rtrancl)
  qed
qed

definition normalized_p :: "csts ⇒ string ⇒ bool" where
  "normalized_p R w ⟷ (w ∈ NF (csr_p_join_step R))"

definition "strongly_p_irreducible R s ⟷ (∀((l, r), cs) ∈ R. ¬ (subseq l s))"

lemma sp_irreducible_imp_normalized_p: 
  assumes sp:"strongly_p_irreducible R s"
  shows "normalized_p R s"
proof(rule ccontr)
  assume "¬ ?thesis"
  hence "∃t. (s, t) ∈ (csr_p_join_step R)" unfolding normalized_p_def by auto
  then obtain t where "(s, t) ∈ (csr_p_join_step R)" by auto
  then obtain n where "(s, t) ∈ (csr_p_join_step_n R (Suc n))" 
    by (meson csr_p_join_step_iff csr_p_join_step_n_Suc_mono subsetD)
  then obtain C l r cs where lr:"((l, r), cs) ∈ R"
    and "∀(s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧↓" and s:"s = C⟪l⟫" and "t = C⟪r⟫" by auto
  from s sp[unfolded strongly_p_irreducible_def] 
  show False by (auto split:prod.splits, metis lr list_emb_append2 list_emb_prefix sctxt_app subseq_order.dual_order.refl)
qed

definition strongly_p_deterministic where
  "strongly_p_deterministic R ⟷ (∀((l, r), cs) ∈ R. ∀(s⇩i, t⇩i) ∈ set cs. normalized_p R t⇩i)"

lemma strongly_p_deterministic_equiv: assumes sp:"strongly_p_deterministic R"
  shows "∀l r cs n s⇩i t⇩i w. ((l, r), cs) ∈ R ⟶ (s⇩i, t⇩i) ∈ set cs ⟶  
  ((s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧↓ ⟷ (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧*)" using assms[unfolded strongly_p_deterministic_def]
  by (auto split:prod.splits simp add: NF_iff_no_step NF_join_imp_reach csr_p_join_step_iff normalized_p_def)

proposition csr_join_csr_step_equiv:
  assumes sp:"strongly_p_deterministic R"

  from strongly_p_deterministic_equiv[of R]
  have *:"∀l r cs n s⇩i t⇩i w. ((l, r), cs) ∈ R ⟶ (s⇩i, t⇩i) ∈ set cs ⟶  
    ((s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧↓ ⟷ (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧*)" using sp by auto
  have zero:"csr_p_join_step_n R 0 = csr_step_n R 0" by auto
  have "∀n. csr_p_join_step_n R n = csr_step_n R n"
  proof(intro allI impI)
    fix n

        ((s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧↓ ⟷ (s⇩i, t⇩i) ∈ (csr_step_n R n)⇧*)" using * Suc(1) by metis
      moreover have "csr_p_join_step_n R (Suc n) = {(C⟪l⟫, C⟪r⟫) | C l r cs.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧↓)}" by auto
      moreover have "csr_p_join_step_n R (Suc n) = {(C⟪l⟫, C⟪r⟫) | C l r cs.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧*)}" using * remove_suffix.cases
        by (auto, metis+)
      moreover have "csr_step_n R (Suc n) = {(C⟪l⟫, C⟪r⟫) | C l r cs.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_step_n R n)⇧*)}" by auto
      moreover have "csr_step_n R (Suc n) = {(C⟪l⟫, C⟪r⟫) | C l r cs.
        ((l, r), cs) ∈ R ∧ (∀ (s⇩i, t⇩i) ∈ set cs. (s⇩i, t⇩i) ∈ (csr_p_join_step_n R n)⇧*)}" using IH Suc by auto
