section ‹String Rewriting›

theory String_Rewriting
  imports
   "HOL-Library.Sublist"
    ShortLex
begin

text ‹STS (semi-Thus system) is a set of string rewriting rules ›
type_synonym srule = "string × string"
type_synonym sts  = "srule set"

inductive_set ststep :: "sts ⇒ sts" for R 
  where
    ststep: "⋀bef aft. (l, r) ∈ R ⟹ s = bef @ l @ aft ⟹ t = bef @ r @ aft ⟹ (s, t) ∈ ststep R"

lemmas ststepI = ststep.intros [intro]
lemmas ststepE = ststep.cases [elim]

lemma ststep_incl[simp]: "R ⊆ ststep R" 
proof(intro subrelI)
  fix l r
  assume "(l, r) ∈ R"
  then show "(l, r) ∈ ststep R" unfolding ststep.simps by blast
qed

lemma ststep_mono[simp]: "R ⊆ S ⟹ ststep R ⊆ ststep S" by force

lemma ststep_union[simp]: "ststep (R ∪ S) = ststep R ∪ ststep S" by auto

lemma ststep_empty [simp]: "ststep {} = {}" by auto

lemma ststep_ID [simp]: "ststep Id = Id"
proof
  show "ststep Id ⊆ Id"
  proof(intro subrelI)
    fix s t
    assume "(s, t) ∈ ststep Id"
    then show "(s, t) ∈ Id" by auto
  qed
qed auto

lemma ststep_converse[simp]: "ststep (R¯) = (ststep R)¯" by auto

lemma ststep_closed[simp]:"ststep (ststep R) = ststep R" (is "?lhs = ?rhs")
proof 
  show "?lhs ⊆ ?rhs" unfolding ststep.simps
  proof(intro subrelI)
    fix s t
    assume "(s, t) ∈ ststep (ststep R)"
    hence "∃bef1 aft1 l1 r1. (l1, r1) ∈ ststep R ∧ s = bef1 @ l1 @ aft1 ∧ t = bef1 @ r1 @ aft1" by blast
    then obtain bef1 aft1 l1 r1 where l1r1:"(l1, r1) ∈ ststep R" and s':"s = bef1 @ l1 @ aft1" and t':"t = bef1 @ r1 @ aft1" by auto
    from l1r1 have "∃bef2 aft2 l r. (l, r) ∈ R ∧ l1 = bef2 @ l @ aft2 ∧ r1 = bef2 @ r @ aft2" by blast
    then obtain bef2 aft2 l r where lr:"(l, r) ∈ R" and l1:"l1 = bef2 @ l @ aft2" and r1:"r1 = bef2 @ r @ aft2" by auto
    have s:"s = bef1 @ bef2 @ l @ aft2 @ aft1" and t:"t = bef1 @ bef2 @ r @ aft2 @ aft1" 
      by (auto simp add:s' t' l1 r1) 
    show "(s, t) ∈ ststep R"
      by (simp add:s t ststepI, insert lr append_eq_appendI, blast) 
  qed
next
  show "?rhs ⊆ ?lhs"
  proof(intro subrelI)
    fix s t
    assume "(s, t) ∈ ststep R"
    then show "(s, t) ∈ ststep (ststep R)"
      using ststep_incl by blast
  qed
qed
  
lemma ststep_reflcl[simp]: "ststep (R⇧=) = (ststep R)⇧=" by simp
 
lemma ststep_symcl[simp]: "ststep (R⇧↔) = (ststep R)⇧↔"
proof
  show "ststep (R⇧↔) ⊆ (ststep R)⇧↔"
  proof(intro subrelI)
    fix s t
    assume "(s, t) ∈ ststep (R⇧↔)"
    hence "∃bef aft l r. (l, r) ∈ R⇧↔ ∧ s = bef @ l @ aft ∧ t = bef @ r @ aft" by blast
    then obtain bef aft l r where lr:"(l, r) ∈ R⇧↔" and s:"s = bef @ l @ aft" and t:"t = bef @ r @ aft" by auto
    from lr show "(s, t) ∈ (ststep R)⇧↔"
      by (cases "(l, r) ∈ R", insert s t, blast+)
  qed  
next
  show "(ststep R)⇧↔ ⊆ ststep (R⇧↔)"
  proof(intro subrelI)
    fix s t
    assume "(s, t) ∈ (ststep R)⇧↔"
    then show "(s, t) ∈ ststep (R⇧↔)"
    proof
      assume "(s, t) ∈ (ststep R)¯"
      then show ?thesis by blast
    qed auto
  qed
qed

lemma sts_incl_conv[simp]: "(ststep (E ∪ R))⇧↔ O (ststep R)⇧= ⊆ (ststep (E ∪ R))⇧↔⇧*"
proof(rule subrelI)
  fix l r
  assume "(l, r) ∈ (ststep (E ∪ R))⇧↔ O (ststep R)⇧="
  then obtain u where lu:"(l, u) ∈ (ststep (E ∪ R))⇧↔" and ur:"(u, r) ∈ (ststep R)⇧=" by auto
  from ur show "(l, r) ∈ (ststep (E ∪ R))⇧↔⇧*"
  proof(cases "u = r")
    case True
    then show ?thesis using lu by blast
  next
    case False note F1 = this
    from lu show ?thesis
    proof(cases "(l, u) ∈ ststep (E ∪ R)")
      case True
      from ur have "(u, r) ∈ ststep (E ∪ R)" using F1 by auto
      then show ?thesis 
        by (meson True converse_rtrancl_into_rtrancl conversionI' r_into_rtrancl)
    next
      case False
      from ur have "(u, r) ∈ ststep (E ∪ R)" using F1 by auto
      then show ?thesis
        by (meson UnCI converse_rtrancl_into_rtrancl conversionI lu r_into_rtrancl)
    qed
  qed
qed

lemma ststep_sctxt [intro]: "(s, t) ∈ ststep R ⟹ (C⟪s⟫, C⟪t⟫) ∈ ststep R"
proof -
  assume "(s, t) ∈ ststep R"
  hence "∃bef aft l r. (l, r) ∈ R ∧ s = bef @ l @ aft ∧ t = bef @ r @ aft" by blast
  then obtain bef aft l r where lr:"(l, r) ∈ R" 
    and s:"s = bef @ l @ aft" and t:"t = bef @ r @ aft" by auto
  have "∃bef' aft'. C⟪s⟫ = bef'@ s @ aft' ∧ C⟪t⟫ = bef'@ t @ aft'" 
    using sctxt_apply_string.simps
  proof(induct C)
    case Hole
    then show ?case using hole empty_append by metis
  next
    case (More x1 C x3)
    then show ?case  by (metis append_assoc)
  qed
  then obtain bef' aft' where cs:"C⟪s⟫ = bef'@ s @ aft'" and ct:"C⟪t⟫ = bef'@ t @ aft'" by blast
  then show "(C⟪s⟫, C⟪t⟫) ∈ ststep R" using append.assoc cs ct lr s t ststepI
    by metis
qed

lemma ststeps_closed_sctxt[simp]:
  assumes "(s, t) ∈ (ststep R)⇧*"
  shows "(C⟪s⟫, C⟪t⟫) ∈ (ststep R)⇧*" 
  using assms ststep_sctxt sctxt.closed_rtrancl
  by (simp add: sctxt.closedD sctxt.closedI)

lemma ststeps_sym_closed_sctxt[simp]:
  assumes "(s, t) ∈ (ststep R)⇧↔⇧*"
  shows "(C⟪s⟫, C⟪t⟫) ∈ (ststep R)⇧↔⇧*" 
  by (simp add: assms sctxt.closedD sctxt.closedI sctxt.closed_conversion ststep_sctxt)

lemma sclosed_trancl[simp]: "ststep ((ststep R)⇧↔⇧*) =  (ststep R)⇧↔⇧*"
proof
  show "ststep ((ststep R)⇧↔⇧*) ⊆ (ststep R)⇧↔⇧*"
  proof(rule subrelI)
    fix s t
    assume "(s, t) ∈ ststep ((ststep R)⇧↔⇧*)"
    hence "∃l r bef aft. (l, r) ∈ (ststep R)⇧↔⇧* ∧ s = bef @ l @ aft ∧ t = bef @ r @ aft" by blast
    then obtain l r bef aft where lr:"(l, r) ∈ (ststep R)⇧↔⇧*" and s:"s = bef @ l @ aft" and t:"t = bef @ r @ aft" by auto
    let ?C = "More bef ○ aft"
    from ststeps_sym_closed_sctxt[OF lr, of ?C]
    show "(s, t) ∈ (ststep R)⇧↔⇧*"
      by (simp add: s t)
  qed
qed auto

lemma ststep_rule [intro]: "(l, r) ∈ R ⟹ (l, r) ∈ ststep R"
  using ststep.ststep empty_append by blast

lemma sctxt_closed_ststeps [intro]: "sctxt.closed ((ststep R)⇧*)" by blast
lemma sctxt_closed_ststep [intro]: "sctxt.closed (ststep R)" by blast

lemma ststep_eq_closure: "ststep R = sctxt.closure R"
proof (auto elim: sctxt.closure.cases)
  fix a b
  assume asm:"(a,b) ∈ ststep R"
  hence "∃l r. (l, r) ∈ R" by (meson ststepE)
  then obtain l r where lr:"(l, r) ∈ R" 
    and "∃ bef aft. a = bef @ l @ aft ∧ b = bef @ r @aft" by (meson asm ststepE)
  then obtain bef aft where "a = bef @ l @ aft" and "b = bef @ r @aft" by auto
  then have "∃C. a = C⟪l⟫ ∧ b = C⟪r⟫" 
    using sctxt_apply_string.simps by metis
  then obtain C where a:"a = C⟪l⟫" and b:"b = C⟪r⟫" by auto
  from sctxt.closureI2[OF lr a b]
    show "(a, b) ∈ sctxt.closure R" by auto
qed

lemma sts_CR_WCR:
  fixes R :: "sts"
  assumes sn: "SN (ststep R)"
  shows "CR (ststep R) ⟷ WCR (ststep R)" using assms
    by (meson CR_onE Newman WCR_onI r_into_rtrancl)

definition ststep_c_s :: "sts  ⇒ sts"
  where "ststep_c_s R = {(s,t) | s t rl C. rl ∈ R ∧ s = C⟪fst rl⟫ ∧ t = C⟪snd rl⟫}"

lemma ststep_iff_ststep_r_s:
  "(s, t) ∈ ststep R ⟷ (s, t) ∈ ststep_c_s R" (is "?lhs = ?rhs")
proof
  assume asm:"(s, t) ∈ ststep R"
  hence "∃l r. (l, r) ∈ R" by (meson ststepE)
  then obtain l r where lr:"(l, r) ∈ R" 
    and "∃ bef aft. s = bef @ l @ aft ∧ t = bef @ r @aft" by (meson asm ststepE)
  then obtain bef aft where "s = bef @ l @ aft" and "t = bef @ r @aft" by auto
  then have "∃C. s = C⟪l⟫ ∧ t = C⟪r⟫" 
    using sctxt_apply_string.simps by metis
  then obtain C where a:"s = C⟪l⟫" and b:"t = C⟪r⟫" by auto
  then show ?rhs unfolding ststep_c_s_def using lr by auto
next
  assume ?rhs
  then obtain l r C where rl:"(l, r) ∈ R" and s:"s = C⟪l⟫" and t:"t = C⟪r⟫" 
    unfolding ststep_c_s_def by auto
  then show ?lhs by auto
qed

lemma ststep_ststep_c_s_iff:
  "ststep R = ststep_c_s R" using ststep_iff_ststep_r_s by auto

lemma sctxt_closed_SN_on_sublist:
  assumes sc:"sctxt.closed R" and "SN_on R {t}" and sub:"sublist s t"
  shows "SN_on R {s}"
proof (rule ccontr)
  assume "¬ SN_on R {s}"
  then obtain A where s:"A 0 = s" and A_chain:"∀i. (A i,A (Suc i)) ∈ R"
    unfolding SN_on_def by best
  from sub have "∃bef aft. t = bef @ s @ aft" 
    by (simp add: sublist_def)
  then obtain C where t:"t = C⟪s⟫" using sctxt_apply_string.simps
    by metis
  let ?B = "λi. C⟪A i⟫"
  have "∀i. (?B i,?B(Suc i)) ∈ R"
  proof
    fix i
    from A_chain have "(?B i,?B(Suc i)) ∈ sctxt.closure R" by fast
    then show "(?B i,?B(Suc i)) ∈ R" using sc by auto
  qed
  with s have "?B 0 = t ∧ (∀i. (?B i,?B(Suc i)) ∈ R)" using t by simp
  then have "¬ SN_on R {t}" unfolding SN_on_def by auto
  with assms show "False" by simp
qed

lemma sublist_preserves_SN_gen:
  assumes sctxt: "sctxt.closed R"
  and SN: "SN_on R {t}" and sslist: "strict_sublist s t"
  shows "SN_on R {s}"
proof -
  from sslist have "sublist s t" by auto
  then show ?thesis 
    using sctxt_closed_SN_on_sublist[OF sctxt SN, of s] by simp
qed

lemma sublist_preserves_SN:
  "SN_on (ststep R) {t} ⟹ sublist s t ⟹ SN_on (ststep R) {s}"
  using sublist_preserves_SN_gen strict_sublist_def by blast

end
