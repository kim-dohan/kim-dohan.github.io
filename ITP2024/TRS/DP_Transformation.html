(* @{term "Tinf"} characterizes termination. *)
lemma Tinf_SN_conv: "(Tinf R = {}) = (SN R)"
proof
  assume "Tinf R = {}" then show "SN R" using not_SN_imp_Tinf by blast
next
  assume "SN R" then show "Tinf R = {}" unfolding Tinf_def SN_defs by blast
qed

definition "wwf_inn_qtrs nfs Q R ≡ nfs ∧ NF_terms Q ⊆ NF_trs R ∧ Ball (fst ` R) is_Fun"

lemma wwf_inn_qtrs_imp_left_fun: 
  "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R ⟹ (l,r) ∈ R ⟹ is_Fun l"
  using wwf_qtrs_imp_left_fun[of Q R l r] unfolding wwf_inn_qtrs_def by auto

lemma Tinf_lemma_1:
  assumes wwf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R"
    and Tinf: "Fun f ts ∈ Tinf (qrstep nfs Q R)" (is "?t ∈ _")
  shows "∃l r u σ.
  (l, r) ∈ applicable_rules Q R ∧ 
  r ⊵ u ∧ 
  is_Fun u ∧ 
  (?t, l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧* ∧ 
  (l ⋅ σ, r ⋅ σ) ∈ rqrstep nfs Q R ∧ 
  r ⋅ σ ⊵ u ⋅ σ ∧
  l ⋅ σ ∈ Tinf (qrstep nfs Q R) ∧
  u ⋅ σ ∈ Tinf (qrstep nfs Q R) ∧
  ¬ (l ⊳ u) ∧
  NF_subst nfs (l, r) σ Q"
proof -
  let ?U = "applicable_rules Q R"
  let ?qr = "qrstep nfs Q R"
  let ?rqr = "rqrstep nfs Q R"
  let ?nrqr = "nrqrstep nfs Q R"
  from Tinf_imp_SN_nr_first_root_step [OF Tinf] obtain lhs and rhs
    where nsteps: "(Fun f ts, lhs) ∈ ?nrqr⇧*" and step: "(lhs, rhs) ∈ ?rqr"
    and nSN: "¬ SN_on ?qr {rhs}" by auto
  then obtain l r σ where NF: "∀v ⊲ l ⋅ σ. v ∈ NF_terms Q"
    and "(l, r) ∈ R" and lhs: "l ⋅ σ = lhs" and rhs: "r ⋅ σ = rhs" 
    and nfs: "NF_subst nfs (l, r) σ Q" by auto
  with step have "(l ⋅ σ, r ⋅ σ) ∈ ?rqr" by auto
  from ‹(l, r) ∈ R› and only_applicable_rules [OF NF]
    have "(l, r) ∈ ?U" by (auto simp: applicable_rules_def)
  from nrqrsteps_preserve_root_fun [OF nsteps] obtain ls' where "lhs = Fun f ls'" by auto
  have A: "⋀i. (ts ! i, ls' ! i) ∈ ?qr⇧*"
  proof -
    fix j
    from nsteps have "(args ?t ! j, args (Fun f ls') ! j) ∈ ?qr⇧*"
      unfolding ‹lhs = Fun f ls'› by (rule nrqrsteps_imp_arg_qrsteps)
    then show "?thesis j" by simp
  qed
  from nsteps and ‹lhs = Fun f ls'› have "(?t, Fun f ls') ∈ ?nrqr⇧*" by simp
  from ‹l ⋅ σ = lhs› and ‹lhs = Fun f ls'› have "l ⋅ σ = Fun f ls'" by simp
  with ‹(?t, Fun f ls') ∈ ?nrqr⇧*› have "(?t, l ⋅ σ) ∈ ?nrqr⇧*" by simp
  from ‹(?t, Fun f ls') ∈ ?nrqr⇧*›
    have numargs: "num_args ?t = num_args (Fun f ls')" by (rule nrqrsteps_num_args)
  from wwf_inn_qtrs_imp_left_fun [OF wwf ‹(l, r) ∈ R›] obtain g ls where "l = Fun g ls" by force
  then have "l ⋅ σ = Fun g (map (λt. t ⋅ σ) ls)" by simp
  with ‹l ⋅ σ = Fun f ls'› have "g = f" and "ls' = map (λt. t ⋅ σ) ls" by auto
  then have B: "∀i < length ls'. ls' ! i = (ls ! i) ⋅ σ" by auto
  have "∀i < num_args ?t. (ts ! i, (ls ! i) ⋅ σ) ∈ ?qr⇧*"
  proof (intro allI impI)
    fix i assume "i < num_args (Fun f ts)"
    with numargs have "i < length ls'" by simp
    with A [of i] show "(ts ! i, ls ! i ⋅ σ) ∈ ?qr⇧*" using B by simp
  qed
  from ‹l ⋅ σ = Fun f ls'› numargs have "num_args ?t = num_args (l ⋅ σ)"  by simp
  with ‹l ⋅ σ = Fun f ls'› and ‹∀i<length ls'. ls' ! i = (ls ! i) ⋅ σ›
  have "⋀i. i < num_args ?t ⟹ (ls ! i) ⋅ σ = ls' ! i" by auto
  have "⋀i. i < num_args ?t ⟹ ?t ⊳ (ts ! i)"
  proof -
    fix i assume "i < num_args ?t"
    then have "ts ! i ∈ set ts" by simp
    then show "?t ⊳ (ts ! i)" by (rule supt.arg)
  qed
  with Tinf [unfolded Tinf_def] have "⋀i. i < num_args ?t ⟹ SN_on ?qr {ts ! i}" by auto
  with ‹∀i < num_args ?t. (ts ! i, (ls ! i) ⋅ σ) ∈ ?qr⇧*› 
    have "⋀i. i < num_args ?t ⟹ SN_on ?qr {(ls ! i) ⋅ σ}"
    using steps_preserve_SN_on [of _ _ ?qr] by blast
  with ‹⋀i. i < num_args ?t ⟹ (ls ! i) ⋅ σ = ls' ! i› and ‹num_args ?t = num_args (l ⋅ σ)›
    and ‹l ⋅ σ = Fun f ls'› and ‹⋀i. i < num_args ?t ⟹ SN_on ?qr {(ls ! i) ⋅ σ}›
    have argsSN: "⋀i. i < num_args (Fun f ls') ⟹ SN_on ?qr {ls' ! i}" by auto
  have "∀s ⊲ Fun f ls'. SN_on ?qr {s}" 
  proof (intro allI impI)
    fix s
    assume tmp: "Fun f ls' ⊳ s"
    then obtain C where "C ≠ □" and "Fun f ls' = C⟨s⟩" by auto
    then obtain bef aft C where id: "Fun f ls' = Fun f (bef @ C⟨s⟩ # aft)" by (cases C) auto
    then have len: "length bef < num_args (Fun f ls')" by auto
    with argsSN [OF len] and id have "SN_on ?qr {C⟨s⟩}" by auto
    with ctxt_closed_SN_on_subt [OF ctxt_closed_qrstep _ ctxt_imp_supteq [of C s]]
      show "SN_on ?qr {s}" by auto
  qed
  then have "∀s ⊲ l ⋅ σ. SN_on ?qr {s}" unfolding ‹l ⋅ σ = Fun f ls'› by simp 
  from nSN rhs obtain B where B0: "B 0 = r ⋅ σ" and Bsteps: "⋀ i. (B i, B (Suc i)) ∈ ?qr" by auto
  from ‹(l ⋅ σ, r ⋅ σ) ∈ ?rqr› have "(l ⋅ σ, r ⋅ σ) ∈ ?qr" 
    unfolding qrstep_iff_rqrstep_or_nrqrstep by auto
  let ?B = "λj. case j of 0 ⇒ l ⋅ σ | Suc j ⇒ B j"
  have "?B 0 = l ⋅ σ" using lhs by simp
  { fix i
    have "(?B i, ?B (Suc i)) ∈ ?qr"
      using Bsteps [of "i - Suc 0"] and ‹(l ⋅ σ, r ⋅ σ) ∈ ?qr› and B0 by (cases i) auto }
  note Bsteps = this
  with ‹?B 0 = l ⋅ σ› have "¬ SN_on ?qr {l ⋅ σ}" unfolding SN_on_def by best
  with ‹∀s ⊲ l ⋅ σ. SN_on ?qr {s}› have tinf_ls: "l ⋅ σ ∈ Tinf ?qr" unfolding Tinf_def by simp
  { fix x
    assume x: "x ∈ vars_term r"
    from wwf have "SN_on ?qr {Var x ⋅ σ}"
    proof
      assume "wwf_qtrs Q R"
      with x have "x ∈ vars_term l" using ‹(l, r) ∈ ?U› unfolding wwf_qtrs_def wwf_rule_def applicable_rules_def by auto
      then have "l ⊳ Var x"
        using ‹l = Fun g ls› supteq_Var [OF ‹x ∈ vars_term l›] unfolding supt_supteq_conv by auto
      from supt_subst [OF this, of σ] and tinf_ls show ?thesis by (auto simp: Tinf_def)
    next
      assume "wwf_inn_qtrs nfs Q R"
      with nfs x have "σ x ∈ NF_trs R" unfolding NF_subst_def wwf_inn_qtrs_def vars_rule_def by auto
      then have "SN_on (rstep R) {σ x}" by (rule NF_imp_SN_on)
      from SN_on_mono [OF this]
        show ?thesis by auto
    qed }
  note xSN = this
  from nSN rhs have "¬ SN_on ?qr {r ⋅ σ}" by simp
  then obtain t' where "r ⋅ σ ⊵ t'" and "t' ∈ Tinf ?qr" using not_SN_imp_subt_Tinf by auto
  have "∀x ∈ vars_term r. ¬ (Var x ⋅ σ ⊵ t')"
  proof
    fix x assume "x ∈ vars_term r"
    then have "SN_on ?qr {Var x ⋅ σ}" using xSN by simp
    show "¬ Var x ⋅ σ ⊵ t'"
    proof
      assume "Var x ⋅ σ ⊵ t'"
      with ‹SN_on ?qr {Var x ⋅ σ}› have "SN_on ?qr {t'}" using ctxt_closed_SN_on_subt [OF ctxt_closed_qrstep] by auto
      with ‹t' ∈ Tinf ?qr› show "False" unfolding Tinf_def by auto
    qed
  qed
  with ‹r ⋅ σ ⊵ t'› have "∃u. r ⊵ u ∧ t' = u ⋅ σ"
    using subt_instance_and_not_subst_imp_subt [where s = r and σ = σ] by auto
  then obtain u where "r ⊵ u" and "t' = u ⋅ σ" by best
  have "is_Fun u"
  proof (rule ccontr)
    assume "¬ is_Fun u"
    then have "∃x. u = Var x" by (cases u) simp_all
    then obtain x where "u = Var x" by best
    with ‹r ⊵ u› have "r ⊵ Var x" by auto
    then have "x ∈ vars_term r" using subteq_Var_imp_in_vars_term by best
    with xSN ‹u = Var x› have "SN_on ?qr {u ⋅ σ}" by simp
    with ‹t' = u ⋅ σ› have "SN_on ?qr {t'}" by simp
    with ‹t' ∈ Tinf ?qr› show "False" by (simp add: Tinf_def)
  qed
  from ‹t' = u ⋅ σ› and ‹t' ∈ Tinf ?qr› have "u ⋅ σ ∈ Tinf ?qr" by simp
  have "¬ l ⊳ u"
  proof
    assume "l ⊳ u"
    then have "l ⋅ σ ⊳ u ⋅ σ" by (rule supt_subst)
    with ‹l ⋅ σ ∈ Tinf ?qr› have "SN_on ?qr {u ⋅ σ}" unfolding Tinf_def by best
    with ‹u ⋅ σ ∈ Tinf ?qr› show "False" by (auto simp: Tinf_def)
  qed
  from ‹r ⊵ u› have "r ⋅ σ ⊵ (u ⋅ σ)" by (rule supteq_subst)
  from ‹(l, r) ∈ ?U› and ‹r ⊵ u› and ‹is_Fun u› and ‹(?t, l ⋅ σ) ∈ ?nrqr⇧*› and ‹(l ⋅ σ,r ⋅ σ) ∈ ?rqr›
    and ‹u ⋅ σ ∈ Tinf ?qr› and ‹r ⋅ σ ⊵ u ⋅ σ› and ‹l ⋅ σ ∈ Tinf ?qr› and ‹¬ l ⊳ u› and nfs
  show ?thesis by blast
qed

lemma Tinf_var:
  assumes "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" shows "Var x ∉ Tinf (qrstep nfs Q R)"
  using SN_on_Var_gen[of R nfs Q x] wwf_inn_qtrs_imp_left_fun[OF assms] unfolding Tinf_def by force
 
lemma Tinf_lemma:
  assumes wwf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" and tinf: "t∈ Tinf (qrstep nfs Q R)"
  shows "∃l r u σ.
    (l, r) ∈ applicable_rules Q R ∧ r ⊵ u ∧ is_Fun u ∧ (t, l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*
    ∧ (l ⋅ σ, r ⋅ σ) ∈ rqrstep nfs Q R ∧ r ⋅ σ ⊵ u ⋅ σ ∧ l ⋅ σ ∈ Tinf (qrstep nfs Q R)
    ∧ u ⋅ σ ∈ Tinf (qrstep nfs Q R) ∧ ¬ (l ⊳ u) ∧ NF_subst nfs (l,r) σ Q"
proof (cases t)
  case (Var x)
  with Tinf_var[OF wwf] tinf have False by auto
  then show ?thesis ..
next
  case (Fun f ts)
  from Tinf_lemma_1[OF wwf tinf[simplified Fun]]
  show ?thesis unfolding Fun .
qed

lemma Tinf_imp_def_root_qrstep:
  assumes wwf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" and tinf: "t ∈ Tinf (qrstep nfs Q R)"
  shows "∃f ss. t = Fun f ss ∧ defined (applicable_rules Q R) (f, length ss)"
proof -
  from Tinf_var[OF wwf] tinf obtain f ts where t: "t = Fun f ts" by (cases t, auto)
  with tinf have tinf: "Fun f ts ∈ Tinf (qrstep nfs Q R)" by auto
  from Tinf_lemma[OF wwf tinf]
  obtain l r σ where lr: "(l,r) ∈ applicable_rules Q R" and "(Fun f ts,l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*" 
    by blast
  from lr have "(l,r) ∈ R" unfolding applicable_rules_def by auto
  from wwf_inn_qtrs_imp_left_fun[OF wwf this]
  obtain g ls where l: "l = Fun g ls" by blast
  with nrqrsteps_preserve_root[OF ‹(Fun f ts,l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*›]
    nrqrsteps_num_args[OF ‹(Fun f ts,l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*›]
  have "g = f" and "length ls = length ts" by auto
  with lr l have "defined (applicable_rules Q R) (f, length ts)" unfolding defined_def by auto
  with t show ?thesis by auto
qed

lemma Tinf_imp_Fun: 
   assumes wwf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" and "t ∈ Tinf (qrstep nfs Q R)"
   shows "∃f ts. t = Fun f ts"
   using Tinf_var[OF wwf] assms(2) by (cases t, auto)

definition wf_qtrs :: "bool ⇒ ('f, 'v) terms ⇒ ('f, 'v) trs ⇒ bool" where
  "wf_qtrs nfs Q R ≡ (wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R) ∧ (∀t∈Q. is_Fun t)"

lemma wf_qtrs_wf_trs_conv: "wf_qtrs nfs {} = wf_trs"
proof (rule ext)
  fix R :: "('f,'v)trs"
  show "wf_qtrs nfs {} R = wf_trs R"
    unfolding wf_qtrs_def wf_trs_def wwf_qtrs_def wwf_inn_qtrs_def
    by (force simp: applicable_rule_empty)
qed

lemma Tinf_lemma_DP:
  assumes wf: "wf_qtrs nfs Q R" and s: "s ∈ Tinf (qrstep nfs Q R)"
    and ndef:
    "⋀f n ss. defined (applicable_rules Q R) (f, n) ⟹ length ss = n ⟹ Fun (♯ f) ss ∉ Q"
  shows "∃l r u σ. (l,r) ∈ DP ♯ (applicable_rules Q R) ∧ r ⋅ σ = ♯ u
    ∧ u ∈ Tinf (qrstep nfs Q R) ∧ (♯ s, l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*
    ∧ l ⋅ σ ∈ NF_terms Q ∧ NF_subst True (l, r) σ Q"
proof -
  let ?qr = "qrstep nfs Q R"
  let ?U = "applicable_rules Q R"
  let ?nrqr = "nrqrstep nfs Q R"
  let ?rqr = "rqrstep nfs Q R"
  from wf have wf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" and wfQ: "∀t∈Q. is_Fun t"
    unfolding wf_qtrs_def by auto
  let ?Q = "Q" and ?R = "R"
  let ?Iqr = "nrqrstep nfs ?Q ?R"
  obtain l r u σ where lr: "(l,r) ∈ ?U" and "r ⊵ u" and "is_Fun u" and "(s,l ⋅ σ) ∈ ?nrqr⇧*"
    and step: "(l ⋅ σ,r ⋅ σ) ∈ ?rqr" and "l ⋅ σ ∈ Tinf ?qr" and "u ⋅ σ ∈ Tinf ?qr" and "¬ l ⊳ u"
    and nfs: "NF_subst nfs (l,r) σ Q"
    using Tinf_lemma[OF wf s] by auto
  from lr have "(l,r) ∈ R" unfolding applicable_rules_def by auto
  let ?u = "u ⋅ σ"
  let ?l = "♯ l"
  let ?r = "♯ u"
  let ?σ = σ
  obtain f us where "u = Fun f us" using ‹is_Fun u› by (cases u) auto
  have "root u = root ?u" unfolding ‹u = Fun f us› by simp
  obtain h vs where "?u = Fun h vs" and "defined ?U (h, length vs)"
    using Tinf_imp_def_root_qrstep[OF wf ‹?u ∈ Tinf ?qr›] by auto
  from ‹(s,l ⋅ σ) ∈ ?nrqr⇧*› have "(♯ s, ♯ (l ⋅ σ)) ∈ ?Iqr⇧*"
    by (rule nrqrsteps_imp_sharp_qrsteps)
  from wwf_inn_qtrs_imp_left_fun[OF wf ‹(l,r) ∈ R›] obtain g ts where l: "l = Fun g ts" by force
  have "♯ (l ⋅ σ) = ?l⋅?σ" unfolding ‹l = Fun g ts› by simp
  with ‹(♯ s,♯ (l ⋅ σ)) ∈ ?Iqr⇧*› have "(♯ s,?l⋅?σ) ∈ ?Iqr⇧*" by simp
  from ‹(u ⋅ σ) = Fun h vs› and ‹u = Fun f us› have "f = h" and "length us = length vs" by auto
  with ‹defined ?U (h, length vs)› have "defined ?U (f, length us)" by simp
  from ‹(l,r) ∈ ?U› and ‹r ⊵ u› and ‹defined ?U (f, length us)› and ‹¬(l ⊳ u)› have "(?l,?r) ∈ DP ♯ ?U" unfolding DP_on_def ‹u = Fun f us› by auto
  have "?r⋅?σ = ♯ ?u" unfolding ‹u = Fun f us› by simp
  from step have NF: "∀u⊲l ⋅ σ. u ∈ NF_terms Q" unfolding rqrstep_def rstep_r_c_s_def by auto
  from lr l have gdef: "defined ?U (g, length ts)" unfolding defined_def by auto
  note ndef = ndef[OF gdef]
  note NF = NF[unfolded l]
  have "?l ⋅ ?σ ∈ NF_terms ?Q"
  proof (rule ccontr)
    assume "¬ ?thesis"
    from not_NF_termsE[OF this]
    obtain q C γ where q: "q ∈ ?Q" and match: "?l ⋅ ?σ = C⟨q ⋅ γ⟩" .
    from q have qg: "q ⋅ γ ∉ NF_terms Q" unfolding NF_def by force
    show False
    proof (cases C)
      case (More ff bef D aft)
      then have "Fun g ts ⋅ ?σ ⊳ q ⋅ γ" using match l by auto
      with NF have "q ⋅ γ ∈ NF_terms Q" by auto
      with qg show False by simp
    next
      case Hole
      with match have match: "?l⋅?σ = q ⋅ γ" by auto
      show ?thesis 
      proof (cases q)
        case (Fun ff qq)
        from match[unfolded l this] have ff: "ff = ♯ g" and qq: "length ts = length qq" using map_eq_imp_length_eq[of _ ts _ qq] by auto
        from ndef[OF qq[symmetric]] q[unfolded Fun ff]  
        show ?thesis by auto
      next
        case (Var x)
        from q[unfolded Var] wfQ show False by auto
      qed
    qed 
  qed
  then have NF: "?l ⋅ ?σ ∈ NF_terms ?Q" by auto 
  have nfs: "NF_subst True (?l,?r) σ Q"
    unfolding NF_subst_def
  proof (intro impI subsetI)
    fix t
    assume t: "t ∈ σ ` vars_rule (?l,?r)"
    then obtain x where x: "x ∈ vars_rule (?l,?r)" and t: "t = σ x" by auto
    from x have "x ∈ vars_term ?l ∨ x ∈ vars_term ?r" unfolding vars_rule_def by auto
    then have "x ∈ vars_term ?l ∨ σ x ∈ NF_terms Q"
    proof 
      assume "x ∈ vars_term ?r"
      then have x: "x ∈ vars_term u" unfolding ‹u = Fun f us› by auto
      with supteq_imp_vars_term_subset [OF ‹r ⊵ u›] have x: "x ∈ vars_term r" by auto
      from wf show ?thesis
      proof
        assume "wwf_qtrs Q R"
        with lr x have "x ∈ vars_term l" unfolding wwf_qtrs_def applicable_rules_def by auto
        with ‹l = Fun g ts›
        show ?thesis by simp
      next
        assume "wwf_inn_qtrs nfs Q R"
        from this[unfolded wwf_inn_qtrs_def] have nfs by auto
        from nfs[unfolded NF_subst_def] vars_rule_def ‹(l,r) ∈ R› ‹nfs› x
        have "σ x ∈ NF_terms Q" by force
        then show ?thesis by simp
      qed
    qed simp
    then have "σ x ∈ NF_terms Q"
    proof
      assume "x ∈ vars_term ?l"
      then have "Var x ⊴ ?l" by auto
      then have "Var x ⋅ σ ⊴ ?l ⋅ σ" by auto
      from NF_subterm[OF NF this] show ?thesis by simp
    qed
    then show "t ∈ NF_terms Q" unfolding t .
  qed
  from NF nfs ‹(?l,?r) ∈ DP ♯ ?U› and ‹?r⋅?σ = ♯ ?u›
    and ‹?u ∈ Tinf ?qr› and ‹(♯ s, ?l⋅?σ) ∈ ?Iqr⇧*›
  show ?thesis by blast
qed

fun
  next4 :: "bool ⇒ ('f,'v)terms ⇒ ('f,'v)trs ⇒ ('f,'v)term ⇒
    ('f,'v)term × ('f,'v)term × ('f,'v)term × ('f,'v)subst"
where
  "next4 nfs Q R s = (SOME (l, r, u, σ).
    (l, r) ∈ DP ♯ (applicable_rules Q R) ∧ r ⋅ σ = ♯ u ∧ u ∈ Tinf (qrstep nfs Q R)
    ∧ (♯ s, l ⋅ σ) ∈ (nrqrstep nfs Q R)⇧*
    ∧ l ⋅ σ ∈ NF_terms Q ∧ NF_subst True (l,r) σ Q)"

fun
  concrete_seq :: "bool ⇒ ('f, 'v) terms ⇒ ('f, 'v) trs ⇒ ('f, 'v) term ⇒ nat ⇒
    ('f, 'v) term × ('f, 'v) term × ('f, 'v) term × ('f, 'v) term × ('f, 'v) subst"
where
  "concrete_seq nfs Q R s 0 = (s, next4 nfs Q R s)" |
  "concrete_seq nfs Q R r (Suc n) = (
    let ( _, _, _, s, _) = concrete_seq nfs Q R r n in
    (s, next4 nfs Q R s))"

theorem not_SN_imp_min_ichain:
  assumes "wf_qtrs nfs Q R" and "¬ SN (qrstep nfs Q R)" 
  and ndef: "⋀ f n ss. defined (applicable_rules Q R) (f,n) ⟹ length ss = n ⟹ Fun (♯ f) ss ∉ Q"
  and m: "m = (∀ f n. defined (applicable_rules Q R) (f,n) ⟶ ¬ defined (applicable_rules Q R) (♯ f,n))"
  shows "∃s t σ. min_ichain (initial_dpp nfs m Q R) s t σ"
proof -
  let ?qr = "qrstep nfs Q R"
  let ?Q = "Q" and ?R = "R"
  let ?Iqr = "nrqrstep nfs ?Q ?R"
  from ‹¬(SN(qrstep nfs Q R))› obtain s' where "¬ (SN_on ?qr {s'})"
    unfolding SN_on_def by auto
  then obtain s where "s' ⊵  s" and "s ∈ Tinf ?qr" using not_SN_imp_subt_Tinf by auto
  from ‹wf_qtrs nfs Q R›
  have wwf: "wwf_qtrs Q R ∨ wwf_inn_qtrs nfs Q R" unfolding wf_qtrs_def by auto
  from wwf_inn_qtrs_imp_left_fun[OF wwf]
  have nvar: "∀ (l,r) ∈ R. is_Fun l" by auto    
  let ?I = "λi. fst(concrete_seq nfs Q R s i)"
  let ?S = "λi. fst(snd(concrete_seq nfs Q R s i))"
  let ?T = "λi. fst(snd(snd(concrete_seq nfs Q R s i)))"
  let ?σ = "λi. snd(snd(snd(snd(concrete_seq nfs Q R s i))))"
  let ?P = "λs l r u σ. (l,r) ∈ DP ♯ (applicable_rules Q R) ∧ r ⋅ σ = ♯ u
    ∧ u ∈ Tinf ?qr ∧ (♯ s,l ⋅ σ) ∈ ?Iqr⇧* ∧ l ⋅ σ ∈ NF_terms ?Q ∧ NF_subst True (l,r) σ Q"
  let ?P' =  "λ(l,r,u,σ). ?P s l r u σ"
  have main: "∀i. ?P (?I i) (?S i) (?T i) (?I(Suc i)) (?σ i)"
  proof
    fix i show "?P (?I i) (?S i) (?T i) (?I(Suc i)) (?σ i)"
    proof (induct i)
      case 0 
      have "∃l r u σ. ?P'(l,r,u,σ)"
        using Tinf_lemma_DP[OF ‹wf_qtrs nfs Q R› ‹s ∈ Tinf ?qr› ndef] unfolding split by auto
      then have "?P'(next4 nfs Q R s)" unfolding next4.simps using someI_ex4[where P = "?P'"] by best
      have ini: "?I 0 = s" unfolding concrete_seq.simps by simp
      have s: "?S 0 = fst(next4 nfs Q R s)" unfolding concrete_seq.simps by simp
      have t: "?T 0 = fst(snd(next4 nfs Q R s))" unfolding concrete_seq.simps by simp
      have sub: "?σ 0 = snd(snd(snd(next4 nfs Q R s)))"
        unfolding concrete_seq.simps Let_def split_def by auto
      have nxt: "?I(Suc 0) = fst(snd(snd(next4 nfs Q R s)))"
        unfolding concrete_seq.simps Let_def split_def by auto
      show ?case unfolding ini s t sub nxt using ‹?P'(next4 nfs Q R s)› by auto
    next
      case (Suc i)
      let ?P'' = "λ(w,x,y,z). ?P (?I(Suc i)) w x y z"
      from Suc have "?I (Suc i) ∈ Tinf ?qr" by simp
      have "∃l r u σ. ?P''(l,r,u,σ)"
        using Tinf_lemma_DP[OF ‹wf_qtrs nfs Q R› ‹?I (Suc i) ∈ Tinf ?qr› ndef] by auto
      then have IH: "?P''(next4 nfs Q R (?I(Suc i)))"
        unfolding next4.simps using someI_ex4[where P = "?P''"] by simp
      have s: "?S(Suc i) = fst(next4 nfs Q R (?I(Suc i)))"
        unfolding concrete_seq.simps Let_def split_def by simp
      have t: "?T(Suc i) = fst(snd(next4 nfs Q R (?I(Suc i))))"
        unfolding concrete_seq.simps Let_def split_def by simp
      have sub: "?σ(Suc i) = snd(snd(snd(next4 nfs Q R (?I(Suc i)))))"
        unfolding concrete_seq.simps Let_def split_def by simp
      have nxt: "?I(Suc(Suc i)) = fst(snd(snd(next4 nfs Q R (?I(Suc i)))))"
        unfolding concrete_seq.simps Let_def split_def by simp
      show ?case unfolding s t sub nxt using IH by auto
    qed
  qed
  from main have P: "∀i. (?S i,?T i) ∈ DP ♯ (applicable_rules Q R)" by simp
  from main have NF: "⋀i. (?S i ⋅ ?σ i) ∈ NF_terms ?Q" by auto
  from main have nfs: "⋀i. NF_subst nfs (?S i, ?T i) (?σ i) Q" unfolding NF_subst_def by blast
  from main have eq: "⋀i. ♯ (?I(Suc i)) = ?T i ⋅ ?σ i" by auto
  have R: "∀i. (?T i ⋅ ?σ i,?S(Suc i) ⋅ ?σ(Suc i)) ∈ ?Iqr⇧*"
  proof
    fix i show "(?T i ⋅ ?σ i,?S(Suc i)⋅?σ(Suc i)) ∈ ?Iqr⇧*"
    proof (cases i)
      case 0
      from main have first: "?T 0 ⋅ ?σ 0 = ♯ (?I(Suc 0))" by best
      from main have "(♯ (?I(Suc 0)),?S(Suc 0)⋅?σ(Suc 0)) ∈ ?Iqr⇧*" by best
      then show ?thesis unfolding 0 unfolding first by simp
    next
      case (Suc i) show ?thesis unfolding eq[symmetric] using main by best
    qed
  qed
  have Tinf: "∀i. ?I i ∈ Tinf ?qr"
  proof
    fix i show "?I i ∈ Tinf ?qr" using main by (induct i) (auto simp: ‹s ∈ Tinf ?qr›)
  qed
  {
    fix i 
    from Tinf have Tinf: "?I (Suc i) ∈ Tinf ?qr" by best
    from main have "(?S i, ?T i) ∈ DP ♯ (applicable_rules Q R)" by auto
    then obtain f ts where ti: "?T i = Fun (♯ f) ts" and "defined (applicable_rules Q R) (f,length ts)" unfolding DP_on_def by auto
    then obtain ts where id: "?T i ⋅ ?σ i = Fun (♯ f) ts" and d: "defined (applicable_rules Q R) (f,length ts)"  by auto
    from eq id have eq: "♯ (?I(Suc i)) = Fun (♯ f) ts" by simp
    from eq obtain g where "?I (Suc i) = Fun g ts" by (cases "?I (Suc i)", auto)
    from Tinf[unfolded this] have "Fun g ts ∈ Tinf ?qr" .
    from this[unfolded Tinf_def] have SN: "⋀ t. t ∈ set ts ⟹ SN_on ?qr {t}" by auto
    then have "⋀ t. t ∈ set (args (?T i ⋅ ?σ i)) ⟹ SN_on ?qr {t}" unfolding id by simp
  } note SN_args = this
  have RR: "?Iqr⇧* ⊆ ?qr⇧*" by (rule rtrancl_mono, unfold qrstep_iff_rqrstep_or_nrqrstep, auto)
  from R have R: "∀ i. (?T i ⋅ ?σ i,?S(Suc i) ⋅ ?σ(Suc i)) ∈ ?qr⇧*" using RR by blast
  note idd = initial_dpp.simps
  from P and R and NF nfs have ichain: "ichain (initial_dpp nfs m Q R) ?S ?T ?σ" 
    by (auto simp: ichain.simps)
  show ?thesis
  proof (intro exI, unfold idd min_ichain.simps, unfold idd[symmetric], rule conjI[OF ichain], rule impI)
    assume m
    then have ndef2: "∀ f n. defined (applicable_rules Q R) (f,n) ⟶ ¬ defined (applicable_rules Q R) (♯ f, n)"
      unfolding m .
    have SN_t: "∀i. SN_on ?qr {?T i ⋅ ?σ i}"
    proof
      fix i
      from Tinf have Tinf: "?I (Suc i) ∈ Tinf ?qr" by best
      from main have "(?S i, ?T i) ∈ DP ♯ (applicable_rules Q R)" by auto
      then obtain f ts where "?T i = Fun (♯ f) ts" and "defined (applicable_rules Q R) (f,length ts)" unfolding DP_on_def by auto
      then obtain ts where id: "?T i ⋅ ?σ i = Fun (♯ f) ts" and d: "defined (applicable_rules Q R) (f,length ts)"  by auto
      show "SN_on ?qr {?T i ⋅ ?σ i}" unfolding id 
      proof (rule SN_args_imp_SN[OF _ nvar ndef2[THEN spec, THEN spec, THEN mp[OF _ d]]])
        fix t
        assume t: "t ∈ set ts"
        with SN_args[of t i, unfolded id]
        show "SN_on ?qr {t}" by auto
      qed
    qed
    then show "minimal_cond nfs Q ({} ∪ R) ?S ?T ?σ"
      unfolding minimal_cond_def by simp
  qed
qed

corollary not_SN_imp_ichain:
  assumes prem: "wf_qtrs nfs Q R" "¬ SN (qrstep nfs Q R)" 
  "⋀ f n ss. defined (applicable_rules Q R) (f,n) ⟹ length ss = n ⟹ Fun (♯ f) ss ∉ Q"
  shows "∃s t σ. ichain (initial_dpp nfs m Q R) s t σ"
proof -
  obtain mm where mm: "mm = (∀f n. defined (applicable_rules Q R) (f, n) ⟶
                    ¬ defined (applicable_rules Q R) (♯ (f), n))" by auto
  from not_SN_imp_min_ichain[OF prem mm] 
  have "∃s t σ. min_ichain (initial_dpp nfs mm Q R) s t σ" .
  then show ?thesis unfolding initial_dpp.simps min_ichain.simps ichain.simps by blast
qed

corollary not_SN_imp_ichain_rstep:
  fixes R :: "('f, 'v) trs"
  assumes wf: "wf_trs R" and nSN: "¬ SN (rstep R)" 
  shows "∃ s t σ. ichain (nfs,m,DP ♯ R, {}, {}, {}, R) s t σ"
proof -
  let ?DP = "DP ♯ (applicable_rules {} R)"
  have U: "applicable_rules {} R = R" unfolding applicable_rules_def applicable_rule_def by auto
  from wf U have wf: "wf_qtrs nfs {} R" unfolding wf_qtrs_def wwf_qtrs_wf_trs by auto
  from nSN have nSN: "¬ SN (qrstep nfs {} R)" by auto
  have id: "initial_dpp nfs m {} R = (nfs,m,DP ♯ R, {}, {}, {}, R)" by (simp add: applicable_rules_def applicable_rule_def)
  from not_SN_imp_ichain[OF wf nSN, of m, unfolded id U] show ?thesis by auto
qed

(* completeness lemma *)
lemma SN_imp_finite_dpp_simple:
  fixes R :: "('f, 'v) trs"
  assumes SN: "SN (qrstep nfs Q R)"  
    and shp: "⋀ f n. unshp ≠ id ⟹ (f, n) ∈ D ⟹ ¬ defined R (♯ f, n)"
    and unshp: "⋀ f n. (f, n) ∈ D ⟹ unshp (♯ f) = f"
    and D: "⋀ fn. defined R fn ⟹ fn ∈ D"
    and nfs: "nfs ⟹ ∀ l r. (l, r) ∈ R ⟶ is_Fun l"
  shows "finite_dpp (nfs, m, DP_simple ♯ D R, {}, Q, {}, R)"
proof (rule ccontr)
  let ?DP = "DP_simple ♯ D R"
  let ?u = "sharp_term unshp :: ('f,'v)term ⇒ ('f,'v)term"
  let ?r = "qrstep nfs Q R ∪ {⊳}"
  assume "¬ ?thesis"
  then obtain s t σ where "ichain (nfs,m,?DP,{},Q,{},R) s t σ" 
    unfolding finite_dpp_def by auto
  note ichain = this[unfolded ichain.simps]
  {
    fix l r x
    assume lr: "(l,r) ∈ R" and l: "l = Var x" 
    have False
    proof (cases nfs)
      case False
      from left_Var_imp_not_SN_qrstep[OF lr[unfolded l] False, of Q]
        SN show False unfolding SN_def by auto
    next
      case True
      with nfs[OF this] lr l show False by auto
    qed
  } note nvar = this
  then have varcond: "∀ (l,r) ∈ R. is_Fun l" by force
  {
    fix i
    from ichain have "(s i, t i) ∈ ?DP" by auto
    from this[unfolded DP_simple_def] obtain l r h us where s: "s i = ♯ l"
      and t: "t i = ♯ (Fun h us)" and lr: "(l,r) ∈ R" and ru: "r ⊵ Fun h us"
      and h: "(h,length us) ∈ D" by auto
    from nvar[OF lr]
    obtain f ls where l: "l = Fun f ls" by (cases l, auto)
    from lr[unfolded l] have f: "defined R (f, length ls)" unfolding defined_def by auto
    from ru obtain C where r: "r = C⟨Fun h us⟩" by auto
    have si: "?u (s i ⋅ σ i) =  l ⋅ σ i" unfolding s l using unshp[OF D[OF f]]
      by simp
    define τ where "τ = (λ x. if x ∈ vars_term (s i) ∪ vars_term (t i) then σ i x else s 0 ⋅ σ 0)" 
    have lτ: "l ⋅ τ = l ⋅ σ i" 
      by (rule term_subst_eq, insert s, auto simp: τ_def)
    have h_us_τ: "(Fun h us) ⋅ τ = (Fun h us) ⋅ σ i" 
      by (rule term_subst_eq, insert t, auto simp: τ_def)
    from ichain have "s i ⋅ σ i ∈ NF_terms Q" and nfs: "NF_subst nfs (s i, t i) (σ i) Q"
      by auto
    from NF_imp_subt_NF[OF this(1)]
    have "∀ t ∈ set (args (s i ⋅ σ i)). t ∈ NF_terms Q"
      unfolding NF_terms_args_conv .
    then have "∀ t ∈ set (args (l ⋅ σ i)). t ∈ NF_terms Q" unfolding s l by simp
    then have NF: "∀ t ⊲ l ⋅ τ. t ∈ NF_terms Q" unfolding lτ NF_terms_args_conv .
    {
      assume nfs
      with nfs[unfolded NF_subst_def] have nfs: 
        "σ i ` vars_term (s i)  ⊆ NF_terms Q" 
        "σ i ` vars_term (t i)  ⊆ NF_terms Q"
        unfolding vars_rule_def by auto
      then have tau: "τ x ∈ NF_terms Q" for x unfolding τ_def using ichain by auto
      then have "NF_subst nfs (l,r) τ Q" by auto
    }
    then have nfs: "NF_subst nfs (l,r) τ Q" by (cases nfs, auto)
    have step: "(l ⋅ σ i, r ⋅ τ) ∈ qrstep nfs Q R" unfolding lτ[symmetric]
      by (rule qrstepI[OF _ lr, of τ _ _ □, OF NF], auto simp: nfs)
    have supt: "(r ⋅ τ, Fun h us ⋅ σ i) ∈ {⊳}⇧=" unfolding h_us_τ[symmetric] 
      using supteq_subst[OF ru, of τ] unfolding supteq_supt_set_conv .
    from ichain have steps: "(t i ⋅ σ i, s (Suc i) ⋅ σ (Suc i)) ∈ (qrstep nfs Q R)⇧*" by auto    
    have "(?u (t i ⋅ σ i), ?u (s (Suc i) ⋅ σ (Suc i))) ∈ (qrstep nfs Q R)⇧*" 
    proof (cases "unshp = id")
      case False
      have steps: "(t i ⋅ σ i, s (Suc i) ⋅ σ (Suc i)) ∈ (nrqrstep nfs Q R)⇧*"
      proof (rule qrsteps_imp_nrqrsteps[OF varcond ndef_applicable_rules steps])
        show "¬ defined R (the (root (t i ⋅ σ i)))"
          unfolding t using shp False h by auto
      qed
      from nrqrsteps_imp_sharp_qrsteps[OF steps, of unshp]
      show ?thesis
        unfolding qrstep_iff_rqrstep_or_nrqrstep by regexp
    next
      case True
      from steps show ?thesis unfolding True sharp_term_id .
    qed
    then have steps: "(Fun h us ⋅ σ i, ?u (s (Suc i) ⋅ σ (Suc i))) ∈ (qrstep nfs Q R)⇧*"
      unfolding t using unshp[OF h] by simp
    from si step supt steps have steps: "(?u (s i ⋅ σ i), ?u (s (Suc i) ⋅ σ (Suc i))) ∈
      qrstep nfs Q R O {⊳}^= O (qrstep nfs Q R)⇧*" by auto
    have "(?u (s i ⋅ σ i), ?u (s (Suc i) ⋅ σ (Suc i))) ∈ ?r^+"
      by (rule set_mp[OF _ steps], regexp)
  } note main = this
  obtain u where u: "u = (λ i. ?u (s i ⋅ σ i))" by auto
  from main have "⋀ i. (u i, u (Suc i)) ∈ ?r^+" unfolding u .
  then have "¬ SN (?r^+)" by auto
  then have "¬ SN ?r" using SN_imp_SN_trancl[of ?r] by auto
  then show False using SN_imp_SN_union_supt[OF SN ctxt_closed_qrstep] ..
qed

lemma SN_imp_finite_dpp:
  fixes R :: "('f, 'v) trs"
  assumes SN: "SN (qrstep nfs Q R)"
    and shp: "⋀ f n. unshp ≠ id ⟹ defined R (f, n) ⟹ ¬ defined R (♯ f, n)"
    and unshp: "⋀ f n. defined R (f,n) ⟹ unshp (♯ f) = f"
    and nfs: "nfs ⟹ ∀ l r. (l, r) ∈ R ⟶ is_Fun l"
  shows "finite_dpp (nfs, m, DP ♯ R, {}, Q, {}, R)"
by (rule finite_dpp_mono[OF SN_imp_finite_dpp_simple[OF SN shp unshp _ nfs, of "{fn . defined R fn}"]])
   (auto simp: DP_on_subset_DP_simple)

end
