theory Tcap

begin

no_notation Subterm_and_Context.Hole ("□")
notation Ground_Context.GCHole ("□")
notation Ground_Context.equiv_class ("[_]")

(* a semantic version of tcap which is not executable *)
fun tcap :: "('f, 'v) trs ⇒ ('f, 'v) term  ⇒ ('f, 'v) gctxt" where
  "tcap _ (Var _)    = □"
| "tcap R (Fun f ts) = (
    let h = GCFun f (map (tcap R) ts) in
    if (∃r ∈ R. Ground_Context.match h (fst r)) then □ else h)"

lemma tcap_sound_main:
  assumes "(s ⋅ σ, t) ∈ (rstep R)⇧*" and p: "p ∈ pos_gctxt (tcap R s)" 
  shows "t ∈ [tcap R s] ∧ (s ⋅ σ |_ p, t |_p) ∈ (rstep R)⇧*"
using assms
proof (induct s arbitrary: t p rule: term.induct)
  case (Fun f ss)
  let ?h = "GCFun f (map (tcap R) ss)" 
  show ?case 
  proof (cases "tcap R (Fun f ss)")
    case (GCFun g hs)
    then have res: "tcap R (Fun f ss) = ?h"
      by (cases "∃ r ∈ R. Ground_Context.match ?h (fst r)", auto simp: Let_def)
    {
      fix u
      assume steps: "(Fun f ss ⋅ σ, u) ∈ (nrrstep R)⇧*"
      then have cond: "root (Fun f ss ⋅ σ) = root u ∧ (∀i<num_args (Fun f ss ⋅ σ). (Fun f ss ⋅ σ |_ [i], u |_ [i] ) ∈ (rstep R)⇧*)"	
        by (rule nrrsteps_imp_eq_root_arg_rsteps)
      then have "root (Fun f ss) = root u" by simp
      then have "∃ us. u = Fun f us" by (cases u, auto)
      from this obtain us where u: "u = Fun f us" by force
      with cond have len: "length ss = length us" by simp
      {
        fix i p
        assume ilen: "i < length us" and p: "p ∈ pos_gctxt (tcap R (Fun f ss |_ [i]))"
        with len have "[i] ∈ poss (Fun f ss)" by simp
        then have "Fun f ss ⋅ σ|_[i] = Fun f ss |_[i] ⋅ σ" by simp
        with ilen len cond have one: "(Fun f ss |_[i] ⋅ σ, u|_[i]) ∈ (rstep R)⇧*" by simp	
        from ilen len have "Fun f ss |_[i] ∈ set ss" by simp
        from Fun(1)[OF this one p] u
        have "us ! i ∈ [tcap R (ss ! i)]" and "(ss ! i ⋅ σ |_ p, us ! i |_ p) ∈ (rstep R)⇧*" by auto
      } note IH = this
      from IH[OF _ pos_gctxt_epsilon] len
      have "Fun f us ∈ [?h]" by auto
      with u have uh: "u ∈ [?h]" by simp
      have "(Fun f ss ⋅ σ |_ p, u |_p ) ∈ (rstep R)⇧*"
      proof (cases p)
        case (Cons i q)
        from Fun(3) len have i: "i < length us" and "q ∈ pos_gctxt (tcap R (Fun f ss |_ [i]))" unfolding Cons
          by (auto simp: Let_def split: if_splits)
        from IH(2)[OF this] have "(ss ! i ⋅ σ |_ q, us ! i |_ q) ∈ (rstep R)⇧*" .
        then show ?thesis unfolding Cons u using i len by simp
      next
        case Nil
        from steps have "(Fun f ss ⋅ σ, u) ∈ (rstep R)⇧*" unfolding rstep_iff_rrstep_or_nrrstep by regexp
        with Nil show ?thesis by simp
      qed 
      note uh this
    } note nrr = this    
    have "rstep R = rrstep R ∪ nrrstep R" by (rule rstep_iff_rrstep_or_nrrstep)
    from this and ‹(Fun f ss ⋅ σ, t) ∈ (rstep R)⇧*›
    have  "(Fun f ss ⋅ σ, t) ∈ (nrrstep R)⇧* ∨ (Fun f ss ⋅ σ, t) ∈ (nrrstep R)⇧* O (rrstep R) O (rstep R)⇧*" by (rule firstStep)
    then show ?thesis
    proof 
      assume "(Fun f ss ⋅ σ, t) ∈ (nrrstep R)⇧*"
      with nrr res show ?thesis by auto
    next
      assume "(Fun f ss ⋅ σ, t) ∈ (nrrstep R)⇧* O rrstep R O (rstep R)⇧*"
      from this obtain u v where fu: "(Fun f ss ⋅ σ, u) ∈ (nrrstep R)⇧*" and uv: "(u,v) ∈ rrstep R" by force
      from fu nrr have uh: "u ∈ [?h]" by simp
      from uv have "∃ l r σ. (l,r) ∈ R ∧ l ⋅ σ = u ∧ r ⋅ σ = v" by (rule rrstep_imp_rule_subst)
      from this obtain l r σ where lrR: "(l, r) ∈ R" and match: "l ⋅ σ = u" by force
      from uh and match  have "Ground_Context.match ?h l" by (unfold Ground_Context.match_def, auto)
      with lrR have tcap: "tcap R (Fun f ss) = □" by (simp add: Let_def, force)
      with Fun(3) have p: "p = []" by simp
      from tcap p Fun(2) show ?thesis by simp
    qed
  qed (insert Fun, auto)
qed simp

lemma tcap_sound:
  assumes steps: "(s ⋅ σ, t) ∈ (rstep R)⇧*" shows "t ∈ [tcap R s]"
  using tcap_sound_main[OF steps pos_gctxt_epsilon] by simp

lemma tcap_instance_equiv_class[simp]: "s ⋅ σ ∈ [tcap R s]"
  by (rule tcap_sound, auto)

lemma tcap_subterm_subst_rsteps:
  assumes steps: "(s ⋅ σ, t) ∈ (rstep R)⇧*" and p: "p ∈ pos_gctxt (tcap R s)" 
  shows "(s ⋅ σ |_ p, t |_ p) ∈ (rstep R)⇧*"
  using tcap_sound_main[OF steps p] by simp

lemma tcap_subterm_rsteps:
  assumes steps: "(s, t) ∈ (rstep R)⇧*" and p: "p ∈ pos_gctxt (tcap R s)" 
  shows "(s |_ p, t |_ p) ∈ (rstep R)⇧*"
  using tcap_subterm_subst_rsteps[OF _ p, of Var] steps by simp

lemma tcap_instance_subset: "[tcap R (t ⋅ σ)] ⊆ [tcap R t]" 
proof (induct t, simp)
  case (Fun f ts)
  let ?h = "GCFun f (map (tcap R) ts)"
  let ?hs = "GCFun f (map (tcap R) (map (λ t. (t ⋅ σ)) ts))"
  let ?hss = "GCFun f (map (λ t. tcap R (t ⋅ σ)) ts)"
  show ?case 
  proof (cases "∃ r ∈ R. Ground_Context.match ?h (fst r)", simp)
    case False
    then have nMatch: "∀ r ∈ R. ¬ Ground_Context.match ?h (fst r)" by auto
    {
      fix l
      assume "Ground_Context.match ?hs l"
      from this obtain δ where "l ⋅ δ ∈ [?hs]" unfolding Ground_Context.match_def by auto
      then have "Ground_Context.match ?h l" unfolding Ground_Context.match_def using Fun by force
    }
    with nMatch have "∀ r ∈ R. ¬ Ground_Context.match ?hs (fst r)" by auto
    with ‹¬ (∃ r ∈ R. Ground_Context.match ?h (fst r))›
    have id: "tcap R (Fun f ts) = ?h ∧ tcap R (Fun f ts ⋅ σ) = ?hss" by auto
    from Fun have "[?hss] ⊆ [?h]" by (force simp: nth_mem)
    with id show ?thesis by simp
  qed
qed

lemma tcap_refl: "t ∈ [tcap R t]"
  by  (rule tcap_sound[of _ Var], simp)

lemma tcap_lhs_var:
  assumes left_var: "∃lr ∈ R. is_Var (fst lr)" 
  shows "tcap R s = □"
proof -
  from left_var obtain x r where lr: "(Var x,r) ∈ R" by auto
  show ?thesis
  proof (induct s)
    case (Fun f ts)
    show ?case
      unfolding tcap.simps Let_def    
      by (clarsimp, rule bexI[OF _ lr], simp add: Ground_Context.match_def,
          rule exI[of _ "(λ_. Fun f ts)"], simp add: tcap_refl)
  qed auto
qed

lemma tcap_lhs:
  assumes "(l, r) ∈ R"
  shows "tcap R (l ⋅ σ) = □"
proof (cases l)
  case (Fun f ts)
  show ?thesis
  proof (unfold Fun, simp add: Let_def o_def, intro bexI[OF _ assms], simp add: Fun)
    show "Ground_Context.match (GCFun f (map (λ x. tcap R (x ⋅ σ)) ts)) (Fun f ts)"
      by  (unfold Ground_Context.match_def, rule exI[of _ σ], unfold equiv_class.simps, simp add: tcap_refl)
  qed
next
  case (Var x)
  show ?thesis
    by (rule tcap_lhs_var, rule bexI[OF _ assms], auto simp: Var)
qed

lemma match_lhs_var:
  assumes left_var: "∃lr ∈ R. is_Var (fst lr)" 
  shows "Ground_Context.match (tcap R s) t"
  unfolding tcap_lhs_var[OF left_var] Ground_Context.match_def by auto

abbreviation
  tcap_below :: "('f, 'v) trs ⇒ 'f ⇒ ('f, 'v) term list ⇒ ('f, 'v) gctxt"
where
  "tcap_below R f ts ≡ GCFun f (map (tcap R) ts)"

fun match_tcap_below :: "('f, 'v) term ⇒ ('f, 'v) trs ⇒ ('f, 'v) term ⇒ bool" where
  "match_tcap_below l R (Fun f ts) = Ground_Context.match (tcap_below R f ts) l"
| "match_tcap_below l R (Var x) = False"

lemma match_below:
  assumes match: "Ground_Context.match (GCFun f (bef @ s # aft)) l" (is "Ground_Context.match ?s l")
    and subset: "[s] ⊆ [t]"
  shows "Ground_Context.match (GCFun f (bef @ t # aft)) l" (is "Ground_Context.match ?t l")
proof -
  from subset have "[?s] ⊆ [?t]" by (rule equiv_class_mono)
  with match show ?thesis unfolding Ground_Context.match_def by blast
qed

lemma tcap_rewrite:
  fixes t :: "('f, 'v) term"
  assumes "(t, s) ∈ rstep R"
  shows "[tcap R s] ⊆ [tcap R t]"
using assms
proof (induct)
  fix C σ l r assume "(l, r) ∈ R"
  then show "[tcap R C⟨r ⋅ σ⟩] ⊆ [tcap R C⟨l ⋅ σ⟩]"
  proof (induct C)
    case (More f bef D aft)
    then have ind: "[tcap R D⟨r ⋅ σ⟩] ⊆ [tcap R D⟨l ⋅ σ⟩]" by auto
    let ?h = "λt. GCFun f (map (tcap R) bef @ tcap R D⟨t⟩ # map (tcap R) aft)"
    from ind have res: "⋀t. Ground_Context.match (?h (r ⋅ σ)) t ⟹ Ground_Context.match (?h (l ⋅ σ)) t"
      using match_below by force
    show ?case
    proof (cases "∃lr∈R. Ground_Context.match (?h (l ⋅ σ)) (fst lr)")
      case False 
      then have one: "¬ (∃ lr ∈ R. Ground_Context.match (?h (r ⋅ σ)) (fst lr))" using res by blast
      from ind have subset: "[?h (r ⋅ σ)] ⊆ [?h (l ⋅ σ)]" by (rule equiv_class_mono)
      with one show ?thesis by (auto simp: Let_def)
    qed simp
  next
    case Hole
    have "tcap R (l ⋅ σ) = □"
    proof (cases l)
      case (Var x)
      {
        fix h
        from equiv_class_nonempty obtain s :: "('f, 'v) term" where "s ∈ [h]" by auto
        with Var have "l ⋅ subst x s ∈ [h]" unfolding subst_def by auto
        then have "Ground_Context.match h (fst (l,r))" unfolding Ground_Context.match_def by (simp, blast)
      }
      with ‹(l, r) ∈ R› show ?thesis by (cases "l ⋅ σ", auto simp: Let_def, blast)
    next
      case (Fun f ll)
      then have one: "l ⋅ σ = Fun f (map (λ u. u ⋅ σ) ll)" (is "_ = Fun f ?ls") by auto
      {
        fix i 
        have "ll ! i ⋅ σ ∈ [tcap R (ll ! i ⋅ σ)]"
          using tcap_sound[OF rtrancl_refl, of "ll!i ⋅ σ" "Var"] by simp
      }
      with one have "(fst (l,r)) ⋅ σ ∈ [GCFun f (map (λx. tcap R (x ⋅ σ)) ll)]" by auto
      with one and ‹(l, r) ∈ R›
      show ?thesis by (auto simp: Let_def o_def, unfold Ground_Context.match_def, blast)
    qed
    then show "[tcap R Hole⟨r ⋅ σ⟩] ⊆ [tcap R Hole⟨l ⋅ σ⟩]" by simp
  qed
qed

lemma tcap_rewrites:
  assumes  "(s, t) ∈ (rstep R)⇧*" 
  shows "[tcap R t] ⊆ [tcap R s]"
using assms
proof (induct)
  case (step t u)
  from step(3) tcap_rewrite[OF step(2)] show ?case by auto
qed simp

lemma match_tcap_sound: 
  assumes derivation: "(t ⋅ δ, s ⋅ σ) ∈ (rstep R)⇧*"
  shows "Ground_Context.match (tcap R t) s"
proof -
  from derivation have "s ⋅ σ ∈ [tcap R t]" by (rule tcap_sound)
  then show ?thesis by (unfold Ground_Context.match_def, auto)
qed

lemma match_tcap_inv_sound: 
  assumes derivation: "(t ⋅ δ, s ⋅ σ) ∈ (rstep R)⇧*"
  shows "Ground_Context.match (tcap (R¯) s) t"
proof -
  have id: "((rstep R)⇧*)¯ = (rstep (R¯))⇧*"
    unfolding rtrancl_converse [symmetric] and rstep_converse ..
  from derivation have "(s ⋅ σ, t ⋅ δ) ∈ ((rstep R)⇧*)¯" by auto
  from match_tcap_sound[OF this[unfolded id]]
  show "Ground_Context.match (tcap (R¯) s) t" .
qed

lemma join_imp_unifiable_tcaps:
  assumes "(s ⋅ σ, t ⋅ τ) ∈ (rstep R)⇧↓"
  shows "Ground_Context.unifiable (tcap R s) (tcap R t)"
proof -
  obtain u where "(s ⋅ σ, u) ∈ (rstep R)⇧*" and "(t ⋅ τ, u) ∈ (rstep R)⇧*" using assms by auto
  then have "u ∈ [tcap R s]" and "u ∈ [tcap R t]" by (blast intro: tcap_sound)+
  then show ?thesis by (auto simp: Ground_Context.unifiable_def)
qed

lemma match_tcap_below: assumes steps: "(s, l ⋅ σ) ∈ (nrrstep R)⇧*"
  and l: "is_Fun l"
  shows "match_tcap_below l R s"
proof -
  from l obtain f ls where l: "l = Fun f ls" by (cases l, auto)
  note steps = nrrsteps_imp_eq_root_arg_rsteps[OF steps, unfolded l]
  from steps[THEN conjunct1] obtain ss where s: "s = Fun f ss" and len: "length ls = length ss" by (cases s, auto)
  {
    fix i
    assume "i < length ss"
    with steps[unfolded s] have steps: "(ss ! i, ls ! i ⋅ σ) ∈ (rstep R)⇧*" by auto
    have "ls ! i ⋅ σ ∈ [tcap R (ls ! i ⋅ σ)]" by (rule tcap_refl)
    also have "[tcap R (ls ! i ⋅ σ)] ⊆ [tcap R (ss ! i)]" using tcap_rewrites[OF steps] by auto
    finally have "ls ! i ⋅ σ ∈ [tcap R (ss ! i)]" .
  }
  then show ?thesis unfolding l s
    by (auto simp: len Ground_Context.match_def intro!: exI[of _ σ])
qed

no_notation Ground_Context.equiv_class ("[_]")
