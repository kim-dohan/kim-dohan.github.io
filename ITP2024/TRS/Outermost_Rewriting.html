theory Outermost_Rewriting
imports
  Trs
  Context_Substitution
  "HOL-Library.Monad_Syntax"
begin

definition ostep_cond_single :: "('f,'v)term ⇒ pos ⇒ ('f,'v)term ⇒ bool"
  where "ostep_cond_single l p t ≡ ∀ q. q <⇩p p ⟶ ¬ (∃ σ. t |_ q = l ⋅ σ)"

definition ostep_cond :: "('f,'v)terms ⇒ pos ⇒ ('f,'v)term ⇒ bool"
  where "ostep_cond Q p t ≡ ∀ l ∈ Q. ostep_cond_single l p t"

definition ostep_p :: "('f,'v)terms ⇒ ('f,'v)trs ⇒ pos ⇒ ('f,'v)trs" 
  where "ostep_p Q R p ≡ {(s,t). (∃ lr σ. (s,t) ∈ rstep_r_p_s R lr p σ) ∧ ostep_cond Q p s}"

definition ostep :: "('f,'v)terms ⇒ ('f,'v)trs ⇒ ('f,'v)trs"
  where "ostep Q R ≡ {(s,t). ∃ p. (s,t) ∈ ostep_p Q R p}"

lemmas ostep_defs = ostep_def ostep_p_def

lemma ostep_pI: "(s,t) ∈ rstep_r_p_s R lr p σ ⟹ ostep_cond Q p s ⟹ (s,t) ∈ ostep_p Q R p" 
  unfolding ostep_defs by blast

lemma ostep_pE[elim]: "(s,t) ∈ ostep_p Q R p ⟹ ⟦⋀ σ lr. (s,t) ∈ rstep_r_p_s R lr p σ ⟹ ostep_cond Q p s ⟹ P⟧ ⟹ P"
  unfolding ostep_defs by blast

lemma ostep_mono: assumes R: "R ⊆ R'" 
  shows "ostep Q R ⊆ ostep Q R'"
  unfolding ostep_defs rstep_r_p_s_def Let_def using R by blast

end
