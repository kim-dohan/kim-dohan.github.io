theory Forbidden_Patterns
imports
  Trs
  "HOL-Library.Monad_Syntax"
  Auxx.Shows_Literal
begin

hide_const (open) FuncSet.Pi

datatype location = H | A | B | R

type_synonym ('f,'v)forb_pattern = "('f,'v)ctxt × ('f,'v)term × location"
type_synonym ('f,'v)forb_patterns = "('f,'v)forb_pattern set"

definition fpstep_cond_single :: "('f,'v) forb_pattern ⇒ pos ⇒ ('f,'v)term ⇒ bool"
where "fpstep_cond_single pt p t ≡ case pt of (C,u,loc) ⇒
  ¬(∃ C' σ. t = C'⟨C⟨u⟩ ⋅ σ ⟩  ∧ 
   ((loc = H ∧ p = (hole_pos C') @ (hole_pos C)) ∨ 
    (loc = A ∧ p <⇩p (hole_pos C') @ (hole_pos C)) ∨ 
    (loc = B ∧ hole_pos C' @ hole_pos C <⇩p p) ∨ 
    (loc = R ∧ right_of_pos p (hole_pos C'))))"

definition fpstep_cond :: "('f,'v) forb_patterns ⇒ pos ⇒ ('f,'v)term ⇒ bool"
where "fpstep_cond Pi p t ≡ ∀ pt ∈ Pi. fpstep_cond_single pt p t"

definition fpstep_r_p_s :: 
 "('f,'v) forb_patterns ⇒ ('f,'v)trs ⇒ ('f, 'v) rule ⇒ pos ⇒ ('f, 'v) subst ⇒ 
  ('f,'v)term ⇒ ('f,'v)term ⇒ bool"
 where "fpstep_r_p_s Pi rs lr p σ s t ≡ (s, t) ∈ rstep_r_p_s rs lr p σ ∧ fpstep_cond Pi p s"

definition fpstep_p :: "('f,'v) forb_patterns ⇒ ('f,'v)trs ⇒ pos ⇒ ('f,'v)trs"
where "fpstep_p Pi rs p = {(s,t) | s t.(∃ lr σ. (s,t) ∈ rstep_r_p_s rs lr p σ ∧ fpstep_cond Pi p s)}"

definition fpstep :: "('f,'v) forb_patterns ⇒ ('f,'v)trs  ⇒ ('f,'v)trs"
where "fpstep Pi rs = {(s,t). ∃ p. (s,t) ∈  fpstep_p Pi rs p}"

lemmas fpstep_defs = fpstep_def fpstep_p_def

lemma fpstep_mono: "r ⊆ r' ⟹ fpstep p r ⊆ fpstep p r'"
  unfolding fpstep_defs rstep_r_p_s_def Let_def by blast

lemma fpstep_pI: "(s,t) ∈ rstep_r_p_s rs lr p σ ⟹ fpstep_cond Pi p s ⟹ (s,t) ∈ fpstep_p Pi rs p" 
  unfolding fpstep_defs by blast

lemma fpstep_pE[elim]: "(s,t) ∈ fpstep_p Pi rs p ⟹ 
 ⟦⋀ σ lr. (s,t) ∈ rstep_r_p_s rs lr p σ ⟹ fpstep_cond Pi p s ⟹ P⟧ ⟹ P"
  unfolding fpstep_defs by blast

instantiation location :: showl
begin

fun showsl_location where
  "showsl_location A = (showsl_lit (STR ''above''))"
| "showsl_location B = (showsl_lit (STR ''below''))"
| "showsl_location H = (showsl_lit (STR ''here''))"
| "showsl_location R = (showsl_lit (STR ''right''))"
definition "showsl_list (xs :: location list) = default_showsl_list showsl xs"
instance ..
end

hide_const (open) A B H R

end
