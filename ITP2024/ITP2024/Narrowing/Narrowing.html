lemma mgu_ex:
  assumes uv:"u = v ⋅ θ"
    and empty:"subst_domain θ ∩ range_vars θ = {}"
  shows "∃δ. mgu u v = Some δ"
proof -
  from subst_idemp_iff empty have idem:"θ ∘⇩s θ = θ" by auto
  with uv have "u ⋅ θ = v ⋅ θ" 
    by (simp add: subst_subst)
  with the_mgu_is_imgu[unfolded is_imgu_def]
  have "unifiers {(u, v)} ≠ {}" by blast
  then show ?thesis 
    by (simp add: ex_mgu_if_unifiers_not_empty)
qed

lemma restrict_subst_compose:
  assumes asm:"(B - subst_domain σ) ∪ range_vars σ ⊆ A"
    and θθ':"∀x. restrict_subst_domain A θ x = restrict_subst_domain A θ' x"
  shows "∀x. restrict_subst_domain B (σ ∘⇩s θ) x =  restrict_subst_domain B (σ ∘⇩s θ') x" 
proof -
  let ?θ1 = "λx. restrict_subst_domain (range_vars σ) θ x"
  let ?θ2 = "λx. restrict_subst_domain (B ∩ subst_domain σ) (σ ∘⇩s ?θ1) x"
  let ?θ3 = "λx. restrict_subst_domain (B - subst_domain σ) θ x"
  let ?θ1' = "λx. restrict_subst_domain (range_vars σ) θ' x"
  let ?θ2' = "λx. restrict_subst_domain (B ∩ subst_domain σ) (σ ∘⇩s ?θ1') x"
  let ?θ3' = "λx. restrict_subst_domain (B - subst_domain σ) θ' x"
  let ?θl = "?θ2 ∪⇩s ?θ3"
  let ?θr = "?θ2' ∪⇩s ?θ3'"
  let ?σθB = "λx. restrict_subst_domain B (σ ∘⇩s θ) x"
  let ?σθ'B = "λx. restrict_subst_domain B (σ ∘⇩s θ') x"
  from asm have "range_vars σ ⊆ A" by auto
  hence *:"∀x. (Var x) ⋅ ?θ1  = (Var x) ⋅ ?θ1'" using θθ' 
    by (metis eval_term.simps(1) in_mono restrict_subst_domain_def)
  hence **:"∀x. (Var x) ⋅ ?θ2  = (Var x) ⋅ ?θ2'" using assms by simp
  from asm have "(B - subst_domain σ) ⊆ A" by auto
  hence ***:"∀x. (Var x) ⋅ ?θ3 = (Var x) ⋅ ?θ3'" using θθ' *
    by (metis eval_term.simps(1) restrict_subst_domain_def subsetD)
  have "?θl = ?θr" using * ** *** by simp
  moreover have "∀x. (Var x) ⋅ ?σθB = (Var x) ⋅ ?θl"
  proof(auto, goal_cases)
    case (1 x)
    hence "(σ ∘⇩s θ) x = (σ ∘⇩s restrict_subst_domain (range_vars σ) θ) x"
      by (metis (no_types, opaque_lifting) Diff_eq_empty_iff Term.term.simps(17) 
          Un_Diff_Int eval_term.simps(1) inf.order_iff insert_Diff1 le_sup_iff 
          subsetI subst_apply_term_restrict_subst_domain subst_compose vars_term_subst_apply_term_subset)
    then show ?case using assms 1 by (simp add: restrict_subst_domain_def)
  next
    case (2 x)
    then show ?case 
      by (metis Diff_iff eval_term.simps(1) notin_subst_domain_imp_Var restrict_subst_domain_def subst_compose)
  next
    case (3 x)
    then show ?case by (simp add: restrict_subst_domain_def)
  next
    case (4 x)
    from notin_subst_domain_imp_Var[OF this]
    have *:"(σ ∘⇩s restrict_subst_domain (range_vars σ) θ) x = Var x" by simp
    from subst_domain_restrict_subst_domain
    have "subst_domain (restrict_subst_domain (range_vars σ) θ) = (range_vars σ) ∩ subst_domain θ" by simp
    then show ?case using * 4
    proof(cases "x ∈ B")
      case True
      then show ?thesis 
        by (smt (verit, ccfv_threshold) "*" Diff_iff mem_Collect_eq restrict_subst_domain_def 
            subst_apply_eq_Var subst_compose_def subst_domain_def subst_monoid_mult.mult.left_neutral)
    next
      case False
      then show ?thesis by (simp add: restrict_subst_domain_def)
    qed
  qed
  moreover have "∀x. (Var x) ⋅ ?θr = (Var x) ⋅ ?σθ'B"
  proof(auto, goal_cases)
    case (1 x)
    hence "(σ ∘⇩s θ') x = (σ ∘⇩s restrict_subst_domain (range_vars σ) θ') x"
      by (metis (no_types, opaque_lifting) Diff_eq_empty_iff Term.term.simps(17) 
          Un_Diff_Int eval_term.simps(1) inf.order_iff insert_Diff1 le_sup_iff 
          subsetI subst_apply_term_restrict_subst_domain subst_compose vars_term_subst_apply_term_subset)
    then show ?case using assms 1 by (simp add: restrict_subst_domain_def)
  next
    case (2 x)
    then show ?case 
      by (metis Diff_iff eval_term.simps(1) notin_subst_domain_imp_Var restrict_subst_domain_def subst_compose)
  next
    case (3 x)
    then show ?case by (simp add: restrict_subst_domain_def)
  next
    case (4 x)
    from notin_subst_domain_imp_Var[OF this]
    have *:"(σ ∘⇩s restrict_subst_domain (range_vars σ) θ') x = Var x" by simp
    from subst_domain_restrict_subst_domain
    have "subst_domain (restrict_subst_domain (range_vars σ) θ') = (range_vars σ) ∩ subst_domain θ'" by simp
    then show ?case using * 4
    proof(cases "x ∈ B")
      case True
      then show ?thesis 
        by (smt (verit, ccfv_threshold) "*" Diff_iff mem_Collect_eq restrict_subst_domain_def 
            subst_apply_eq_Var subst_compose_def subst_domain_def subst_monoid_mult.mult.left_neutral)
    next
      case False
      then show ?thesis by (simp add: restrict_subst_domain_def)
    qed
  qed
  ultimately show ?thesis by simp
qed

lemma subst_union_sound:assumes st:"s ⋅ α = t ⋅ β"
  and sα:"subst_domain α ⊆ vars_term s "
  and tβ:"subst_domain β ⊆ vars_term t"
  and disj:"vars_term s ∩ vars_term t = {}"
shows "s ⋅ (α ∪⇩s β) = t ⋅ (α ∪⇩s β)" 
proof -
  from term_subst_eq_conv[of s α "α ∪⇩s β"]
  have *:"s ⋅ α = s ⋅ (α ∪⇩s β)" using sα unfolding subst_union.simps
    by (metis disj disjoint_iff notin_subst_domain_imp_Var subsetD tβ)
  from term_subst_eq_conv[of t β "α ∪⇩s β"]
  have "t ⋅ β = t ⋅ (α ∪⇩s β)" using tβ disj sα by auto
  then show ?thesis using * st by auto
qed

lemma normal_subst__perm: assumes norm:"normal_subst R σ"
  and equiv:"∀x. σ x = σr (ω ∙ x)"
shows "normal_subst R σr" 
proof -
  from norm have "∀x. σ (-ω ∙ x) ∈ NF (rstep R)"
    by (simp add: normal_subst_def)
      (metis NF_I local.wf no_Var_rstep notin_subst_domain_imp_Var)
  moreover have "∀x. σ (ω ∙ -ω ∙ x) ∈ NF (rstep R)" 
    by (simp add: norm normal_subst_def) 
      (metis atom_pt.permute_minus_cancel(2) calculation)
  moreover have "∀x. σ (-ω ∙ x) = σr (ω ∙ -ω ∙ x)" using equiv by simp
  moreover have "∀x. σr (ω ∙ -ω ∙ x) = σr x" by simp
  ultimately show ?thesis 
    by (simp add: normal_subst_def)
qed

  shows "funas_rule (ω ∙ rl) ⊆ S"

lemma subst_app_subterm: fixes θ::"('f, 'v)subst"
  assumes "x ∈ vars_term t"
  shows "θ x ⊴ t ⋅ θ" using assms
proof(induct t)
  case (Var x)
  then show ?case by simp
next
  case (Fun f ss)
  then show ?case by (meson subst_image_subterm supt_imp_supteq)
qed

lemma NF_rstep_subterm:
  assumes "t ∈ NF (rstep R)" and "t ⊵ s"
  shows "s ∈ NF (rstep R)"
proof (rule ccontr)  
  assume "¬ ?thesis"
  then obtain u where "(s, u) ∈ rstep R" by auto
  from ‹t ⊵ s› obtain C where "t = C⟨s⟩" by auto
  with ‹(s, u) ∈ rstep R› have "(t, C⟨u⟩) ∈ rstep R" by auto
  then have "t ∉ NF (rstep R)" by auto
  with assms show False by simp
qed

(* Proposition 3.7 in MH94*)
lemma restricted_normalized: fixes A::"('v::infinite) set" and B::"('v::infinite) set" and σ θ θ'::"('f, 'v)subst"
  assumes "normal_subst R (θ |s A)"
    and "(σ ∘⇩s θ') |s A  = θ |s A"
    and B:"B ⊆ (A - subst_domain σ) ∪ range_vars (σ |s A)"
  shows "normal_subst R (θ' |s B)" using assms
proof -
  { fix x
    assume "x ∈ B"
    with B have "x ∈ (A - subst_domain σ) ∨ x ∈ range_vars (σ |s A)" by auto
    hence "(θ' x) ∈ NF (rstep R)" 
    proof 
      assume asm:"x ∈ (A - subst_domain σ)"
      hence "θ' x = (σ ∘⇩s θ') x" 
        by (metis DiffD2 eval_term.simps(1) notin_subst_domain_imp_Var subst_compose_def)
      also have "... = θ x" 
        by (metis Diff_iff asm assms(2) in_subst_restrict)
      then show "(θ' x) ∈ NF (rstep R)" using assms by (metis DiffD1 NF_I NF_Var asm calculation 
            in_subst_restrict local.wf normal_subst_def notin_subst_domain_imp_Var)
    next
      assume "x ∈ range_vars (σ |s A)"
      hence "∃y ∈ A. x ∈ vars_term (σ y)" unfolding range_vars_def subst_domain_def by auto
          (metis IntE in_subst_restrict restrict_subst subst_domain_restrict_subst_domain)
      then obtain y where y:"y ∈ A" and x:"x ∈ vars_term (σ y)" by auto
      hence *:"(σ ∘⇩s θ') y = θ y" by (metis assms(2) in_subst_restrict)
      from subst_app_subterm have "θ' x ⊴ (σ y) ⋅ θ'" using x by auto
      moreover have "θ y ∈ NF (rstep R)" 
        by (metis NF_I assms(1) in_subst_restrict local.wf no_Var_rstep normal_subst_def 
            notin_subst_domain_imp_Var y)
      ultimately show "(θ' x) ∈ NF (rstep R)"
        by (metis NF_rstep_subterm * subst_compose_def)
    qed
  } then show ?thesis unfolding subst_restrict_def normal_subst_def 
    by (meson NF_I local.wf no_Var_rstep)
qed

lemma vars_term_range: fixes δ::"('f, 'v)subst"
  assumes "x ∈ range_vars δ"
    and "x ∈ vars_term t"
    and "subst_domain δ ∩ range_vars δ = {}"
  shows "x ∈ vars_term (t ⋅ δ)" using assms 
proof(induct t)
  case (Var x)
  then show ?case unfolding range_vars_def subst_domain_def by auto
      (metis Var.prems(1) assms(3) disjoint_iff notin_subst_domain_imp_Var term.set_intros(3)) 
next
  case (Fun f ss)
  then show ?case unfolding range_vars_def subst_domain_def by auto
qed 

lemma subst_restricted_range_vars: fixes δ::"('f, 'v)subst" 
  assumes "x ∈ vars_term (t ⋅ δ)" 
    and disj:"subst_domain δ ∩ range_vars δ = {}"
    and "x ∉ vars_term t"
    and "x ∈ range_vars δ"
  shows "x ∈ range_vars (δ |s vars_term t)" using assms
proof(induct t)
  case (Var x)
  have "vars_term (Var x ⋅ δ) = range_vars (δ |s vars_term (Var x))" 
  proof(cases "x ∈ subst_domain δ")
    case True
    then show ?thesis unfolding range_vars_def subst_domain_def by auto 
        (metis IntI True in_subst_restrict insertI1 restrict_subst subst_domain_restrict_subst_domain,
          metis IntE in_subst_restrict restrict_subst singletonD subst_domain_restrict_subst_domain)
  next
    case False
    then show ?thesis unfolding range_vars_def subst_domain_def using False Var.prems(1) Var.prems(3) by simp
  qed 
  then show ?case using Var.prems(1) by force 
next
  case (Fun f ss)
  then show ?case unfolding range_vars_def subst_domain_def subst_restrict_def by auto 
      (smt (verit, del_insts) IntE IntI mem_Collect_eq subst_domain_def)
qed

(* Lemma 3.4 in MH94. The first version is the lifting lemma for ordinary terms. *)
lemma lifting_lemma:
  fixes V::"('v::infinite) set" and s::"('f, 'v)term" and t::"('f, 'v)term"
  assumes "normal_subst R θ"
    and "t = s ⋅ θ"
    and "vars_term s ∪ subst_domain θ ⊆ V"
    and rs:"(t,  t') ∈ (rstep R)⇧*"
    and fv:"finite V"
  shows "∃σ θ' s'. narrowing_derivation s s' σ ∧ t' = s' ⋅ θ' ∧
      normal_subst R θ' ∧ (σ ∘⇩s θ') |s V = θ |s V "
proof -
  from rs
  obtain f n where f0:"f 0 = t" and fn:"f n = t'" and rel_chain:"∀i < n. (f i,  f (Suc i)) ∈ rstep R" 
    by (metis rtrancl_imp_seq)
  then have "∃σ θ' s'. narrowing_derivation_num s s' σ n ∧ t' = s' ⋅ θ' ∧ 
    normal_subst R θ' ∧ (σ ∘⇩s θ') |s V = θ |s V" using assms
  proof(induct n arbitrary: s t t' θ f V rule: wf_induct[OF wf_measure [of "λ n. n"]])
    case (1 n)
    note IH1 = 1(1)[rule_format]
    then show ?case
    proof(cases "n = 0")
      case True
      show ?thesis
        by (rule exI[of _ Var], rule exI[of _ θ], rule exI[of _ "s"], insert 1 True)
          (simp add: narrowing_derivation_num_def subst_rule_def relpow_fun_conv, force) 
    next
      case False
      hence f0f1:"(f 0, f 1) ∈ rstep R" using 1 by auto
      then show ?thesis
      proof -
        from f0f1 obtain C σ l r  where f0:"f 0 = C⟨l ⋅ σ⟩" and f1:"f 1 = C⟨r ⋅ σ⟩"
          and rl:"(l, r) ∈ R" by auto
        have normθ:"normal_subst R θ" by fact
        have sθt:"s ⋅ θ = t" using f0 1 by auto
        obtain ω where varempty':"V ∩ vars_rule (ω ∙ (l, r)) = {}" using ‹finite V›
          by (metis rule_fs.rename_avoiding supp_vars_rule_eq vars_rule_def)
        have "is_Fun l" using wf[unfolded wf_trs_def] rl 
          by (simp add: is_Fun_Fun_conv wf_trs_imp_lhs_Fun)
        hence hpC:"hole_pos C ∈ fun_poss (f 0)" 
          by (metis f0 hole_pos_poss is_VarE is_VarI poss_is_Fun_fun_poss subst_apply_eq_Var subt_at_hole_pos)
        obtain ωr where ωr:"ωr ∙ (ω ∙ (l, r)) ∈ R" by (metis rl rule_pt.permute_minus_cancel(2))
        let ?p = "hole_pos C"
        have f0p:"(f 0) |_ ?p = l ⋅ σ" using f0 by auto
        have pf0:"?p ∈ fun_poss (f 0)" using hpC by auto
        hence pp0:"?p ∈ poss (f 0)" by (simp add: fun_poss_imp_poss) 
        have p:"?p ∈ poss s" 
        proof(rule ccontr)
          assume "¬ ?thesis"
          hence pns:"?p ∉ poss s" by simp
          hence pnfs:"?p ∉ fun_poss s" using fun_poss_imp_poss by blast
          have sub_eq:"(s ⋅ θ) |_ ?p = l ⋅ σ" using 1(2) f0p sθt by auto
          have psθ:"?p ∈ fun_poss (s ⋅ θ)" using 1(2) 1(7) pf0 using sθt by auto
          from poss_subst_apply_term[of ?p s θ]
          obtain q r x where qpr:"?p = q @ r" and qs:"q ∈ poss s" and sqx:"s |_ q = Var x" and r:"r ∈ poss (θ x)"
            using fun_poss_imp_poss pnfs psθ by blast
          hence *:"(s ⋅ θ) |_ ?p = (Var x) ⋅ θ |_ r" by force
          have "((Var x) ⋅ θ) ∈ NF (rstep R)" using normθ by (simp add: normal_subst_def)
              (metis NF_I NF_Var local.wf notin_subst_domain_imp_Var)
          hence "((Var x) ⋅ θ |_ r) ∈ NF (rstep R)" unfolding normal_subst_def using r
            by (metis NF_rstep_subterm eval_term.simps(1) subt_at_imp_supteq)
          then show False by (metis * NF_instance lhs_notin_NF_rstep rl sub_eq)
        qed
        hence pfun':"?p ∈ fun_poss (s ⋅ θ)" using p pp0 pf0 sθt f0p f0 ‹ f 0 = t› ‹is_Fun l› normθ[unfolded normal_subst_def]
          by auto 
        have sub:"is_Fun s" by (metis 1(2) NF_E eval_term.simps(1) f0f1 is_Fun_Fun_conv is_VarE local.wf normθ 
              normal_subst_def notin_subst_domain_imp_Var rstep_imp_Fun sθt)
        hence pfun:"?p ∈ fun_poss s" using pfun' p  p normθ[unfolded normal_subst_def]
          by (smt (verit, ccfv_SIG) "1.prems"(1) NF_instance eval_term.simps(1) f0p fun_poss_fun_conv 
              is_Fun_Fun_conv is_Var_def lhs_notin_NF_rstep notin_subst_domain_imp_Var poss_is_Fun_fun_poss rl sθt subt_at_subst)
        hence vuS':"vars_term (s |_ ?p) ⊆ vars_term s"
        proof -
          from vars_term_subt_at[OF p]
          have "vars_term (s |_ ?p) ⊆ vars_term s" by auto
          then show ?thesis by auto
        qed
        with varempty' have varcond:"vars_term (s |_ ?p) ∩ vars_rule (ω ∙ (l, r)) = {}" using 1(7) by auto
        have "∃σr. ∀x. σr (ω ∙ x) = σ x" using atom_pt.permute_minus_cancel(2) by (metis o_apply)
        then obtain σr where σrdef:"∀t. σr (ω ∙ t) = σ t" by auto
        hence "∀xs. subst_list σr (ω ∙ xs) = subst_list σ xs" unfolding subst_list_def
        proof(auto, goal_cases)
          case (1 xs s t)
          hence "(ω ∙ s) ⋅ σr = s ⋅ σ" 
            by (metis permute_term.simps(1) permute_term_subst_apply_term 
                subst_compose_def subst_monoid_mult.mult.left_neutral term_subst_eq_conv) 
          then show ?case 
            by (metis fst_conv rule_pt.fst_eqvt)
        next
          case (2 xs s t)
          hence "(ω ∙ t) ⋅ σr = t ⋅ σ" 
            by (metis permute_term.simps(1) permute_term_subst_apply_term 
                subst_compose_def subst_monoid_mult.mult.left_neutral term_subst_eq_conv) 
          then show ?case 
            by (metis snd_conv rule_pt.snd_eqvt)
        qed
        hence σr1:"subst_list σr [ω ∙ (l,r)]  = subst_list σ [(l,r)]" unfolding subst_list_def  
          using rule_pt.fst_eqvt rule_pt.snd_eqvt 
          by (smt (verit) list.simps(8) list.simps(9) rules_pt.permute_list_def)
        hence subeq:"subst_rule σr (ω ∙ (l,r))  = subst_rule σ (l, r)" unfolding subst_rule_def subst_list_def by auto
        let ?σdom = "vars_rule (ω ∙ (l, r))"
        let ?σr = "σr |s (?σdom)"
        let ?θ = " θ |s V"
        have sub_σr:"subst_domain ?σr ⊆ (vars_rule (ω ∙ (l, r)))"
          by (metis inf_le1 restrict_subst subst_domain_restrict_subst_domain)
        have sub_θ:"subst_domain ?θ ⊆ V" using 1(7) by auto
        have inter_empty:"subst_domain ?σr ∩ subst_domain ?θ = {}" using varempty' sub_σr sub_θ by auto
        from  sθt f0p have *:"(s ⋅ θ |_ ?p) = fst (ω ∙ (l, r)) ⋅ σr" using subeq
          by (simp add: 1(2) subst_rule_def)
        have varcond':"vars_term (s |_ ?p) ∩ vars_term (ω ∙ l) = {}" 
        proof -
          have "vars_term (ω ∙ l) ⊆ vars_rule (ω ∙ (l, r))" 
            by (metis fst_conv rule_pt.fst_eqvt sup_ge1 vars_rule_def)
          then show ?thesis using varcond by auto
        qed
        with normθ * have **:"(s |_ ?p) ⋅ θ = fst (ω ∙ (l, r)) ⋅ σr" by (simp add: p)
        have sub_eq:"(s |_ ?p) ⋅ ?θ = (ω ∙ l) ⋅ ?σr" using **
          by (metis 1(7) coincidence_lemma' fst_conv le_sup_iff rule_pt.permute_prod.simps subst_domain_neutral sup_ge1 vars_rule_def)
        from subst_union_sound[OF sub_eq]
        have subst_eq_θσ:"(s |_ ?p) ⋅ (?θ ∪⇩s ?σr)  = (ω ∙ l) ⋅ (?θ ∪⇩s ?σr)" using inter_empty sub_σr sub_θ inter_empty 
          by (smt (verit, best) disjoint_iff fst_conv local.wf rl sub_comm sub_eq subsetD subst_union.elims 
              term_subst_eq_conv varcond' varempty' vars_rule_eqvt vars_rule_lhs vars_term_eqvt)
        then obtain δ where mgu_uv:"mgu (s |_ ?p) (ω ∙ l) = Some δ" using mgu_ex 
          by (meson ex_mgu_if_subst_apply_term_eq_subst_apply_term)
        from mgu_sound[OF mgu_uv] have δ:"(?θ ∪⇩s ?σr) =  δ ∘⇩s (?θ ∪⇩s ?σr)" using subst_eq_θσ 
          by (smt (verit, ccfv_SIG) is_imgu_def subst_monoid_mult.mult_assoc the_mgu the_mgu_is_imgu)
        have subst_range_disjδ:"subst_domain δ ∩ range_vars δ = {}" using mgu_uv mgu_subst_domain_range_vars_disjoint by blast
        have crl:"(ω ∙ r) ⋅ σr = (r ⋅ σ)" by (metis rule_pt.snd_eqvt snd_conv subeq subst_rule_def)
        from subst_domain_restrict_subst_domain[of ?σdom σr]
        have subinter:"subst_domain ?σr = subst_domain σr ∩ ?σdom" by auto
        have sndeq:"(ω ∙ r) ⋅ (?θ ∪⇩s ?σr) = r ⋅ σ" by (smt (verit, ccfv_threshold) Un_Int_eq(4) coincidence_lemma' crl 
              inf.absorb_iff2 inf_bot_right inf_commute inf_left_commute restrict_subst rule_pt.snd_eqvt snd_conv 
              subst_domain_restrict_subst_domain subst_union_term_reduction varempty' vars_rule_def)
        let ?s1 = "(replace_at s ?p (ω ∙ r))⋅ δ"
        have nar:"(s, ?s1, δ) ∈ narrowing_step" unfolding subst_rule_def narrowing_step_def using  ωr mgu_uv rl varempty' pfun by auto 
            (smt (verit, ccfv_threshold) 1(7) Int_assoc fst_conv inf.orderE inf_bot_right le_sup_iff narrowing_step.simps 
              narrowing_stepp_narrowing_step_eq rule_pt.fst_eqvt rule_pt.snd_eqvt snd_conv subst_apply_term_ctxt_apply_distrib)
        from narrowing_set_imp_rtran[OF nar]
        have condn1:"narrowing_derivation_num s ?s1 δ 1" by auto
        let ?V = "(V - subst_domain δ) ∪ range_vars δ"
        let ?θ1 = "(?θ ∪⇩s ?σr) |s ?V"
        have subθ1:"subst_domain ?θ1 ⊆ ?V" unfolding subst_restrict_def 
          by (smt (verit, del_insts) mem_Collect_eq subset_eq subst_domain_def)
        have reseq:"?θ1 |s ?V = (?θ ∪⇩s ?σr)|s ?V" 
          by (simp add: restrict_subst_domain_def)
        have reseqδ:"(δ ∘⇩s ?θ1) |s V = (δ ∘⇩s (?θ ∪⇩s ?σr))|s V" 
          using r_subst_compose reseq by blast
        have reseqθ:"(δ ∘⇩s ?θ1) |s V = θ |s V " using δ by auto (smt (verit, best) δ disjoint_iff notin_subst_domain_imp_Var 
              reseqδ restrict_subst restrict_subst_domain_def subst_domain_neutral subst_ext subst_union.elims varempty')
        have "normal_subst R (?θ1 |s ?V)"
        proof -
          let ?δ = "δ |s V"
          let ?B = "(V - subst_domain δ) ∪ range_vars ?δ"
          have normV:"normal_subst R (θ |s V)" using normθ 1(7) by auto
          have BV:"?B ⊆ V - subst_domain δ ∪ range_vars (δ |s V)" by auto
          from restricted_normalized[OF normV reseqθ BV]
          have normB:"normal_subst R (?θ1 |s ?B)" by auto 
          have ranB:"range_vars δ ⊆ ?B"
          proof
            fix x
            assume asm:"x ∈ range_vars δ"
            hence xnδ:"x ∉ subst_domain δ" using δ
              by (meson disjoint_iff mgu_subst_domain_range_vars_disjoint mgu_uv)
            have subst_x:"x ∈ vars_term (s |_ ?p) ∪ vars_term (ω ∙ l)" using mgu_uv 
                asm mgu_range_vars by auto
            then show "x ∈ ?B"
            proof
              assume "x ∈ vars_term (s |_ ?p)"
              hence "x ∈ V - subst_domain δ" using 1(7) xnδ vuS' by auto
              then show ?thesis by auto
            next
              assume asm2:"x ∈ vars_term (ω ∙ l)"
              hence xnu:"x ∉ vars_term (s |_ ?p)" using varcond' by blast
              from vars_term_range[OF asm asm2 subst_range_disjδ]
              have xvδ:"x ∈ vars_term ((ω ∙ l) ⋅ δ)" by auto
              have *:"x ∈ vars_term (s |_ ?p ⋅ δ)" using mgu_uv xvδ
                by (simp add: subst_apply_term_eq_subst_apply_term_if_mgu) 
              hence "vars_term ((ω ∙ l) ⋅ δ) = vars_term (s |_ ?p ⋅ δ)" using mgu_uv 
                by (simp add: subst_apply_term_eq_subst_apply_term_if_mgu)
              from subst_restricted_range_vars[OF * subst_range_disjδ xnu asm]
              have "x ∈ range_vars (δ |s vars_term (s |_ ?p))" by auto
              moreover have upV:"vars_term (s |_ ?p) ⊆ V" using 1(7) vuS' by auto
              ultimately have "x ∈ range_vars (δ |s V)" unfolding range_vars_def subst_restrict_def
                by (auto simp add: subsetD subst_domain_def) 
              then show ?thesis by auto
            qed
          qed
          hence "?B = ?V"
          proof -
            have "range_vars ?δ ⊆ range_vars δ"  unfolding subst_restrict_def range_vars_def  
              by (auto, smt (verit) mem_Collect_eq subst_domain_def, simp add: subst_domain_def)
            then show ?thesis using ranB by blast
          qed
          then show ?thesis using normB by auto
        qed
        hence normθ1:"normal_subst R ?θ1" using subθ1 reseq by fastforce
        have vars_s1:"vars_term ?s1 ⊆ ?V" 
        proof -
          have *:"vars_term (ω ∙ r) ⊆ vars_term (ω ∙ l)" using wf[unfolded wf_trs_def] 
            by (metis fst_eqD local.wf rl rule_pt.fst_eqvt rule_pt.snd_eqvt snd_eqD sup.orderI vars_rule_def 
                vars_rule_eqvt vars_rule_lhs vars_term_eqvt)
          from var_cond_stable[OF this]
          have "vars_term ((ω ∙ r) ⋅ δ ) ⊆ vars_term ((ω ∙ l) ⋅ δ)" by fastforce
          from replace_var_stable[OF this]
          have "vars_term ?s1 ⊆ vars_term (replace_at s ?p ((ω ∙ l))⋅ δ)" 
            by (meson * replace_var_stable var_cond_stable)
          moreover have "vars_term ((replace_at s ?p (ω ∙ l))⋅ δ) = vars_term (s ⋅ δ)"
            by (metis ctxt_supt_id mgu_uv p subst_apply_term_ctxt_apply_distrib subst_apply_term_eq_subst_apply_term_if_mgu)
          moreover have "vars_term (s ⋅ δ) ⊆ ?V" 
            by (smt (verit) 1(7) Un_Diff Un_iff subsetI subset_Un_eq vars_term_subst_apply_term_subset)
          ultimately show ?thesis by auto
        qed
        have rel_chain':"⋀i. i < n - 1 ⟹ (f (i + 1), f (Suc i + 1)) ∈ rstep R" using rel_chain 
          by (simp add: 1(4))
        let ?f = "λi. f (i + 1)"
        have relstar:"(f 1, f n) ∈ (rstep R)⇧*" using False 1(4) less_Suc_eq 
          by (induct n, blast, metis (no_types, lifting) One_nat_def rtrancl.simps)
        have "∃δ' θ' s'. narrowing_derivation_num ?s1 s' δ' (n - 1) ∧  f n = s' ⋅ θ' ∧ normal_subst R θ' ∧
         (δ' ∘⇩s θ') |s ?V = ?θ1 |s ?V"
        proof (rule IH1[of "n - 1"  ?f "f 1" "f n" ?θ1 ?s1], goal_cases)
          case 1
          then show ?case using False by auto
        next
          case 2
          then show ?case by simp
        next
          case 3
          then show ?case using False by auto
        next
          case (4 i)
          then show ?case using rel_chain' by auto
        next
          case 5
          then show ?case using normθ1 by auto
        next
          case 6
          have inempty:"subst_domain ?σr ∩ V = {}" using varempty' subinter by auto
          hence θsv:"(?θ ∪⇩s ?σr) |s V = θ |s V"
            using δ reseqδ reseqθ by auto
          have "(?θ ∪⇩s ?σr) |s (vars_term (ω ∙ r)) = ?σr |s (vars_term (ω ∙ r))" 
          proof -
            { fix v :: 'v
              have "vars_rule (ω ∙ (l, r)) - V = vars_rule (ω ∙ (l, r))" using varempty' by fastforce
              then have "V ∩ vars_term (ω ∙ r) = {}"  by (metis (no_types, lifting) disjoint_iff 
                    rule_pt.snd_eqvt snd_conv subsetD sup_ge2 varempty' vars_rule_def)
              then have "(?θ ∪⇩s ?σr) |s (vars_term (ω ∙ r)) = ?σr |s (vars_term (ω ∙ r))"
                by (smt (verit, best) Int_iff disjoint_iff inf.absorb_iff2 sub_θ subst_ext subst_union.simps)
            } then show ?thesis by fastforce
          qed
          have "f 1 = ?s1 ⋅ ?θ1"
          proof -
            have "f 1 = replace_at (s ⋅ θ) ?p ((ω ∙ r) ⋅ σr)" using crl f1 sθt 
              by (metis 1(2) ctxt_of_pos_term_hole_pos f0) 
            also have "...  = replace_at (s ⋅ (?θ ∪⇩s ?σr)) ?p ((ω ∙ r) ⋅ (?θ ∪⇩s ?σr))" 
              by (metis 1(7) θsv coincidence_lemma' crl sndeq sup.boundedE)
            also have "... = replace_at s ?p (ω ∙ r) ⋅ (?θ ∪⇩s ?σr)" by (simp add: ctxt_of_pos_term_subst p)
            also have "... = replace_at s ?p (ω ∙ r) ⋅ (δ ∘⇩s (?θ ∪⇩s ?σr))" using δ by auto
            also have "... = ?s1 ⋅ (?θ ∪⇩s ?σr)" by simp
            also have *:"f 1 = ?s1 ⋅ ?θ1" using vars_s1 using calculation subst_apply_term_restrict_subst_domain by fastforce
            finally show ?thesis using * by fastforce
