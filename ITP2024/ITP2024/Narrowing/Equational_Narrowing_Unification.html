section ‹Formalization of equational narrowing based E-unification›

theory Equational_Narrowing_Unification
  imports
    Equational_Narrowing_Reachability
begin

locale equational_narrowing_unification = equational_narrowing_reachability R DOTEQ TOP R' F D x
  for R::"('f, 'v:: infinite) trs" 
    and DOTEQ :: 'f ("≐")
    and TOP :: "'f" ("⊤") 
    and R' :: "('f, 'v:: infinite) trs" 
    and F :: "'f sig" 
    and D :: "'f sig"
    and x :: 'v
begin

lemma R_unifiable: assumes funas_uv:"funas_rule (u, v) ⊆ F"
  and fun_uv:"(Fun (≐) [u, v], Fun (⊤) []) ∈ (rstep R)⇧+" (* It needs at least one rewrite step *)
shows "(u, v) ∈ (rstep R')⇧↔⇧*" 
proof -
  let ?srule = "λx. (Fun (≐) [Var x, Var x], Fun (⊤) [])"
  have fr:"∀x. funas_rule (?srule x) = {(≐, 2), (⊤, 0)}" using D 
    unfolding funas_defs by (auto simp add: numeral_2_eq_2)
  have frD:"∀x. funas_rule (?srule x) ⊆ D" using D 
    unfolding funas_defs by (auto simp add: numeral_2_eq_2)
  from fun_uv obtain n where n1:"n ≥ 1" and fuv:"(Fun (≐) [u, v], Fun (⊤) []) ∈ (rstep R)^^n"
    by (metis order_refl relpow_1 trancl_steps_relpow)
  then show ?thesis using funas_uv
  proof(induct n arbitrary:u v)
    case 0
    then show ?case by auto
  next
    case (Suc n)
    from Suc(4) have funas_u:"funas_term u ⊆ F" unfolding funas_defs by auto
    from Suc(4) have funas_v:"funas_term v ⊆ F" unfolding funas_defs by auto
    from Suc(3) obtain U where U1:"(Fun ≐ [u, v], U) ∈ rstep R" and U2:"(U, Fun (⊤) []) ∈ (rstep R)^^ n"
      using Suc(2) by (meson relpow_Suc_E2)
    then show ?case
    proof(cases "Suc n = Suc 0")
      case True
      with Suc(3) have "(Fun ≐ [u, v], Fun ⊤ []) ∈ rstep R" by simp
      then obtain C σ l r where lr:"(l, r) ∈ R" and lhs:"Fun ≐ [u, v] = C⟨l ⋅ σ⟩" and rhs:"Fun ⊤ [] = C⟨r ⋅ σ⟩" by auto
      have rule_type:"(l, r) ∈ R' ∨ (∃x. (l, r) = ?srule x)" using R' lr by fastforce
      hence frule_type:"funas_rule (l, r) ⊆ F ∨ (∃x. (l, r) = ?srule x)" using R_sig 
        by (metis fst_conv funas_rule_def le_supI lhs_wf rhs_wf snd_conv)
      have fnlr:"¬ funas_rule (l, r) ⊆ F"
      proof(rule ccontr)
        assume "¬ ?thesis"
        hence flr:"funas_rule (l, r) ⊆ F" using D D_fresh by auto
        from rhs have "funas_term  (C⟨r ⋅ σ⟩) ⊆ D" using D by auto
        hence *:"funas_term (r ⋅ σ) ⊆ D" by auto
        from flr have "funas_term r ⊆ F" unfolding funas_defs by auto
        hence **:"funas_term (r ⋅ σ) ⊆ F" using wf_F_subst by simp
        have "C = Hole" using rhs by (metis nectxt_imp_supt_ctxt supt_const)
        hence "funas_term (r ⋅ σ) ≠ {}" using rhs by auto 
        then show False using * ** D_fresh by auto
      qed
      hence **:"(∃x. (l, r) = ?srule x)" using fr frule_type by auto
      hence "u = v" using lr lhs rhs by auto
      then show ?thesis by auto
    next
      case False
      hence UN:"U ≠ Fun (⊤) []" using not_reducible_T U2 by (metis relpow_E2)
      have *:"(Fun ≐ [u, v], U) ∈ rstep R'"
      proof(rule ccontr)
        assume asm:"¬ ?thesis"
        hence "(Fun ≐ [u, v], U) ∉ rstep R'" by auto
        hence "∃x. (Fun ≐ [u, v], U) ∈ rstep {?srule x}" using R' by (metis U1 Un_iff rstep_union)
        then obtain x where "(Fun ≐ [u, v], U) ∈ rstep {?srule x}" by auto
        then obtain C' σ' l' r' where l'r':"(l', r') ∈ {?srule x}" and fuv:"Fun ≐ [u, v] = C'⟨l' ⋅ σ'⟩" 
          and UC':"U = C'⟨r' ⋅ σ'⟩" by auto
        have "r' ⋅ σ' =  Fun (⊤) []" using l'r' by auto
        with UN UC' have NC':"C' ≠ Hole" by auto
        hence "l' ⋅ σ' ⊲ Fun ≐ [u, v]" using fuv by auto
        hence *:"l' ⋅ σ' ⊴ u ∨ l' ⋅ σ' ⊴ v" by auto
        have "(≐, 2) ∈ funas_term (l' ⋅ σ')" using l'r' by auto
        then show False using funas_u funas_v D D_fresh *
          by (meson insert_disjoint(1) subset_iff supteq_imp_funas_term_subset) 
      qed         
      from funas_R_reachable_case[OF funas_u funas_v this]
      obtain u' v' where U:"U = (Fun ≐ [u', v'])" and uu'vv':"(u = u' ∨ v = v')" and fu':"funas_term u' ⊆ F" 
        and fv':"funas_term v' ⊆ F" using funas_u funas_v by blast
      have IH:"(u', v') ∈ (rstep R')⇧↔⇧*" using Suc n1 U2 fu' fv'
        by (metis (no_types, lifting) False One_nat_def U fst_conv funas_rule_def le_Suc_eq le_sup_iff snd_conv)
      show ?thesis using uu'vv'
      proof
        assume asm:"u = u'"
        hence vv':"(v, v') ∈ (rstep R')⇧↔⇧*" using U U1 asm fun_rstep_case1 * funas_u funas_v by blast
        then show ?thesis using IH asm by (meson conversion_inv conversion_trans transD)
      next
        assume asm:"v = v'"
        hence uu':"(u, u') ∈ (rstep R')⇧↔⇧*"  using fun_rstep_case2 U U1 * funas_u funas_v by blast
        then show ?thesis using IH asm by (meson conversion_trans transD)
      qed
    qed
  qed
qed

lemma narrowing_based_E_unif_infeasibility: 
  assumes ensrp:"E_unif_normalized_subst_reachability_property s t"
    and funas_C:"funas_rule (s, t) ⊆ F"
    and cr:"CR (rstep R)"
  shows "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ ¬ E_unifiable (s, t)" 
proof -
  assume "¬ narrowing_derivation_reaches_to_success (s, t)"
  hence ncs:"¬ (∃σ. narrowing_derivation (Fun (≐) [s, t]) ( Fun (⊤) []) σ) " unfolding narrowing_derivation_reaches_to_success_def by auto
  from convert_equation_into_term_sound[OF funas_C]
  have wfC:"wf_equational_term (Fun (≐) [s, t])" by auto
  from infeasibility_using_narrowing[OF wfC ncs]
  have *:"¬ (∃θ. normal_subst R θ ∧ (Fun (≐) [s, t] ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*)" by auto
  show ?thesis
  proof
    assume "E_unifiable (s, t)"
    hence "(¬ funas_rule (s, t) ⊆ F ∨ (∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*))"
      unfolding E_unifiable_def by auto
    with funas_C have reach:"∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*" by auto
    hence "∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R')⇧↔⇧*" using funas_C funas_rstep_R'_conv unfolding funas_defs 
      by (auto, meson wf_F_subst)
    then obtain τ where "(s ⋅ τ, t ⋅ τ) ∈ (rstep R')⇧↔⇧*" by auto
    with ensrp[unfolded E_unif_normalized_subst_reachability_property_def] obtain τ' where 
      τ':"(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*" and normτ':"normal_subst R τ'" and "∀x. (τ x, τ' x) ∈ (rstep R)⇧*" 
      by (metis (no_types, opaque_lifting) R' Un_upper1 conversion_mono rstep_union subsetD)
    from rstep_reduction_R_unif_normalized_equiv
    have "∃θ. normal_subst R θ ∧ (Fun ≐ [s ⋅ θ, t ⋅ θ], Fun (⊤) []) ∈ (rstep R)⇧*" using τ' normτ' funas_C cr by blast
    then show False using * by auto
  qed
qed

(* Sufficient condition for E_unif_normalized_subst_reachability_property. Only WN is needed. *)

lemma E_unif_normalized_subst_reachability_cond: assumes wn:"WN (rstep R)"
  shows "E_unif_normalized_subst_reachability_property s t"
proof -
  { fix τ
    assume asm:"(s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*"
    from WN_obtains_normalizable_subst[OF wn]
    have normτ:"normalizable_subst R τ" by auto
    from obtains_normalized_subst[OF normτ]
    obtain τ' where normτ':"normal_subst R τ'" and ττ':"∀x. (τ x, τ' x) ∈ (rstep R)⇧*" by auto
    have rstepτ':"(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*"
    proof -
      have "(s ⋅ τ, s ⋅ τ') ∈ (rstep R)⇧↔⇧*" using ττ' 
        by (simp add: conversionI' substs_rsteps)
      moreover have "(t ⋅ τ, t ⋅ τ') ∈ (rstep R)⇧*" using ττ'  
        by (simp add: substs_rsteps)
      moreover have "(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*" 
        by (meson asm calculation(1) calculation(2) conversionI' conversion_inv conversion_trans transD)
      ultimately show ?thesis by auto
    qed
    hence "∃τ'. (normal_subst R τ' ∧ (s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧* ∧ (∀x. (τ x, τ' x) ∈ (rstep R)⇧*))" 
      using normτ' ττ' by auto
  } then show ?thesis unfolding E_unif_normalized_subst_reachability_property_def by auto
qed

(* If a narrowing derivation reaches the 'success' state,then the solution of an E-unifiability problem exists. *)

lemma narrowing_based_E_unifiable:
  assumes funas_uv:"funas_rule (u, v) ⊆ F"
    and nd:"∃σ. narrowing_derivation (Fun ≐ [u, v]) (Fun ⊤ []) σ"
  shows "∃θ. (u ⋅ θ, v ⋅ θ) ∈ (rstep R)⇧↔⇧*"
proof -
  let ?S = "Fun (≐) [u, v]"
  from funas_uv have "(≐, 2) ∉ funas_term u ∧ (≐, 2) ∉ funas_term v" using D D_fresh R_sig
    unfolding funas_defs by auto
  moreover from funas_uv  have "(⊤, 0) ∉ funas_term u ∧ (⊤, 0) ∉ funas_term v" using D D_fresh R_sig
    unfolding funas_defs by auto
  ultimately have wf_eq:"wf_equational_term ?S" unfolding wf_equational_term_def by auto
  from nd obtain σ where nar:"narrowing_derivation ?S (Fun (⊤) []) σ" by auto
  from equational_narrowing_based_reachability'[of ?S]
  have "∃θ. (?S ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*"
    by (simp add: nd wf_eq)
  then obtain θ where *:"(?S ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*" by auto
  have "?S ⋅ θ = Fun (≐) [u ⋅ θ, v ⋅ θ]" by auto
  have "funas_term (u ⋅ θ) ⊆ F" using wf_F_subst funas_uv unfolding funas_defs by simp
  moreover have "funas_term (v ⋅ θ) ⊆ F" using wf_F_subst funas_uv unfolding funas_defs by simp
  ultimately have fuvθ:"funas_rule (u ⋅ θ, v ⋅ θ) ⊆ F" unfolding funas_defs by simp
  from R_unifiable[OF fuvθ]
  have "(u ⋅ θ, v ⋅ θ) ∈ (rstep R')⇧↔⇧*" using * by (simp add: rtrancl_eq_or_trancl)
  hence "(u ⋅ θ, v ⋅ θ) ∈ (rstep R)⇧↔⇧*" using R'
    by (metis (no_types, opaque_lifting) Un_upper1 conversion_mono in_mono rstep_mono)
  then show ?thesis by (rule exI[of _ θ])
qed

(* Hullot's completeness of E-unification via narrowing, where R is viewed as a set of equations E. 
  If σ is an R-unifier of s and t, then a narrowing derivation provides a more general unifier than σ if it reaches to ⊤. *)

theorem narrowing_based_completeness_of_E_unification:
  assumes semi_comp:"semi_complete (rstep R)"
    and funas_st:"funas_rule (s, t) ⊆ F"
    and st_unif:"(s ⋅ σ, t ⋅ σ) ∈ (rstep R)⇧↔⇧*"
  shows nd:"∃τ θ. narrowing_derivation (Fun ≐ [s, t]) (Fun ⊤ []) τ ∧ 
            (τ ∘⇩s θ) |s (vars_term s ∪ vars_term t) =⇩R σ |s (vars_term s ∪ vars_term t)"
proof -
  from semi_comp have wn:"WN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: semi_complete_on_def, insert semi_comp, auto)
  from wn have normσ:"normalizable_subst R σ"
    using WN_obtains_normalizable_subst by auto
  from obtains_normalized_subst[OF normσ]
  obtain σ' where normσ':"normal_subst R σ'" and σσ':"(∀x. (σ x, σ' x) ∈ (rstep R)⇧*)" by auto
  with st_unif have *:"(s ⋅ σ', t ⋅ σ') ∈ (rstep R)⇧↔⇧*"
    by (metis (no_types, opaque_lifting) all_ctxt_closed_rsteps all_ctxt_closed_subst_step converse_iff 
        conversionI' conversion_converse conversion_rtrancl r_r_into_trancl rtrancl_trancl_absorb)
  hence "(s ⋅ σ', t ⋅ σ') ∈ (rstep R)⇧↓" using cr wn by (simp add: CR_imp_conversionIff_join)
  hence **:"(Fun ≐ [s ⋅ σ', t ⋅ σ'], Fun ⊤ []) ∈ (rstep R)⇧*"
    by (metis * cr fst_eqD funas_rule_def funas_st le_sup_iff rstep_reduction_step_R_unif_eq snd_eqD wf_F_subst)
  let ?S = "Fun ≐ [s, t]"
  have wf:"wf_equational_term ?S" using funas_st convert_equation_into_term_sound by force
  let ?V = "vars_term ?S ∪ subst_domain σ'"
  have varcond':"vars_term ?S ∪ subst_domain σ' ⊆ ?V" ..
  have fn:"finite ?V" by (simp add: finite_subst_domain)
  have ***:"(?S ⋅ σ',  Fun ⊤ []) ∈ (rstep R)⇧*" using ** by auto
  have wf_eq:"wf_equational_term (Fun ⊤ [])" unfolding wf_equational_term_def by auto
  have vacuous_subst:"∀θ. Fun ⊤ [] = (Fun ⊤ []) ⋅ θ" by auto
  from lifting_lemma_equational_terms[of σ' ?S "?S ⋅ σ'" "?V" "Fun ⊤ []"]
  have "∃τ θ S'. narrowing_derivation (Fun ≐ [s, t]) S' τ ∧ Fun ⊤ [] = S' ⋅ θ ∧ wf_equational_term S' ∧
       normal_subst R θ ∧ τ ∘⇩s θ |s (vars_term (Fun ≐ [s, t]) ∪ subst_domain σ') = σ' |s (vars_term (Fun ≐ [s, t]) ∪ subst_domain σ')" 
    using "***" fn local.wf normσ' by fastforce
  then obtain S' where ob:"∃τ θ. narrowing_derivation (Fun ≐ [s, t]) S' τ ∧ Fun ⊤ [] = S' ⋅ θ ∧ wf_equational_term S' ∧
       normal_subst R θ ∧ τ ∘⇩s θ |s (vars_term (Fun ≐ [s, t]) ∪ subst_domain σ') = σ' |s (vars_term (Fun ≐ [s, t]) ∪ subst_domain σ')"
    by auto
  hence"S' = Fun ⊤ []" using vacuous_subst using wf_equational_term_def by fastforce
  hence "∃τ θ. narrowing_derivation ?S (Fun ⊤ []) τ ∧ (τ ∘⇩s θ) |s ?V = σ' |s ?V" using ob wf wf_eq *** fn varcond' by auto
  then obtain τ θ where nd:"narrowing_derivation ?S (Fun ⊤ []) τ" and subst_cond:"(τ ∘⇩s θ) |s ?V = σ' |s ?V" by auto
  let ?Vst = "(vars_term s ∪ vars_term t)"
  from subst_cond have subVst:"(τ ∘⇩s θ) |s ?Vst = σ' |s ?Vst"
    by (auto, metis Un_Int_eq(1) subst_restrict_Int)
  have "∀x. (σ x, σ' x) ∈ (rstep R)⇧↔⇧*" using σσ' by auto
  hence var_rest_cond:"(τ ∘⇩s θ) |s ?Vst  =⇩R σ |s ?Vst" using subVst
    by (smt (verit, ccfv_threshold) conversion_def conversion_inv in_subst_restrict 
        notin_subst_restrict rtrancl.simps subst_modulo_def)
  show ?thesis by (rule exI[of _ τ], rule exI[of _ θ], insert nd var_rest_cond, auto) 
qed

lemma narrowing_based_non_E_unifiability: 
  assumes semi_comp:"semi_complete (rstep R)"
    and funas_C:"funas_rule (s, t) ⊆ F"
  shows "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ ¬ E_unifiable (s, t)"
proof -
  from semi_comp have wn:"WN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: semi_complete_on_def, insert semi_comp, auto)
  then show "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ ¬ E_unifiable (s, t)"
    by (simp add: cr funas_C narrowing_based_E_unif_infeasibility E_unif_normalized_subst_reachability_cond wn)
qed

(* Narrowing based E-unifiability, providing a decision procedure of E-unifiability. 
  Here, (rstep R)⇧↔ is viewed as (rstep E)⇧↔ *)

theorem narrowing_based_E_unifiability: 
  assumes semi_comp:"semi_complete (rstep R)"
    and funas_C:"funas_rule (s, t) ⊆ F"
  shows "narrowing_derivation_reaches_to_success (s, t) ⟹ E_unifiable (s, t)"
        "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ ¬ E_unifiable (s, t)"
proof -
  from semi_comp have wn:"WN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: semi_complete_on_def, insert semi_comp, auto)
  show "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ ¬ E_unifiable (s, t)" 
    by (simp add: cr funas_C narrowing_based_E_unif_infeasibility E_unif_normalized_subst_reachability_cond wn)
  show "narrowing_derivation_reaches_to_success (s, t) ⟹ E_unifiable (s, t)"
    by (simp add: E_unifiable_def funas_C narrowing_based_E_unifiable narrowing_derivation_reaches_to_success_def)
qed  

end
