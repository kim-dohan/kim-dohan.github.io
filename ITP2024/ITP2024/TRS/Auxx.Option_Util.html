theory Option_Util
imports Main
begin

primrec option_to_list :: "'a option ⇒ 'a list"
where
  "option_to_list (Some a) = [a]" |
  "option_to_list None = []"

lemma set_option_to_list_sound [simp]:
  "set (option_to_list t) = set_option t"
  by (induct t) auto

end
