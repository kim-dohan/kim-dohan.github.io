theory Context_Sensitive
  imports
    Unification_More
    Trs_Impl
    "HOL-Library.Monad_Syntax"
    Auxx.Name
    Forbidden_Patterns
begin

(* context sensitive replacement map μ *)
type_synonym 'f mu = "'f × nat ⇒ nat set"

definition csstep_cond :: "'f mu ⇒ pos ⇒ ('f,'v)term ⇒ bool"
  where "csstep_cond c p t ≡ 
 ∀ p1 i p2. p = p1 @ i # p2 ∧ p ∈ poss t ⟶ (∃ f ts. t |_ p1 = Fun f ts ∧ i ∈ c (f,length ts))" 

definition csstep_r_p_s :: 
  "'f mu ⇒ ('f,'v)trs ⇒ ('f, 'v) rule ⇒ pos ⇒ ('f, 'v) subst ⇒ 
  ('f,'v)term ⇒ ('f,'v)term ⇒ bool"
  where "csstep_r_p_s μ rs lr p σ s t ≡ (s, t) ∈ rstep_r_p_s rs lr p σ ∧ csstep_cond μ p s"

definition csstep_p :: "'f mu ⇒ ('f,'v)trs ⇒ pos ⇒ ('f,'v)trs"
  where "csstep_p μ rs p = {(s,t) | s t.(∃ lr σ. (s,t) ∈ rstep_r_p_s rs lr p σ ∧ csstep_cond μ p s)}"

definition csstep :: "'f mu ⇒ ('f,'v)trs  ⇒ ('f,'v)trs"
  where "csstep μ rs = {(s,t). ∃ p. (s,t) ∈  csstep_p μ rs p}"

(* Transformation of context sensitive replacement map μ to forbidden patterns *)
definition to_pi :: "'f mu ⇒ ('f, string) forb_patterns"
  where "to_pi c ≡ {(ct, xi, loc). ∃ f a xs i. i < a ∧ i ∉ c (f,a) ∧
 xs = map (λ x . Var x) (fresh_strings_list ''x'' 0 [] a) ∧ xi = xs!i ∧
 ct = ctxt_of_pos_term [i] (Fun f xs) ∧ loc ∈ {Forbidden_Patterns.B, Forbidden_Patterns.H}}"


lemma fp_is_cs_cond: assumes posst:"p ∈ poss t" shows "fpstep_cond (to_pi c) p t = csstep_cond c p t"
proof
  assume fp:"fpstep_cond (to_pi c) p t" show "csstep_cond c p t"  unfolding csstep_cond_def
  proof((rule allI)+)
    fix p1 i p2
    show "p = p1 @ i # p2 ∧ p ∈ poss t ⟶ (∃f ts. t |_ p1 = Fun f ts ∧ i ∈ c (f, length ts))"
    proof
      assume "p = p1 @ i # p2 ∧ p ∈ poss t"
      then have p':"p = p1 @ i # p2" and pt:"p ∈ poss t" by auto
      let ?t' = "t |_ p1"
      from p' pt have "is_Fun ?t'" by (metis is_VarE list.distinct(1) poss_append_poss var_pos_maximal)
      then obtain f ts where t:"?t' = Fun f ts" by blast
      let ?a = "length ts"
      let ?H = Forbidden_Patterns.H 
      let ?B = Forbidden_Patterns.B
      have "funas_term ?t' ⊆ funas_term t" by (metis p' poss_append_poss pt subt_at_imp_supteq supteq_imp_funas_term_subset)
      from pt have "i # p2 ∈ poss ?t'" using poss_append_poss pt[unfolded p'] by blast 
      from this[unfolded t] have ia:"i < ?a" by (metis term.sel(4) poss_Cons_poss)
      have "i ∈ c (f,?a)"
      proof(rule ccontr)
        assume nic:"i ∉ c (f, length ts)"
        let ?fresh = "fresh_strings_list ''x'' 0 [] ?a"
        from nic ia have "∃ xs l ci xi. ∀ l ∈ {?B, ?H}.
      (ci, xi, l) ∈ to_pi c ∧ ci = ctxt_of_pos_term [i] (Fun f xs) ∧
      xs = map (λ x . Var x) ?fresh ∧ xi = xs!i" unfolding to_pi_def by blast
        then obtain xs ci xi where pH:"(ci, xi, ?H) ∈ to_pi c" and pB:"(ci, xi, ?B) ∈ to_pi c" and
          ci:"ctxt_of_pos_term [i] (Fun f xs) = ci" and xs:"xs = map (λ x . Var x) ?fresh" and xi:"xi = xs!i" by auto
        from xs fresh_name_gen_for_strings_list[unfolded fresh_name_gen_list_def] have 
          "length xs = ?a ∧ distinct ?fresh" by (metis length_map)
        then have len:"length xs = ?a" and  disj:"set ?fresh ∩ set [] = {}" and dist:"distinct ?fresh" by auto 
        with ia have ip:"[i] ∈ poss (Fun f xs)" by force
        with ctxt_supt_id[OF ip] xi have a:"(ctxt_of_pos_term [i] (Fun f xs))⟨xi⟩ = Fun f xs" by force
        from a have b:"(ctxt_of_pos_term [i] (Fun f xs))⟨xi⟩ = Fun f xs" by fast
        define σ where "σ = mk_subst Var (zip ?fresh ts)" 
        from len[unfolded xs] have len':"length ts = length ?fresh" by auto
        with mk_subst_distinct[OF dist] have c:"map σ ?fresh = ts" unfolding σ_def by (metis map_nth_eq_conv)
        have d:"map (λ t. t ⋅ σ) xs = map ((λ t. t ⋅ σ) ∘ Var) ?fresh" unfolding xs by (metis List.map.compositionality)
        have d:"map σ ?fresh =  map ((λt. t ⋅ σ) ∘ Var) ?fresh" by (metis comp_apply eval_term.simps(1))
        from c have "map (λ t. t ⋅ σ) (map Var ?fresh) = ts" unfolding map_map by (metis d)
        from this mk_subst_distinct[OF dist] have "(Fun f xs) ⋅ σ = ?t'" unfolding xs t eval_term.simps(2) by force
        with b ci have "∃ σ. ?t' = ci⟨xi⟩ ⋅ σ" by fastforce
        with p' pt have sg:"∃ σ. t = (ctxt_of_pos_term p1 t)⟨ci⟨xi⟩ ⋅ σ⟩" by (metis ctxt_supt_id poss_append_poss)
        from pt[unfolded p'] have pt':"p1 ∈ poss t" by simp
        have "¬ fpstep_cond (to_pi c) p t" 
        proof(cases p2)
          case Nil
          from hole_pos_ctxt_of_pos_term[OF ip]  hole_pos_ctxt_of_pos_term[OF pt'] sg p'[unfolded Nil]
          have "∃ C' σ. t = C'⟨ci⟨xi⟩ ⋅ σ⟩ ∧ p = hole_pos C' @ hole_pos ci" unfolding ci by force
          with pH show ?thesis unfolding fpstep_cond_def fpstep_cond_single_def by fast
        next
          case (Cons j p2')
          from hole_pos_ctxt_of_pos_term[OF ip]  hole_pos_ctxt_of_pos_term[OF pt'] sg p'[unfolded Cons]
          have "∃ C' σ. t = C'⟨ci⟨xi⟩ ⋅ σ⟩ ∧ hole_pos C' @ hole_pos ci <⇩p p" unfolding ci less_pos_def less_eq_pos_def 
            by auto
          with pB show ?thesis unfolding fpstep_cond_def fpstep_cond_single_def by fast
        qed
        with fp show False by auto 
      qed
      with t show "∃f ts. t |_ p1 = Fun f ts ∧ i ∈ c (f, length ts)" by auto
    qed
  qed
next
  assume cs:"csstep_cond c p t" show "fpstep_cond (to_pi c) p t" unfolding fpstep_cond_def
  proof
    fix pt
    assume pt_pi:"pt ∈ to_pi c"
    show "fpstep_cond_single pt p t"
    proof(cases pt)
      fix C u loc
      assume pt:"pt = (C, u, loc)"
      from pt_pi[unfolded pt to_pi_def] obtain f a xs i where ia:"i < a" and nic:"i ∉ c (f,a)" and 
        C:"C = ctxt_of_pos_term [i] (Fun f xs)" and xs:"xs = map Var (fresh_strings_list ''x'' 0 [] a)" and
        u:"u = xs ! i" and BH:"loc ∈ {location.B, location.H}" by blast
      let ?fresh = "fresh_strings_list ''x'' 0 [] a"
      let ?C = "ctxt_of_pos_term [i] (Fun f xs)"
      from xs fresh_name_gen_for_strings_list[unfolded fresh_name_gen_list_def] have 
        len:"length xs = a ∧ distinct ?fresh" by (metis length_map)
      then have len:"length xs = a" and dist:"distinct ?fresh" by auto
      from ia len have iposs:"[i] ∈ poss (Fun f xs)" by simp
      from C hole_pos_ctxt_of_pos_term[OF this] have Ci:"hole_pos C = [i]" by auto
      from cs[unfolded csstep_cond_def] have 
        "∀ p1 i p2. p = p1 @ i # p2 ∧ p ∈ poss t ⟶ (∃f ts. t |_ p1 = Fun f ts ∧ i ∈ c (f, length ts))" by auto
      show "fpstep_cond_single pt p t" unfolding fpstep_cond_single_def pt split
      proof
        assume ex:"∃C' σ. t = C'⟨C⟨u⟩ ⋅ σ⟩ ∧
           (loc = location.H ∧ p = hole_pos C' @ hole_pos C ∨ loc = location.A ∧ p <⇩p hole_pos C' @ hole_pos C ∨
            loc = location.B ∧ hole_pos C' @ hole_pos C <⇩p p ∨ loc = location.R ∧ right_of_pos p (hole_pos C'))"
        from BH ex obtain C' σ where t:"t = C'⟨C⟨u⟩ ⋅ σ⟩" and HB:"(loc = location.H ∧ p = hole_pos C' @ hole_pos C) ∨ 
     (loc = location.B ∧ hole_pos C' @ hole_pos C <⇩p p)" (is "?H ∨ ?B") by auto
        let ?tp1 = "t |_ hole_pos C'"
        have "∃f ts. ?tp1 = Fun f ts ∧ i ∈ c (f, length ts)"
        proof(cases ?H)
          case True
          with Ci have p:"p = hole_pos C' @ [i]" by auto
          from cs[unfolded csstep_cond_def] posst p show ?thesis by auto
        next
          case False
          with HB have B:"loc = location.B" and p:"hole_pos C' @ hole_pos C <⇩p p" by auto
          then obtain q where p:"p = hole_pos C' @ [i] @ q" 
            unfolding less_pos_def less_eq_pos_def Ci by auto
          from cs[unfolded csstep_cond_def] posst p show ?thesis by auto
        qed
        then obtain f' ts' where tp1:"?tp1 = Fun f' ts'" and ic:"i ∈ c (f', length ts')" by blast
        from subt_at_hole_pos have tp1':"?tp1 = ?C⟨u⟩ ⋅ σ" unfolding t[unfolded C] by fast
        from tp1[unfolded tp1' u] ctxt_supt_id iposs weak_match_match have "f = f'" and "length xs = length ts'" by auto
        with ic nic show False unfolding len by auto
      qed
    qed
  qed
qed

lemma csstep_r_p_s_iff_fpstep_r_p_s: 
  "csstep_r_p_s μ rs lr p σ s t = fpstep_r_p_s (to_pi μ) rs lr p σ s t"
  unfolding csstep_r_p_s_def fpstep_r_p_s_def rstep_r_p_s_def
  using fp_is_cs_cond by (metis (lifting) mem_Collect_eq split_conv)

lemma csstep_p_iff_fpstep_p: 
  "csstep_p μ trs p = fpstep_p (to_pi μ) trs p" (is "?C = ?F")
proof(rule, rule)
  fix s t
  assume "(s,t) ∈ csstep_p μ trs p"
  from this[unfolded csstep_p_def]  have r:"∃ lr σ. (s,t) ∈ rstep_r_p_s trs lr p σ" and cs:"csstep_cond μ p s" by auto
  from r have "p ∈ poss s" unfolding rstep_r_p_s_def mem_Collect_eq 
    using rstep_r_p_s_def' by force
  from r fp_is_cs_cond[OF this] cs show "(s,t) ∈ fpstep_p (to_pi μ) trs p" unfolding fpstep_p_def by auto
next
  {
    fix s t
    assume "(s,t) ∈ fpstep_p (to_pi μ) trs p" (is ?FP)
    from this[unfolded fpstep_p_def]  have r:"∃ lr σ. (s,t) ∈ rstep_r_p_s trs lr p σ" and cs:"fpstep_cond(to_pi μ) p s" by auto
    from r have "p ∈ poss s" unfolding rstep_r_p_s_def mem_Collect_eq 
      using rstep_r_p_s_def' by force
    from r fp_is_cs_cond[OF this] cs have "(s,t) ∈ csstep_p μ trs p" unfolding csstep_p_def by auto
  }
  then show "?F ⊆ ?C" by (rule subrelI)
qed

lemma csstep_iff_fpstep: "csstep μ trs = fpstep (to_pi μ) trs"
  unfolding csstep_def fpstep_def using  csstep_p_iff_fpstep_p by blast

end
