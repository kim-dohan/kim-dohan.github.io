fun ctxt_subst :: "('f,'v)ctxt ⇒ ('f,'v)subst ⇒ nat ⇒ ('f,'v)term ⇒ ('f,'v)term"
  where "ctxt_subst _ _ 0 t = t"
     |  "ctxt_subst C μ (Suc n) t = C⟨(ctxt_subst C μ n t) ⋅ μ⟩"

fun ctxt_subst_ctxt :: "('f,'v)ctxt ⇒ ('f,'v)subst ⇒ nat ⇒ ('f,'v)ctxt ⇒ ('f,'v)ctxt"
  where "ctxt_subst_ctxt _ _ 0 D = D"
     |  "ctxt_subst_ctxt C μ (Suc n) D = C ∘⇩c (ctxt_subst_ctxt C μ n D ⋅⇩c μ)"


lemma ctxt_subst_ctxt: "ctxt_subst C μ n (D⟨t⟩) = (ctxt_subst_ctxt C μ n D)⟨t ⋅ μ^^n⟩"
  by (induct n, auto simp: subst_left_right)

lemma ctxt_subst_add[simp]: "ctxt_subst C μ (n + m) t = ctxt_subst C μ n (ctxt_subst C μ m t)"
  by (induct n, auto)

lemma ctxt_subst_ctxt_add[simp]: "ctxt_subst_ctxt C μ (n + m) D = ctxt_subst_ctxt C μ n (ctxt_subst_ctxt C μ m D)"
  by (induct n, auto)

lemma ctxt_subst_Suc: "ctxt_subst C μ (Suc n) t = ctxt_subst C μ n (C⟨t⋅μ⟩)" (is "?l = _")
proof -
  have "?l = ctxt_subst C μ (n + Suc 0) t" by simp
  from this[unfolded ctxt_subst_add] show ?thesis by simp
qed

lemma ctxt_subst_ctxt_Suc: "ctxt_subst_ctxt C μ (Suc n) D = ctxt_subst_ctxt C μ n (C ∘⇩c (D ⋅⇩c μ))" (is "?l = _")
proof -
  have "?l = ctxt_subst_ctxt C μ (n + Suc 0) D" by simp
  from this[unfolded ctxt_subst_ctxt_add] show ?thesis by simp
qed

lemma ctxt_subst_hole_pos[simp]: "hole_pos C ^ n ∈ poss (ctxt_subst C μ n t)"
  by (induct n, auto)

lemma ctxt_subst_ctxt_hole_pos[simp]: "hole_pos (ctxt_subst_ctxt C μ n D) = hole_pos C ^ n @ hole_pos D"
  by (induct n, auto)

lemma ctxt_subst_subst: "(ctxt_subst C μ n t) ⋅ μ = ctxt_subst (C ⋅⇩c μ) μ n (t ⋅ μ)"
  by (induct n, auto) 

lemma ctxt_subst_subst_pow: "(ctxt_subst C μ n t) ⋅ μ ^^ k= ctxt_subst (C ⋅⇩c (μ ^^ k)) μ n (t ⋅ (μ ^^ k))"
 proof (induct k, simp)
  case (Suc j)
   have "(ctxt_subst C μ n t) ⋅ μ ^^ (Suc j) = ((ctxt_subst C μ n t) ⋅ μ ^^ j) ⋅ μ" using subst_right_left by auto
   also have "… =  ctxt_subst (C ⋅⇩c (μ ^^ j)) μ n (t ⋅ (μ ^^ j)) ⋅ μ " using Suc by auto
   also have "… =  ctxt_subst (C ⋅⇩c (μ ^^ j) ⋅⇩c μ) μ n (t ⋅ (μ ^^ j) ⋅ μ)" using ctxt_subst_subst by auto
   finally show ?case using ctxt_compose_subst_compose_distrib subst_subst subst_power_Suc by metis
 qed

lemma ctxt_subst_subt_at[simp]: "(ctxt_subst C μ n t) |_ (hole_pos C ^ n) = t ⋅ μ^^n"
proof (induct n)
  case (Suc n)
  let ?p = "hole_pos C"
  have "(ctxt_subst C μ (Suc n) t) |_ (?p ^ Suc n) = C⟨ctxt_subst C μ n t ⋅ μ⟩ |_ (?p @ ?p ^ n)" by simp
  also have "... = (ctxt_subst C μ n t ⋅ μ) |_ (?p ^ n)" using subt_at_append[OF hole_pos_poss[of C]] by simp
  also have "... = ((ctxt_subst C μ n t) |_ (?p ^ n)) ⋅ μ" by (rule subt_at_subst[OF ctxt_subst_hole_pos])
  also have "... = (t ⋅ μ^^n) ⋅ μ" unfolding Suc by simp
  finally show ?case using subst_left_right by simp
qed simp

lemma ctxt_subst_ctxt_of_pos_term: assumes q: "q ∈ poss s"
  shows "ctxt_subst_ctxt C μ n (ctxt_of_pos_term q s) = ctxt_of_pos_term ((hole_pos C)^n @ q) (ctxt_subst C μ n s)"
proof (induct n)
  case (Suc n)
  have hp: "hole_pos C ^ Suc n = hole_pos C @ hole_pos C ^ n" by auto
  have poss: "hole_pos C ^ n @ q ∈ poss (ctxt_subst C μ n s)"
    unfolding poss_append_poss using q by auto
  show ?case 
    unfolding ctxt_subst_ctxt.simps
    unfolding Suc
    unfolding ctxt_subst.simps
    unfolding hp 
    by (simp add: ctxt_of_pos_term_append[OF hole_pos_poss] ctxt_of_pos_term_subst[OF poss])
qed simp

lemma ctxt_subst_step: fixes C :: "('f,'v)ctxt"
  defines p: "p ≡ hole_pos C"
  assumes step: "(s,t) ∈ rstep_r_p_s R lr q σ"
  shows "(ctxt_subst C μ n s, ctxt_subst C μ n t) ∈ rstep_r_p_s R lr ((p^n) @ q) (σ ∘⇩s μ^^n)"
proof -
  obtain l r where l_r: "lr = (l,r)" by force
  from step[unfolded rstep_r_p_s_def' l_r]
  have q: "q ∈ poss s" and lr: "(l,r) ∈ R" and s: "s |_ q = l ⋅ σ" and t: "t = replace_at s q (r ⋅ σ)" by auto
  show ?thesis unfolding rstep_r_p_s_def'
  proof(rule, unfold split l_r fst_conv snd_conv, intro conjI)
    show "(l,r) ∈ R" by (rule lr)
  next
    show "p ^ n @ q ∈ poss (ctxt_subst C μ n s)" 
      unfolding poss_append_poss p
      using q by auto
  next
    show "(ctxt_subst C μ n s) |_ (p ^ n @ q) = l ⋅ σ ∘⇩s μ ^^ n"
      unfolding p
      unfolding subt_at_append[OF ctxt_subst_hole_pos]
      unfolding ctxt_subst_subt_at
      unfolding subt_at_subst[OF q] s
      by simp
  next
    show "ctxt_subst C μ n t = (replace_at (ctxt_subst C μ n s) (p ^ n @ q) (r ⋅ σ ∘⇩s μ ^^ n))" (is "?l = ?r")
    proof -        
      have "?l = ctxt_subst C μ n (replace_at s q (r ⋅ σ))" unfolding t ..
      also have "... = ?r" unfolding ctxt_subst_ctxt p unfolding ctxt_subst_ctxt_of_pos_term[OF q] by simp
      finally show "?l = ?r" .
    qed
  qed
qed



lemma ctxt_subst_pos_cases: assumes "p ∈ poss(ctxt_subst C μ n t)" shows
"(∃p'. p = hole_pos C ^ n @ p' ∧ p' ∈ poss t) ∨
 (∃x p' p''. p = hole_pos C ^ n @ p' @ p'' ∧ p' ∈ poss t ∧ t|_p' = Var x ∧ 
  p'' ∈ poss (Var x ⋅ μ ^^ n) ∧ n > 0) ∨
 (∃ k p'. k < n ∧ p = hole_pos C ^ k @ p' ∧ p' ∈ possc C) ∨
 (∃ k x p' p''. k < n ∧ p = hole_pos C ^ k @ p' @ p'' ∧ p' ∈ possc C - {hole_pos C} ∧ 
  C⟨t⟩|_p' = Var x ∧ p'' ∈ poss (Var x ⋅ μ ^^ k))"
proof-
 let ?case1 = "λ C t n p p'. p = hole_pos C ^ n @ p' ∧ p' ∈ poss t"
 let ?case2 = "λ C t n p p' p'' x. p = hole_pos C ^ n @ p' @ p'' ∧ p' ∈ poss t ∧ 
  t|_p' = Var x ∧ p'' ∈ poss (Var x ⋅ μ ^^ n) ∧ n > 0"
 let ?case3 = "λ C t n k p p'. k < n ∧ p = hole_pos C ^ k @ p' ∧ p' ∈ possc C"
 let ?case4 = "λ C t n k p p' p'' x.  k < n ∧ p = hole_pos C ^ k @ p' @ p'' ∧ 
  p' ∈ possc C - {hole_pos C} ∧ C⟨t⟩|_p' = Var x ∧ p'' ∈ poss (Var x ⋅ μ ^^ k)"
 let ?th = "λ C t n p. (∃ p'. ?case1 C t n p p') ∨ (∃ x p' p''. ?case2 C t n p p' p'' x) ∨ 
  (∃ k p'. ?case3 C t n k p p') ∨ (∃ k x p' p''. ?case4 C t n k p p' p'' x)"  
 from assms show ?thesis proof(induct n arbitrary:C t p)
 case 0
  then have "p ∈ poss t" by simp
  then show ?case by auto
 next
 case (Suc n)
  then have pposs:"p ∈ poss (C⟨ctxt_subst C μ n t ⋅ μ⟩)" and 
   IH:"⋀q' C t. q' ∈ poss (ctxt_subst C μ n t) ⟹ ?th C t n q'" by auto
  let ?q = "hole_pos C"
  show ?case
  proof (cases "?q ≤⇩p p")
   case False note Case3=this
     with pos_cases have "p <⇩p ?q ∨ ?q ⊥ p" by blast
     from this[unfolded less_pos_def ] par_hole_pos_in_possc[OF _ pposs] less_eq_hole_pos_in_possc 
      have "p ∈ possc C" by blast
     then have "?case3 C t (Suc n) 0 p p" by auto
     then show ?thesis by blast
   next
   case True 
    then obtain q where p:"p = ?q @ q" unfolding less_eq_pos_def by auto
    show ?thesis proof(cases "q ∈ poss (ctxt_subst C μ n t)")
     case True 
      with p IH have ih:"?th C t n q" (is "?A ∨ ?B ∨ ?C ∨ ?D") by auto
       show ?thesis proof(cases "¬ ?A", cases "¬ ?B", cases "¬ ?C")
         (* case (iv) *)
         assume "¬ ?A" and "¬ ?B" and "¬ ?C"
         with ih have "?D" by fast
         then obtain k x p' p'' where a:"k < n" and b:"q = ?q ^ k @ p' @ p''" and 
           c:"p' ∈ possc C - {hole_pos C} ∧ C⟨t⟩ |_ p' = Var x" and d:"p'' ∈ poss (Var x ⋅ μ ^^ k)" by auto
         then have a:"Suc k < Suc n" and b:"p = ?q ^(Suc k) @ p' @ p''" unfolding p by auto
         from poss_imp_subst_poss[OF d, of μ] 
           subst_subst[of "Var x" "μ ^^ k" μ, unfolded subst_power_Suc[symmetric]] 
           have d':"p'' ∈ poss (Var x ⋅ μ ^^ (Suc k))" by fastforce
         with a b c have "?case4 C t (Suc n) (Suc k) p p' p'' x" by fast
         then show ?thesis by metis
        next
         (* case (iii) *)
         assume "¬ ¬(∃ k p'. ?case3 C t n k q p')" 
         then obtain k p' where "k < n ∧ q = ?q ^ k @ p' ∧ p' ∈ possc C" by auto
         then have "Suc k < Suc n ∧ p = ?q ^ Suc k @ p' ∧ p' ∈ possc C" using p by auto
         then show ?thesis by blast
        next
         (* case (ii) *)
         assume "¬ ¬(∃ x p' p''. ?case2 C t n q p' p'' x)"
         then obtain x p' p'' where "q = ?q ^ n @ p' @ p''" and 
           a:"p' ∈ poss t ∧ t |_ p' = Var x" and b:"p'' ∈ poss (Var x ⋅ μ ^^ n)" by auto
         then have c:"p = hole_pos C ^ Suc n @ p' @ p''" unfolding p by auto
         from  subst_subst[of "Var x" "μ ^^ n" μ, unfolded subst_power_Suc[symmetric]] 
            poss_imp_subst_poss[OF b, of μ] have "p'' ∈ poss (Var x ⋅ μ ^^ Suc n)" by auto
         with a c show ?thesis by blast
        next
         (* case (i) *)
         assume "¬ ¬(∃ p'. ?case1 C t n q p')"
         with p show ?thesis by auto
       qed
     next
      assume qn:"q ∉ poss (ctxt_subst C μ n t)"
      from pposs p have qposs:"q ∈ poss (ctxt_subst C μ n t ⋅ μ)" using hole_pos_poss_conv by fast
      have s:"hole_pos (C ⋅⇩c μ) = ?q" by simp
      from qposs have "q ∈ poss (ctxt_subst (C ⋅⇩c μ) μ n (t ⋅ μ))" unfolding ctxt_subst_subst by simp
      with IH[OF this] have ih:"?th (C ⋅⇩c μ) (t ⋅ μ) n q" (is "?A ∨ ?B ∨ ?C ∨ ?D") by auto
      show ?thesis proof(cases "¬ ?A", cases "¬ ?B", cases "¬ ?C")
      (* case (iv) *)
      assume  "¬ ?A" and "¬ ?B" and "¬ ?C"
        with ih have "?D" by fast
        then obtain k x p' p'' where kn:"k < n" and q:"q = ?q ^ k @ p' @ p''" and 
          pposs0:"p' ∈ possc (C ⋅⇩c μ) - {hole_pos C}" and x:"C⟨t⟩ ⋅ μ |_ p' = Var x" and 
          pposs'':"p'' ∈ poss (Var x ⋅ μ ^^ k)" unfolding s by auto
        from pposs0[unfolded possc_def] have pposs':"p' ∈ poss (C⟨t⟩ ⋅ μ)" 
         using subst_apply_term_ctxt_apply_distrib[of C t μ] by simp
        have "¬ (is_Fun (C⟨t⟩ ⋅ μ |_ p'))" using x by auto
        then have yeah:"¬ (p' ∈ poss C⟨t⟩ ∧ is_Fun (C⟨t⟩ |_ p'))"
          by (metis is_VarE is_VarI subst_apply_eq_Var subt_at_subst)
        with pos_into_subst[OF _ pposs', of "C⟨t⟩"] obtain q' q'' y where 
          p':"p' = q' @ q''" and 2:"q' ∈ poss C⟨t⟩" and 3:"C⟨t⟩ |_ q' = Var y" by blast
        have q':"q' ∈ possc (C ⋅⇩c μ)" unfolding possc_def proof(rule,simp, rule allI) 
         fix t
         from DiffD1[OF pposs0[unfolded p']] 
         show "q' ∈ poss (C ⋅⇩c μ)⟨t⟩" using poss_append_poss poss_imp_possc by metis
        qed
        from possc_subst_not_possc_not_poss[OF q'] 2 have 22:"q' ∈ possc C" by fast
        from pposs0 have "p' ≠ hole_pos C" by fast
        with p' possc_not_below_hole_pos[OF DiffD1[OF pposs0], unfolded hole_pos_subst] have "q' ≠ hole_pos C" 
         using less_pos_def by fastforce
        with q' 22 have 22:"q' ∈ possc C - {hole_pos C}" by auto
        from p[unfolded q p'] have 1:"p = ?q ^ (Suc k) @ q' @ q'' @ p''" by auto
        from subt_at_subst[OF 2] x[unfolded p' subt_at_append[OF poss_imp_subst_poss[OF 2]]] 3
          have "Var y ⋅ μ |_ q'' = Var x" by metis
        then have 42:"Var y ⋅ μ |_ q'' ⋅ μ ^^ k = Var x ⋅ μ ^^ k" by auto
        from poss_append_poss[of q' q'' "C⟨t⟩ ⋅ μ"] pposs'[unfolded p'] 3 have 
          41:"q'' ∈ poss (Var y ⋅ μ)" by (metis "2" subt_at_subst)
        then have "q'' ∈ poss (Var y ⋅ μ ⋅ μ ^^ k)" by (metis poss_imp_subst_poss)
        then have 40:"q'' ∈ poss (Var y ⋅ μ ^^ Suc k)"
          by (simp add: subst_compose)
        have "Var y ⋅ μ ^^ Suc k |_ q'' = Var y ⋅ μ |_ q'' ⋅ μ ^^ k" using subt_at_subst[OF 41] by (auto simp add: subst_compose)
        with pposs''[unfolded 42[symmetric]] have "p'' ∈ poss (Var y ⋅ μ ^^ Suc k |_ q'')" by auto
        with 40 poss_append_poss[of q'' p'' "Var y ⋅ μ ^^ (Suc k)"] 
        have 4:"(q'' @ p'') ∈ poss (Var y ⋅ μ ^^ (Suc k))" by fast
        from kn 1 22 3 4 have  "?case4 C t (Suc n) (Suc k) p q' (q''@ p'') y" by (metis Suc_less_eq)
        then show ?thesis by metis
      next
      (* case (iii) *)
      assume "¬ ¬(∃ k p'. ?case3 (C ⋅⇩c μ) (t ⋅ μ) n k q p')" 
      with s obtain k p' where ih:"k < n ∧ q = ?q ^ k @ p'" and ppossc':"p' ∈ possc (C ⋅⇩c μ)" by auto
      then have kn:"Suc k < Suc n" and p:"p = ?q ^ Suc k @ p'" using p by auto
      show ?thesis proof(cases "p' ∈ possc C")
        case True
          with kn p show ?thesis by blast
        next
        case False
         have np:"p' ∉ poss C⟨t⟩" using possc_subst_not_possc_not_poss[OF ppossc' False] by auto
         from False have 33:"p' ≠ hole_pos C" using hole_pos_in_possc by auto
         from ppossc'[unfolded possc_def] subst_apply_term_ctxt_apply_distrib[of C t μ] 
          have pposs':"p' ∈ poss (C⟨t⟩ ⋅ μ)" by fastforce
         from pos_into_subst[OF _ this, of "C⟨t⟩"] np 
         have "∃q q' x. p' = q @ q' ∧ q ∈ poss C⟨t⟩ ∧ C⟨t⟩ |_ q = Var x" by fast
         then obtain q' q'' x where p':"p' = q' @ q''" and 3:"q' ∈ poss C⟨t⟩" and 
           4:"C⟨t⟩ |_ q' = Var x" by blast
         from 33 hole_pos_poss_conv possc_subst_not_possc_not_poss[OF ppossc', unfolded p'] 
          have 33:"q' ≠ hole_pos C" by (metis False p' poss_append_poss pposs')
         have a3:"q' ∈ possc C - {hole_pos C}" 
          proof-
           from possc_not_below_hole_pos[OF ppossc', unfolded hole_pos_subst] have nlt:"¬ hole_pos C <⇩p p'" by auto
           with pos_cases 33 less_pos_def have "p' <⇩p hole_pos C ∨ p' ⊥ hole_pos C" 
             by (metis False hole_pos_in_possc)
           with p' pos_cases[of "q'" "hole_pos C"] have "q' ≤⇩p hole_pos C ∨ q' ⊥ hole_pos C"
             apply auto
             using less_pos_simps(5) order_pos.less_trans apply blast
             by (metis append_Nil2 less_pos_simps(1) nlt order_pos.dual_order.strict_implies_order order_pos.dual_order.strict_trans2)
           then show ?thesis using less_eq_hole_pos_in_possc[of q' C] par_hole_pos_in_possc[OF _ 3] parallel_pos_sym 33 by fast
          qed 
          from poss_append_poss[of q' q'' "C⟨t⟩ ⋅ μ"] pposs'[unfolded p'] have 
           "q'' ∈ poss (C⟨t⟩ ⋅ μ |_ q')" by auto
         with subt_at_subst[OF 3] 4 have 5:"q'' ∈ poss (Var x ⋅ μ)" by metis
         then have 5:"q'' ∈ poss (Var x ⋅ μ ^^ Suc k)" using poss_imp_subst_poss by (auto simp add: subst_compose)
         from kn p[unfolded p'] a3 4 5 have "?case4 C t (Suc n) (Suc k) p q' q'' x" by blast
         then show ?thesis by blast
        qed
        next
       (* case (ii) *)
          assume "¬ ¬(∃ x p' p''. ?case2 (C ⋅⇩c μ) (t ⋅ μ) n q p' p'' x)"
          then obtain x p' p'' where q:"q = ?q ^ n @ p' @ p''" and 
           a:"p' ∈ poss (t ⋅ μ)" and x:"(t ⋅ μ) |_ p' = Var x" and 
           c:"p'' ∈ poss (Var x ⋅ μ ^^ n)" by auto
          have "¬ (is_Fun (t ⋅ μ |_ p'))" using x by auto
          then have yeah:"¬ (p' ∈ poss t ∧ is_Fun (t |_ p'))"
             by (metis is_VarE is_VarI subst_apply_eq_Var subt_at_subst)
          with pos_into_subst[OF _ a] have 
             "∃q' q'' y. p' = q' @ q'' ∧ q' ∈ poss t ∧ t |_ q' = Var y" by auto
          then obtain q' q'' y where p':"p' = q' @ q''" and 2:"q' ∈ poss t"
             and 3:"t |_ q' = Var y" by auto
          from p[unfolded q p'] have 1:"p = ?q ^ (Suc n) @ q' @ q'' @ p''" by auto
          from x[unfolded p' subt_at_append[OF poss_imp_subst_poss[OF 2]]  subt_at_subst[OF 2]]
             have "t |_ q' ⋅ μ |_ q'' = Var x" by fast
          with 3 have 42:"Var y ⋅ μ |_ q'' ⋅ μ ^^ n = Var x ⋅ μ ^^ n" by auto
          from poss_append_poss[of q' q'' "t ⋅ μ", unfolded subt_at_subst[OF 2] 3] a[unfolded p']
            poss_imp_subst_poss[OF 2] have 41:"q'' ∈ poss (Var y ⋅ μ)" by auto
          then have "q'' ∈ poss (Var y ⋅ μ ⋅ μ ^^ n)" by (metis poss_imp_subst_poss)
          then have 40:"q'' ∈ poss (Var y ⋅ μ ^^ Suc n)" by (auto simp add: subst_compose)
          have "Var y ⋅ μ ^^ Suc n |_ q'' = Var y ⋅ μ |_ q'' ⋅ μ ^^ n" using subt_at_subst[OF 41] by (auto simp add: subst_compose)
          with c[unfolded 42[symmetric]] have "p'' ∈ poss (Var y ⋅ μ ^^ Suc n |_ q'')" by auto
          with 40 poss_append_poss[of q'' p'' "Var y ⋅ μ ^^ (Suc n)"] 
            have 4:"(q'' @ p'') ∈ poss (Var y ⋅ μ ^^ (Suc n))" by fast
          from 1 2 3 4 have  "?case2 C t (Suc n) p q' (q''@ p'') y" by auto
          then show ?thesis by metis
          next
           assume "¬ ¬(∃ p'. ?case1 (C ⋅⇩c μ) (t ⋅ μ) n q p')"
           then obtain p' where q:"q = ?q ^ n @ p'" and pposs':"p' ∈ poss (t ⋅ μ)" by auto
           show ?thesis proof(cases "p' ∈ poss t")
            case True
             with p[unfolded q] have "?case1 C t (Suc n) p p'" by auto
             then show ?thesis by blast
            next
            case False
             from pos_into_subst[OF _ pposs', of t] False have
               "∃q q' x. p' = q @ q' ∧ q ∈ poss t ∧ t |_ q = Var x" by auto
             then obtain q q' x where p':"p' = q @ q'" and 2:"q ∈ poss t"
               and 3:"t |_ q = Var x" by auto
             from p[unfolded q p'] have 1:"p = ?q ^ Suc n @ q @ q'" by auto
             from poss_append_poss[of q q'] pposs'[unfolded p'] poss_imp_subst_poss[OF 2]
              have "q' ∈ poss (t ⋅ μ |_ q)" by auto
             with subt_at_subst[OF 2] 3 have "q' ∈ poss (Var x ⋅ μ)" by metis
             with poss_imp_subst_poss[OF this] have 4:"q' ∈ poss (Var x ⋅ μ ^^ Suc n)" by (force simp add: subst_compose)
             from 1 2 3 4 have  "?case2 C t (Suc n) p q q' x" by auto
             then show ?thesis by blast
            qed
          qed
     qed
   qed
 qed
qed
