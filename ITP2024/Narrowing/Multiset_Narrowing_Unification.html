section ‹Formalization of multiset narrowing based (multiset) E-unification and E-unifiability›

theory Multiset_Narrowing_Unification
  imports
    Multiset_Narrowing_Reachability
begin

locale multiset_narrowing_unification =  multiset_narrowing_reachability R DOTEQ TOP R' F D x
  for R::"('f, 'v:: infinite) trs" 
    and DOTEQ :: 'f ("≐")
    and TOP :: "'f" ("⊤") 
    and R' :: "('f, 'v:: infinite) trs" 
    and F :: "'f sig" 
    and D :: "'f sig"
    and x :: 'v
begin

lemma rstep_single_R_unifiability_step_eq: assumes cr: "CR (multiset_reduction_step)"
  and funas_uv: "funas_rule (u, v) ⊆ F"
  and  uv:"(u, v) ∈ (rstep R)⇧↔⇧*"
  shows "(∃T. ({#Fun ≐ [u, v]#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))" 
proof -
  from uv obtain n where "(u::('f, 'v)term, v::('f, 'v)term) ∈ ((rstep R)⇧↔)^^n" by auto
  then show ?thesis
  proof(induct n arbitrary:u v)
    case 0
    then show ?case 
      by (metis multiset_red_refl insert_DiffM r_into_rtrancl relpow_0_E single_eq_add_mset)
  next
    case (Suc n)
    from Suc(2) obtain w where uw:"(u, w) ∈ (rstep R)⇧↔" and wv:"(w, v) ∈ (rstep R)⇧↔ ^^ n"
      by (meson relpow_Suc_E2)
    from wv have vw:"(v, w) ∈ (rstep R)⇧↔ ^^ n" by (metis converseI converse_power symcl_converse)
    from uw obtain C rl σ where rl:"rl ∈ R" and uw:"(u = C⟨(fst rl) ⋅ σ⟩ ∧ w = C⟨(snd rl) ⋅ σ⟩) ∨ 
      (w = C⟨(fst rl) ⋅ σ⟩ ∧ u = C⟨(snd rl) ⋅ σ⟩)" by fastforce
    then obtain T where IH:"({#Fun ≐ [w, v]#}, T) ∈ multiset_reduction_step⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" using Suc(1) 
      using wv by blast
    from uw show ?case
    proof
      assume asm:"u = C⟨(fst rl) ⋅ σ⟩ ∧ w = C⟨(snd rl) ⋅ σ⟩"
      have "is_Fun (fst rl)" using wf[unfolded wf_trs_def] rl
        by (metis is_Fun_Fun_conv prod.collapse)
      hence hf:"hole_pos C ∈ fun_poss u" 
        by (metis asm hole_pos_poss is_Var_def poss_is_Fun_fun_poss subst_apply_eq_Var subt_at_hole_pos)
      let ?p = "0 # hole_pos C"
      let ?S = "{# Fun ≐ [u, v] #}"
      let ?s = "Fun ≐ [u, v]"
      let ?rl = rl
      let ?σ = σ
      let ?T = "(?S - {#?s#}) + {#replace_at ?s  ?p ((snd ?rl) ⋅ ?σ)#}"
      have spσ:"?s |_ ?p = (fst ?rl) ⋅ ?σ" using asm by auto
      have "rl ∈ R" using R' rl by fastforce
      have pfun:"?p ∈ poss ?s" using hf by (simp add: asm)
      have st_step:"(?S, ?T) ∈ multiset_reduction_step_pos" 
      proof - 
        have "?s ∈# {#Fun ≐ [u, v]#}" by simp
        moreover have "?rl ∈ R" using R' rl by fastforce
        moreover have "?p ∈ poss ?s" using hf pfun by fastforce
        moreover have "?s |_ ?p = fst ?rl ⋅ ?σ" using spσ by auto
        ultimately show ?thesis by blast
      qed
      hence st_step:"(?S, ?T) ∈ multiset_reduction_step" by (metis ctxt_supt_id multiset_reduction_stepI 
        pfun rl rstep_rule.intros rstep_rule_imp_rstep spσ union_single_eq_member) 
      hence "replace_at (Fun ≐ [u, v])  ?p ((snd rl) ⋅ σ) = (Fun ≐ [w, v])" using asm by auto
      hence "?T = {#Fun ≐ [w, v]#}"by fastforce
      hence *:"({#Fun ≐ [u, v]#}, {#Fun ≐ [w, v]#}) ∈ multiset_reduction_step" 
        using st_step by simp
      then show ?case using IH by (meson * converse_rtrancl_into_rtrancl)
    next
      assume asm:"w = C⟨(fst rl) ⋅ σ⟩ ∧ u = C⟨(snd rl) ⋅ σ⟩"
      have "is_Fun (fst rl)" using wf[unfolded wf_trs_def] rl 
        by (metis is_Fun_Fun_conv prod.collapse)
      hence hf:"hole_pos C ∈ fun_poss w" 
        by (metis asm hole_pos_poss is_Var_def poss_is_Fun_fun_poss subst_apply_eq_Var subt_at_hole_pos)
      let ?p = "0 # hole_pos C"
      let ?S = "{# Fun ≐ [w, v] #}"
      let ?s = "Fun ≐ [w, v]"
      let ?rl = rl
      let ?σ = σ
      let ?T = "(?S - {#?s#}) + {#replace_at ?s  ?p ((snd ?rl) ⋅ ?σ)#}"
      have spσ:"?s |_ ?p = (fst ?rl) ⋅ ?σ" using asm by auto
      have "rl ∈ R" using R' rl by fastforce
      have pfun:"?p ∈ poss ?s" using hf asm by auto
      have st_step:"(?S, ?T) ∈ multiset_reduction_step_pos" 
      proof - 
        have "?s ∈# {#Fun ≐ [w, v]#}" by simp
        moreover have "?rl ∈ R" using R' rl by fastforce
        moreover have "?p ∈ poss ?s" using hf pfun by blast
        moreover have "?s |_ ?p = fst ?rl ⋅ ?σ" using spσ by auto
        ultimately show ?thesis by blast
      qed
      hence "replace_at (Fun ≐ [w, v])  ?p ((snd rl) ⋅ σ) = (Fun ≐ [u, v])" using asm by auto
      hence "?T = {#Fun ≐ [u, v]#}"by fastforce
      hence "({#Fun ≐ [w, v]#}, {#Fun ≐ [u, v]#}) ∈ multiset_reduction_step_pos" 
        using st_step by simp
      hence *:"({#Fun ≐ [w, v]#}, {#Fun ≐ [u, v]#}) ∈ multiset_reduction_step" 
        using st_step mul_reduction_correspondence  by (smt (verit, ccfv_threshold) ctxt_supt_id 
            multiset_narrowing.multiset_reduction_step.simps multiset_narrowing.multiset_reduction_step_pos.cases 
            multiset_narrowing_axioms rstep_rule.intros rstep_rule_imp_rstep)
      have "∀t∈#T. t = Fun ⊤ []" using IH ..
      hence "T ∈ NF (multiset_reduction_step)" using not_reducible_T_erstep by fastforce
      then show ?case using IH cr[unfolded CR_on_def] exI[of _ T]  * 
        by (auto, metis NF_join_imp_reach r_into_rtrancl)
    qed
  qed
qed

lemma rstep_single_eq_reduction_step_eq_normal: assumes cr_mul:"CR (multiset_reduction_step)"
  and uv:"(u::('f, 'v)term, v::('f, 'v)term) ∈ (rstep R')⇧↔⇧*"
  shows "(∃T. ({#Fun ≐ [u, v]#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))"
proof -
  from uv obtain n where "(u::('f, 'v)term, v::('f, 'v)term) ∈ (rstep R')⇧↔^^n" by auto
  then show ?thesis
  proof(induct n arbitrary: v)
    case 0
    hence "u = v" by auto
    then show ?case 
      by (metis add_mset_eq_single multi_member_split multiset_red_refl r_into_rtrancl)
  next
    case (Suc n)
    from Suc(2) obtain w where "(u, w) ∈ (rstep R')⇧↔ ^^ n" and wv:"(w, v) ∈ (rstep R')⇧↔" by auto
    then obtain T where uwT:"({#Fun ≐ [u, w]#}, T) ∈ multiset_reduction_step⇧*" and tT:"(∀t∈#T. t = Fun ⊤ [])" 
      using Suc(1) by auto
    hence TNF:"T ∈ NF (multiset_reduction_step)" using not_reducible_T by fastforce
    from wv have "(w, v) ∈ (rstep R') ∨ (v, w) ∈ (rstep R')" by auto
    then show ?case
    proof
      assume "(w, v) ∈ (rstep R')"
      hence wv:"(w, v) ∈ (rstep R)" using R' by blast
      hence "(Fun ≐ [u, w], Fun ≐ [u, v]) ∈ (rstep R)" using rstep.intros
      proof -
        let ?C = "ctxt_of_pos_term [1] (Fun ≐ [u, w])"
        from wv have "(?C⟨w⟩, ?C⟨v⟩) ∈ (rstep R)" by blast
        then show ?thesis by auto
      qed
      hence *:"({#Fun ≐ [u, w]#}, {#Fun ≐ [u, v]#}) ∈ multiset_reduction_step" by auto
      show ?thesis by (rule exI[of _T], insert cr_mul[unfolded CR_on_def] TNF uwT tT *,  
        meson NF_join_imp_reach cr_mul partially_localize_CR)
    next
      assume "(v, w) ∈ (rstep R')"
      hence vw:"(v, w) ∈ (rstep R)" using R' by blast
      hence "(Fun ≐ [u, v], Fun ≐ [u, w]) ∈ (rstep R)" using rstep.intros
      proof -
        let ?C = "ctxt_of_pos_term [1] (Fun ≐ [u, v])"
        from vw have "(?C⟨v⟩, ?C⟨w⟩) ∈ (rstep R)" by blast
        then show ?thesis by auto
      qed
      hence *:"({#Fun ≐ [u, v]#}, {#Fun ≐ [u, w]#}) ∈ multiset_reduction_step" by fastforce
      show ?thesis by (rule exI[of _T], insert cr_mul TNF uwT tT *, auto) 
    qed
  qed
qed

lemma rstep_multiset_R_unifiability_eq: assumes funas_A:"funas_trs (set A) ⊆ F"
    and "∀u v. (u, v) ∈ set A ⟶ ((u, v) ∈ (rstep R)⇧↔⇧*)"
    and cr:"CR (multiset_reduction_step)"
  shows "(∃T. ({#Fun ≐ [u, v]. (u, v) ∈# mset A#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀t ∈# T. t = Fun (⊤) []))" using assms
proof(induct A)
  case Nil
  then show ?case by auto
next
  case (Cons pair A)
  hence ftA:"funas_trs (set A) ⊆ F" unfolding funas_trs_def funas_rule_def by auto
  from Cons have *:"∀u v. (u, v) ∈ set A ⟶ ((u, v) ∈ (rstep R)⇧↔⇧*)" by auto
  obtain u v where pair:"pair = (u, v)" by (meson surj_pair)
  have funas_u:"funas_term u ⊆ F" by (metis Cons.prems(1) lhs_wf list.set_intros(1) pair)
  have funas_v:"funas_term v ⊆ F" by (metis Cons.prems(1) list.set_intros(1) pair rhs_wf)
  have uvR:"(u, v) ∈ (rstep R)⇧↔⇧*" using Cons by (simp add: pair)
  hence uv:"(u, v) ∈ (rstep R')⇧↔⇧*" using funas_rstep_R' funas_u funas_v uvR 
    by (simp add: funas_defs(2) funas_rstep_R'_conv)
  obtain T where **:"({#Fun ≐ [u, v]. (u, v) ∈# mset A#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" 
    using Cons * ftA by auto
  hence "(u, v) ∈ (rstep R')⇧↔⇧*" using uv by auto
  hence uvv:"(u, v) ∈ (rstep R)⇧↔⇧*" using uv R' using uvR by blast
  have funas_uv: "funas_rule (u, v) ⊆ F" 
    by (simp add: funas_defs(2) funas_u funas_v)
  from rstep_single_R_unifiability_step_eq
  obtain T' where ***:"({#Fun ≐ [u, v]#}, T') ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T'. t = Fun ⊤ [])" using uv cr funas_uv 
    using uvv by blast
  have multiset_step_ctxt1:"({#Fun ≐ [u, v]#} + {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset A#},   
    T'+ {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset A#}) ∈ (multiset_reduction_step)⇧*" 
    using *** multiset_red_add by blast
  have multiset_step_ctxt2:"(T'+ {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset A#}, T' + T) ∈ (multiset_reduction_step)⇧*" using multiset_red_add ** add.commute by metis
  show ?case by (auto split:prod.splits, rule exI[of _ "T' + T"], insert ** *** pair addFunT[of T' T] 
        multiset_step_ctxt1 multiset_step_ctxt2, fastforce)
qed          

lemma rstep_multiset_unifiability_equiv:
  assumes funas_A:"funas_trs (set A) ⊆ F"
    and cstep_A:"∃τ. (∀u v. ((u, v) ∈ set A) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*))"
    and sn: "SN (rstep R)"
    and cr: "CR (multiset_reduction_step)"
  shows "∃θ T. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset A), T) ∈ (multiset_reduction_step)⇧*
          ∧ (∀ t ∈# T. t = Fun (⊤) [])" using assms
proof(induct A)
  case Nil
  hence nVar:"normal_subst R Var" using wf[unfolded wf_trs_def] 
    by (simp add: empty_subst_normalized normal_subst_def) 
  show ?case unfolding subst_term_multiset_def
    by (rule exI[of _ Var], rule exI[of _ "convert_equations_into_term_multiset []"], insert nVar, auto)
next
  case (Cons pair U)
  obtain u v where pair:"pair = (u, v)" using surjective_pairing by blast
  from funas_A have funU:"funas_trs (set U) ⊆ F" unfolding funas_trs_def funas_rule_def 
    using Cons by (meson lhs_wf rhs_wf subsetD subset_insertI, simp add: funas_rule_def funas_trs_def) 
  have rstepU':"∃τ. (∀s t. ((s, t) ∈ set ((u, v) # U)) ⟶ (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*)" 
    using assms Cons pair by auto
  then obtain τ where τ:"∀s t. ((s, t) ∈ set ((u, v) # U)) ⟶ (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*"  by auto
  from SN_obtains_normalizable_subst[OF sn]
  have normτ:"normalizable_subst R τ" by auto
  from obtains_normalized_subst[OF normτ]
  obtain τ' where normτ':"normal_subst R τ'" and subst_r:"(∀x. (τ x, τ' x) ∈ (rstep R)⇧*)" by auto
  hence ττ':"(∀x. (τ x, τ' x) ∈ (rstep R)⇧↔⇧*)" by auto
  have rstepτ':"∀s t. ((s, t) ∈ set ((u, v) # U)) ⟶ (s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*" 
  proof(intro impI allI)
    fix s t
    assume asm:"(s, t) ∈ set ((u, v) # U)"
    hence *:"(s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*" using τ asm by auto
    have "(s ⋅ τ, s ⋅ τ') ∈ (rstep R)⇧↔⇧*" using ττ' substs_rsteps subst_r by blast
    moreover have "(t ⋅ τ, t ⋅ τ') ∈ (rstep R)⇧↔⇧*" using ττ' 
      by (simp add: conversionI' subst_r substs_rsteps)
    ultimately show "(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*" 
      by (meson "*" conversion_inv conversion_trans transD)
  qed
  have ftτ':"funas_trs (set (subst_equations_list τ' ((u, v) # U))) ⊆ F" 
    unfolding subst_equations_list_def[unfolded subst_equation_def] funas_trs_def funas_rule_def 
    using wf_F_subst Cons(2) pair lhs_wf rhs_wf funas_A funU by (auto, smt (verit, ccfv_SIG) insertCI lhs_wf subsetD, 
        smt (verit, best) insertCI rhs_wf subsetD, blast, smt (verit) rhs_wf subsetD)
  show ?case
  proof(rule exI[of _ τ'], auto split:prod.splits, goal_cases)
    case (1 x1 x2)
    then show ?case 
      using normτ' by auto
  next
    case (2 x1 x2)
    have pair:"x1 = u ∧ x2 = v" using 2 pair by force
    have *:"∀s t. (s, t) ∈ set (subst_equations_list τ' ((u, v) # U)) ⟶ (s, t) ∈ (rstep R)⇧↔⇧*" 
      using rstepτ' unfolding subst_equations_list_def[unfolded subst_equation_def] by auto 
    from rstep_multiset_R_unifiability_eq[of "subst_equations_list τ' ((u, v) # U)"]
    obtain T where **:"(({#Fun ≐ [u, v]. (u, v) ∈# mset (subst_equations_list τ' ((u, v) # U)) #}, T) ∈ 
      (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))" using * ftτ' cr by auto
    have equiv:"{#t ⋅ τ'. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} =
        {#Fun ≐ [u, v]. (u, v) ∈# {#(fst p ⋅ τ', snd p ⋅ τ'). p ∈# mset U#}#}"
    proof -
      have "{#t ⋅ τ'. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#Fun ≐ [u ⋅ τ', v ⋅ τ']. (u, v) ∈# mset U#}"
      proof -
        have "{#t ⋅ τ'. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#(Fun ≐ [u, v]) ⋅ τ'. (u, v) ∈# mset U#}" 
          using subst_Fun_mset_equiv by auto
        also have "... = {#(Fun ≐ [u ⋅ τ', v ⋅ τ']). (u, v) ∈# mset U#}" by auto
        ultimately show ?thesis by auto
      qed
      then show ?thesis by (smt (verit, best) case_prod_conv comp_apply multiset.map_comp multiset.map_cong0 prod.collapse)
    qed     
    show ?case  unfolding subst_term_multiset_def subst_equations_list_def using ** pair 
      by (simp add:subst_equations_list_def subst_equation_def, insert equiv, auto) 
  qed
qed

lemma rstep_multiset_reduction_equiv_normal:
  assumes funas_A:"funas_trs (set A) ⊆ F"
    and cstep_A:"(∀u v. ((u, v) ∈ set A) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*))"
    and sn: "SN (rstep R)"
    and cr: "CR (rstep R)"
    and normτ: "normal_subst R τ"
  shows "∃T. (subst_term_multiset τ (convert_equations_into_term_multiset A), T) ∈ (multiset_reduction_step)⇧*
          ∧ (∀ t ∈# T. t = Fun (⊤) [])" using assms
proof(induct A)
  case Nil
  then show ?case by auto
next
  case (Cons pair U)
  from sn have sn_mul:"SN (multiset_reduction_step)" 
    by (simp add: SN_R_imp_SN_multiset_reduction_step)
  from sn_mul cr have cr_mul:"CR (multiset_reduction_step)" 
    by (simp add: CR_correspondence sn)
  obtain u v where pair:"pair = (u, v)" using surjective_pairing by blast
  from funas_A have funU:"funas_trs (set U) ⊆ F" unfolding funas_trs_def funas_rule_def 
    using Cons unfolding funas_defs by auto
  have rstepτ:"∀s t. ((s, t) ∈ set ((u, v) # U)) ⟶ (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧↔⇧*" using Cons pair by fastforce
  have ftτ:"funas_trs (set (subst_equations_list τ ((u, v) # U))) ⊆ F" 
    unfolding subst_equations_list_def[unfolded subst_equation_def] funas_trs_def funas_rule_def 
    using wf_F_subst Cons(2) pair lhs_wf rhs_wf funas_A funU by (auto, smt (verit, ccfv_SIG) insertCI lhs_wf subsetD, 
        smt (verit, best) insertCI rhs_wf subsetD, blast, smt (verit) rhs_wf subsetD)
  then show ?case
  proof(auto split:prod.splits, goal_cases)
   case (1 x1 x2)
    have pair:"x1 = u ∧ x2 = v" using 1 pair by force
    have *:"∀s t. (s, t) ∈ set (subst_equations_list τ ((u, v) # U)) ⟶ (s, t) ∈ (rstep R)⇧↔⇧*" 
      using rstepτ unfolding subst_equations_list_def[unfolded subst_equation_def] by auto 
    from rstep_multiset_R_unifiability_eq [of "subst_equations_list τ ((u, v) # U)"]
    obtain T where **:"(({#Fun ≐ [u, v]. (u, v) ∈# mset (subst_equations_list τ ((u, v) # U)) #}, T) ∈ 
      (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))" using * ftτ cr_mul by auto 
    have equiv:"{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} =
        {#Fun ≐ [u, v]. (u, v) ∈# {#(fst p ⋅ τ, snd p ⋅ τ). p ∈# mset U#}#}"
    proof -
      have "{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#Fun ≐ [u ⋅ τ, v ⋅ τ]. (u, v) ∈# mset U#}"
      proof -
        have "{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#(Fun ≐ [u, v]) ⋅ τ. (u, v) ∈# mset U#}" 
          using subst_Fun_mset_equiv by auto
        also have "... = {#(Fun ≐ [u ⋅ τ, v ⋅ τ]). (u, v) ∈# mset U#}" by auto
        ultimately show ?thesis by auto
      qed
      then show ?thesis by (smt (verit, best) case_prod_conv comp_apply multiset.map_comp multiset.map_cong0 prod.collapse)
    qed     
    show ?case unfolding subst_term_multiset_def subst_equations_list_def using ** pair 
      by (simp add:subst_equations_list_def subst_equation_def, insert equiv, auto) 
  qed
qed

lemma funas_term_pos_included:assumes "p ∈ poss t"
  and "funas_term t ⊆ F"
shows "funas_term (t |_ p) ⊆ F" using assms
  by (meson fun_poss_imp_poss subset_trans subt_at_imp_supteq' supteq_imp_funas_term_subset)

lemma funas_mgu_F:assumes fun_s:"funas_term s ⊆ F"
  and fun_t:"funas_term t ⊆ F"
  and fun_u:"funas_term u ⊆ F"
shows "funas_term (u ⋅ (the (mgu s t)::('v ⇒ ('f, 'v) Term.term))) ⊆ F" using assms wf_F_subst by auto

lemma mulr_single_correspondence:assumes LM:"(L, M) ∈ multiset_reduction_step"
  shows "∃s t. s ∈# L ∧ (s, t) ∈ rstep R ∧ t ∈# M" using assms
proof -
  from LM obtain s t where *:"s ∈# L ∧ M = ((L - {#s#}) + {#t#}) ∧ (s, t) ∈ rstep R" by auto
  hence "t ∈# M" by simp
  then show ?thesis using * by auto
qed

lemma mulr_cor:assumes LM:"(L, M) ∈ (multiset_reduction_step)⇧*"
  shows "∀ u. u ∈# L ⟶ (∃ u'. u' ∈# M ∧ (u, u') ∈ (rstep R)⇧*)" 
proof -
  from LM obtain n where LMn:"(L, M) ∈ (multiset_reduction_step)^^n" by auto
  then show ?thesis
  proof(induct n arbitrary:M)
    case 0
    then show ?case by auto
  next
    case (Suc n)
    from Suc(2) obtain U where LU:"(L, U) ∈ multiset_reduction_step ^^ n" and UM:"(U, M) ∈ multiset_reduction_step" by auto
    have "∀u. u ∈# L ⟶ (∃u'. u' ∈# U ∧ (u, u') ∈ (rstep R)⇧*)" using Suc(1) LU by auto
    then obtain u' where *:"∀u. u ∈# L ⟶ (u' u ∈# U ∧ (u, u' u) ∈ (rstep R)⇧*)" by metis
    from UM obtain v w where vw:"v ∈# U ∧ M = ((U - {#v#}) + {#w#}) ∧ (v, w) ∈ rstep R" by auto
    hence "∀u. u ∈# L ⟶ (∃u'. u' ∈# M ∧ (u, u') ∈ (rstep R)⇧*)"
    proof -
      { fix u
        assume "u ∈# L"
        hence u'u:"(u' u ∈# U ∧ (u, u' u) ∈ (rstep R)⇧*)" using * by auto
        hence "(∃u'. u' ∈# M ∧ (u, u') ∈ (rstep R)⇧*)"
        proof(cases "u' u = v")
          case True
          hence "(u, w) ∈ (rstep R)⇧*" using vw u'u by auto
          then show ?thesis using True vw diff_single_trivial by auto
        next
          case False
          hence "(u' u) ∈# M" using vw 
            by (auto, metis insert_DiffM insert_noteq_member u'u)
          then show ?thesis using False u'u by auto
        qed
      } then show ?thesis by auto
    qed
    then show ?case by auto
  qed
qed

lemma mulr_cor_rev:assumes LM:"(L, M) ∈ (multiset_reduction_step)⇧*"
  shows "∀ u. u ∈# M ⟶ (∃ w. w ∈# L ∧ (w, u) ∈ (rstep R)⇧*)"
proof -
  from LM obtain n where "(L, M) ∈ (multiset_reduction_step)^^n" by auto
  then show ?thesis
  proof(induct n arbitrary:L M)
    case 0
    then show ?case by auto
  next
    case (Suc n)
    from Suc(2) obtain U where LU:"(L, U) ∈ multiset_reduction_step" and UM:"(U, M) ∈ multiset_reduction_step ^^ n"
      by (meson relpow_Suc_E2)
    have "∀u. u ∈# M ⟶ (∃w. w ∈# U ∧ (w, u) ∈ (rstep R)⇧*)" using UM Suc(1) by auto
    then obtain w where IH:"∀u. u ∈# M ⟶ (w u ∈# U ∧ (w u, u) ∈ (rstep R)⇧*)" by metis
    from LU obtain v v' where vw:"v ∈# L ∧ U = ((L - {#v#}) + {#v'#}) ∧ (v, v') ∈ rstep R" by auto
    hence "∀ u. u ∈# M ⟶ (∃ w. w ∈# L ∧ (w, u) ∈ (rstep R)⇧*)"
    proof -
      { fix u
        assume "u ∈# M"
        hence u'u:"(w u ∈# U ∧ (w u, u) ∈ (rstep R)⇧*)" using IH by auto
        hence "(∃ w'. w' ∈# L ∧ (w', u) ∈ (rstep R)⇧*)"
        proof(cases "w u ∈# L")
          case True
          then show ?thesis using vw
            by (meson converse_rtrancl_into_rtrancl u'u) 
        next
          case False
          hence "w u = v'" using vw by (auto, metis Multiset.diff_right_commute 
                diff_single_trivial diff_union_swap multi_drop_mem_not_eq u'u)
          then show ?thesis using u'u vw by auto
        qed
      } then show ?thesis by simp
    qed
    then show ?case by auto
  qed
qed

    by (meson SN_imp_WN UNIV_I WN_on_def normalizability_E)
  from rm_correspondence uM show ?thesis 
    by (metis (no_types, opaque_lifting) u' union_mset_add_mset_right union_single_eq_member)
qed

lemma multiset_narrowing_based_not_R_unifiable: 
  assumes comp:"complete (rstep R)"
    and funas_C:"funas_trs (set C) ⊆ F"
  shows "¬ multiset_narrowing_reaches_to_success C ⟹ (¬ (∃τ. (∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*)))" 
proof -
  from comp have sn:"SN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: complete_on_def, insert comp, auto)
  assume "¬ multiset_narrowing_reaches_to_success C"
  hence ncs:"(¬ (∃σ S'. multiset_narrowing_derivation (convert_equations_into_term_multiset C) S' σ ∧ 
    (∀ s ∈# S'. s = Fun (⊤) [])))" unfolding multiset_narrowing_reaches_to_success_def by simp
  from convert_equations_into_rule_list_sound[OF funas_C]
  have wfC:"wf_equational_term_mset (convert_equations_into_term_multiset C)" by auto
  from infeasibility_using_multiset_narrowing[OF wfC ncs]
  have *:"¬ (∃θ T V. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []) ∧
      vars_term_set (set_mset (convert_equations_into_term_multiset C)) ∪ subst_domain θ ⊆ V ∧ finite V)" by auto
  show ?thesis
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "¬ funas_trs (set C) ⊆ F ∨ (∃τ. (∀u v. ((u, v) ∈ set C) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*)))"
      unfolding R_unifiable_def by auto
    with funas_C have **:"(∃τ. (∀u v. ((u, v) ∈ set C) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*)))" by auto
    have cr:"CR (multiset_reduction_step)" using sn CR_correspondence by (simp add: cr)
    from rstep_multiset_unifiability_equiv[OF funas_C ** sn cr]
    have **:"∃θ T. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" by auto
    then obtain θ T where ***:"normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" by auto
    let ?V = "vars_term_set (set_mset (convert_equations_into_term_multiset C)) ∪ subst_domain θ"
    from *** have "normal_subst R θ" by auto
    have "finite (subst_domain θ)" using finite_subst_domain by simp
    hence fV:"finite ?V" 
      using ** infeasibility_using_multiset_narrowing[OF wfC ncs] by blast
    have "(∃θ T V. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []) ∧
      vars_term_set (set_mset (convert_equations_into_term_multiset C)) ∪ subst_domain θ ⊆ V ∧ finite V)"
      by (rule exI[of _ θ], rule exI[of _ T], rule exI[of _ ?V], insert fV * ***, auto)
    then show False using * by auto
  qed
qed

lemma vars_preserve_convert_equations:"(vars_trs (set C)) = vars_term_set (set_mset (convert_equations_into_term_multiset C))"
proof(induct C)
  case Nil
  then show ?case unfolding vars_trs_def vars_term_set_def vars_rule_def by auto
next
  case (Cons pair C)
  have "vars_trs (set [pair]) = vars_term_set (set_mset (convert_equations_into_term_multiset [pair]))"
    unfolding vars_trs_def vars_term_set_def vars_rule_def by (auto split:prod.splits) 
  then show ?case unfolding vars_trs_def vars_term_set_def vars_rule_def using Cons by auto
qed

(* Completeness of E-unification w.r.t. multiset narrowing. If a (multi)set of equations has an R-unifier,
  then multiset narrowing can find a more general solution w.r.t. R *)

theorem multiset_narrowing_based_completeness_of_E_unification:
  assumes comp:"complete (rstep R)"
    and funas_C:"funas_trs (set C) ⊆ F"
    and R_unif:"(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*)"
  shows mnd:"∃σ θ S'. multiset_narrowing_derivation (convert_equations_into_term_multiset C) S' σ ∧ 
    (∀ s ∈# S'. s = Fun (⊤) []) ∧ (σ ∘⇩s θ) |s (vars_trs (set C)) =⇩R τ |s (vars_trs (set C))"
proof -
  from comp have sn:"SN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: complete_on_def, insert comp, auto)
  have sn_mul:"SN (multiset_reduction_step)" using sn 
    by (simp add: SN_R_imp_SN_multiset_reduction_step)
  have cr_mul:"CR (multiset_reduction_step)" using cr sn_mul
    by (simp add: CR_correspondence sn)
  from sn have normτ:"normalizable_subst R τ"
    using SN_obtains_normalizable_subst by auto
  from obtains_normalized_subst[OF normτ]
  obtain τ' where normτ':"normal_subst R τ'" and ττ':"(∀x. (τ x, τ' x) ∈ (rstep R)⇧*)" by auto
  with R_unif have *:"(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧↔⇧*)"
  proof -
    { fix u :: "('f, 'v) Term.term" and v :: "('f, 'v) Term.term"
      have "∀t. (t ⋅ τ, t ⋅ τ') ∈ (rstep R)⇧*"
        by (simp add: ττ' term_subst_rsteps)
      hence "(u, v) ∈ set C ⟶ (u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧↔⇧*"
        by (smt (z3) R_unif conversionI' conversion_inv conversion_rtrancl rtrancl.rtrancl_into_rtrancl) 
    } then show ?thesis by auto
  qed
  let ?C = "convert_equations_into_term_multiset C"
  have wf_C:"wf_equational_term_mset ?C"
    by (simp add: convert_equations_into_rule_list_sound funas_C)
  let ?Cτ'= "subst_term_multiset τ' ?C"
  have Cτ':"?Cτ'= subst_term_multiset τ' ?C" ..
  from R_unif have "(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧↔⇧*)" using * by auto
  hence "(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧↓)"
    by (simp add: CR_imp_conversionIff_join cr)
  from rstep_multiset_reduction_equiv_normal[of C τ']
  have reltran:"∃U. (?Cτ', U) ∈ multiset_reduction_step⇧* ∧ (∀ s ∈# U. s = Fun (⊤) [])"
    using cr sn normτ' * funas_C by auto
  then obtain U where mstep:"(?Cτ', U) ∈ multiset_reduction_step⇧*" and U:"(∀ s ∈# U. s = Fun (⊤) [])" by auto
  let ?V = "vars_term_set (set_mset ?C) ∪ subst_domain τ'"
  have "vars_term_set (set_mset ?C) ∪ subst_domain τ' ⊆ ?V" ..
  have "finite (vars_term_set (set_mset ?C))" unfolding vars_term_set_def by simp
  moreover have "finite (subst_domain τ')" using finite_subst_domain by simp
  ultimately have fV:"finite ?V" by simp
  from lifting_lemma_for_multiset_narrowing_for_equational_terms [OF normτ' wf_C Cτ', of ?V U ]
  have "∃σ θ S'. multiset_narrowing_derivation ?C S' σ ∧ subst_term_multiset θ S' = U ∧ 
    normal_subst R θ ∧ wf_equational_term_mset S' ∧ (σ ∘⇩s θ) |s ?V = τ' |s ?V" using mstep fV by auto
  then obtain σ θ S' where mnd:"multiset_narrowing_derivation ?C S' σ" and sub:"subst_term_multiset θ S' = U"
    and "normal_subst R θ" and wf:"wf_equational_term_mset S'" and subσθ:"(σ ∘⇩s θ) |s ?V = τ' |s ?V" by auto
  from wf[unfolded wf_equational_term_mset_def]
  have wfS':"∀ s ∈# S'. wf_equational_term s" by auto
  have no_subst_true:"∀u v. (Fun ≐ [u, v]) ⋅ θ ≠ Fun (⊤) []" by auto
  have wf_eq:"wf_equational_term (Fun ⊤ [])" unfolding wf_equational_term_def by auto
  have vacuous_subst:"∀θ. Fun ⊤ [] = (Fun ⊤ []) ⋅ θ" by auto
  from sub have "{# (t ⋅ θ). t ∈# S' #} = U" using sub unfolding subst_term_multiset_def by auto
  hence *:"∀s ∈# S'. s ⋅ θ = Fun (⊤) []" using U by auto
  have sS':"(∀ s ∈# S'. s = Fun (⊤) [])"
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "∃s. s ∈# S' ∧ (s ≠ Fun (⊤) [])" by auto
    then obtain s where s:"s ∈# S'" and ns:"(s ≠ Fun (⊤) [])" by auto
    from ns wfS' have "∃u v. s = Fun ≐ [u, v] ∧ 
    (≐, 2) ∉ funas_term u ∧ (≐, 2) ∉ funas_term v ∧ (⊤, 0) ∉ funas_term u ∧ (⊤, 0) ∉ funas_term v" 
      using s wf_equational_term_def by auto
    then obtain u v where "s = Fun ≐ [u, v]" by auto
    then show False using * s by auto
  qed
  have vars:"(vars_trs (set C)) = vars_term_set (set_mset (convert_equations_into_term_multiset C))" 
    using vars_preserve_convert_equations by auto
  hence vars:"(vars_trs (set C)) ⊆ ?V"  by auto
  have *:"(σ ∘⇩s θ) |s (vars_trs (set C)) = τ' |s (vars_trs (set C))" using subσθ 
    by (metis vars inf.absorb_iff2 subst_restrict_Int)
  have **:"(∀x. (τ x, τ' x) ∈ (rstep R)⇧↔⇧*)" using ττ' by auto
  hence "τ' |s vars_trs (set C) =⇩R τ |s vars_trs (set C)" unfolding subst_modulo_def subst_restrict_def 
    by (simp add: conversion_inv)
  hence "(σ ∘⇩s θ) |s (vars_trs (set C)) =⇩R τ |s (vars_trs (set C))" using * by auto 
  then show ?thesis using sS' mnd by auto
qed

(* Completeness of E-unifiability w.r.t. multiset narrowing, where R is viewed as a set of equations *)

theorem multiset_narrowing_based_R_unifiability: 
  assumes comp:"complete (rstep R)"
    and funas_C:"funas_trs (set C) ⊆ F"
  shows "multiset_narrowing_reaches_to_success C ⟹ R_unifiable C" 
        "¬ multiset_narrowing_reaches_to_success C ⟹ ¬ R_unifiable C" 
proof -
  from comp have sn:"SN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: complete_on_def, insert comp, auto)
  from sn cr show "multiset_narrowing_reaches_to_success C ⟹ R_unifiable C"
    by (metis R_unifiable_def ex_in_conv funas_C multiset_narrowing.multiset_narrowing_reaches_to_success_def multiset_narrowing_axioms multiset_narrowing_based_R_unifiable)
  show "¬ multiset_narrowing_reaches_to_success C ⟹ ¬ R_unifiable C" 
    by (simp add: comp funas_C multiset_narrowing_based_not_R_unifiable R_unifiable_def)
qed
