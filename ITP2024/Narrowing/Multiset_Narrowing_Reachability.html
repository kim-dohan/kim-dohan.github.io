section ‹Formalization of multiset narrowing based (multiset) reachability analysis›

theory Multiset_Narrowing_Reachability
  imports
    Multiset_Narrowing
begin

locale multiset_narrowing_reachability =  multiset_narrowing R DOTEQ TOP R' F D x
  for R::"('f, 'v:: infinite) trs" 
    and DOTEQ :: 'f ("≐")
    and TOP :: "'f" ("⊤") 
    and R' :: "('f, 'v:: infinite) trs" 
    and F :: "'f sig" 
    and D :: "'f sig"
    and x :: 'v
begin

(* We consider multiset narrowing on multisets of ordinary terms, multisets of equational terms, and multisets of pairs of terms.
  First, consider multiset narrowing on multisets of pairs of terms for (ordinary) reachability analysis.*)

definition subst_pairs_multiset :: "('f, 'v) subst ⇒ (('f, 'v) term × ('f, 'v) term) multiset ⇒ (('f, 'v) term × ('f, 'v) term) multiset" where
  "subst_pairs_multiset σ M = {# (s ⋅ σ, t ⋅ σ). (s, t) ∈# M #}"

inductive_set multiset_pair_reduction_step where 
    "(s, t) ∈# S ∧ T = (S - {#(s,t)#} + {#(u, t)#}) ∧ (s, u) ∈ rstep R ⟹ (S, T) ∈ multiset_pair_reduction_step"

inductive_set multiset_pair_narrowing_step where 
    "(s, t) ∈# S ∧ T = (subst_pairs_multiset σ (S - {#(s,t)#}) + {#(u, t ⋅ σ)#}) ∧ (s, u, σ) ∈ narrowing_step  
        ⟹ (S, T, σ) ∈ multiset_pair_narrowing_step"

lemmas multiset_pair_reduction_stepI = multiset_pair_reduction_step.intros [intro]
lemmas multiset_pair_reduction_stepE = multiset_pair_reduction_step.cases [elim]
lemmas multiset_pair_narrowing_stepI = multiset_pair_narrowing_step.intros [intro]
lemmas multiset_pair_narrowing_stepE = multiset_pair_narrowing_step.cases [elim]

definition multiset_pair_narrowing_derivation where
  "multiset_pair_narrowing_derivation G G' σ ⟷ (∃n. (∃f τ. f 0 = G ∧ f n = G' ∧ 
  (∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step) ∧ (if n = 0 then σ = Var else σ = compose (map (λi. (τ i)) [0 ..< n]))))"

definition multiset_pair_narrowing_derivation_num where
  "multiset_pair_narrowing_derivation_num G G' σ n ⟷ (∃f τ. f 0 = G ∧ f n = G' ∧ 
  (∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step) ∧ (if n = 0 then σ = Var else σ = compose (map (λi. (τ i)) [0 ..< n])))"

definition syntactically_unifiable where
  "syntactically_unifiable σ S ⟷ (∀ (s, t) ∈# S. s ⋅ σ = t ⋅ σ)"

definition trivially_unifiable where
  "trivially_unifiable S ⟷ (∀ (s, t) ∈# S. s = t)"

definition R_solution_of_goal where
  "R_solution_of_goal σ S ⟷ (∀ (s, t) ∈# S. (s ⋅ σ, t ⋅ σ) ∈ (rstep R)⇧*)"

lemma n0_multiset_pair_narrowing_derivation_num:"multiset_pair_narrowing_derivation_num G G' σ 0 ⟹ G = G' ∧ σ = Var"
  unfolding multiset_pair_narrowing_derivation_num_def by auto

lemma multiset_reachability_narrowing_deriv_implication: assumes "multiset_pair_narrowing_derivation_num G G' σ n"
  shows "multiset_pair_narrowing_derivation G G' σ" 
  unfolding multiset_pair_narrowing_derivation_num_def multiset_pair_narrowing_derivation_def 
  using assms multiset_pair_narrowing_derivation_num_def by metis

lemma multiset_reachability_reduction_trivial_solution: assumes gr_steps:"(G, G') ∈ multiset_pair_reduction_step⇧*"
    and un:"trivially_unifiable G'"
  shows "(∀ (s, t) ∈# G. (s, t) ∈ (rstep R)⇧*)"
proof -
  note gdefs = subst_pairs_multiset_def trivially_unifiable_def R_solution_of_goal_def 
  from gr_steps obtain n where "(G, G') ∈ (multiset_pair_reduction_step)^^n" by auto
  then show ?thesis using un
  proof(induct n arbitrary: G G' rule: wf_induct[OF wf_measure [of "λ n. n"]])
    case (1 n)
    note IH1 = 1(1)[rule_format]
    then show ?case
    proof(cases "n = 0")
      case True
      with step have "G = G'" using 1 by auto
      then show ?thesis unfolding gdefs using 1 trivially_unifiable_def
        by (smt (verit, ccfv_SIG) case_prodD case_prodI2 rtrancl.simps)
    next
      case False
      with 1(2) obtain U where step:"(G, U) ∈ (multiset_pair_reduction_step)" and
        UG':"(U, G') ∈ (multiset_pair_reduction_step)^^ (n - 1)" by (metis False One_nat_def diff_Suc_1 relpow_E2)
      with 1(2) obtain s t u where stG:"(s, t) ∈# G" and U:"U = (G - {#(s,t)#} + {#(u, t)#})" and 
        su:"(s, u) ∈ rstep R" by auto
      have IH:"(∀ (s, t) ∈# U. (s, t) ∈ (rstep R)⇧*)" 
      proof
        fix v w
        assume asm:"(v, w) ∈# U"
        then show "(v, w) ∈ (rstep R)⇧*" using IH1[of "n-1" U G' "(v, w)"] 1(3) False UG' by fastforce
      qed
      hence *:"(∀ (s, t) ∈# (G - {#(s,t)#}). (s, t) ∈ (rstep R)⇧*)" using U by auto
      have "(u, t) ∈ (rstep R)⇧*" using U IH by auto
      hence "(s, t) ∈ (rstep R)⇧*" using su by auto
      then show ?thesis using *
      by (metis stG add_mset_remove_trivial_eq case_prodI insert_noteq_member)
    qed
  qed
qed

lemma multiset_reachability_reduction_solution: assumes gr_steps:"(subst_pairs_multiset σ G, G') ∈ multiset_pair_reduction_step⇧*"
    and un:"trivially_unifiable G'"
  shows "R_solution_of_goal σ G" 
proof (rule ccontr)
  note gdefs = subst_pairs_multiset_def trivially_unifiable_def R_solution_of_goal_def 
  assume "¬ ?thesis"
  hence "(∃ (s, t) ∈# G. (s ⋅ σ, t ⋅ σ) ∉ (rstep R)⇧*)" unfolding gdefs by auto
  then obtain s t where "(s, t) ∈# G" and "(s ⋅ σ, t ⋅ σ) ∉ (rstep R)⇧*" by auto
  then show False using multiset_reachability_reduction_trivial_solution[OF gr_steps un] unfolding gdefs by auto 
qed

lemma subst_closed_multiset_reachability_reduction: assumes "(S, T) ∈ (multiset_pair_reduction_step)⇧*"
shows "(subst_pairs_multiset σ S, subst_pairs_multiset σ T) ∈ (multiset_pair_reduction_step)⇧*"
proof -
  from assms obtain n where "(S, T) ∈ (multiset_pair_reduction_step)^^n" by auto
  then show ?thesis
  proof(induct n arbitrary:T σ)
    case 0
    then show ?case by simp
  next
    case (Suc n)
    from Suc(2) obtain U where SU:"(S, U) ∈ multiset_pair_reduction_step ^^ n" and UT:"(U, T) ∈ multiset_pair_reduction_step" by auto
    from Suc(1) have gstep:"(subst_pairs_multiset σ S, subst_pairs_multiset σ U) ∈ multiset_pair_reduction_step⇧*" by (simp add: SU)
    from UT obtain s t u where st:"(s, t) ∈# U" and T:"T = (U - {#(s,t)#} + {#(u, t)#})" and su:"(s, u) ∈ rstep R" by auto
    let ?s = "s ⋅ σ"
    let ?t = "t ⋅ σ"
    let ?u = "u ⋅ σ"
    let ?U = "subst_pairs_multiset σ U"
    let ?T = "subst_pairs_multiset σ T"
    have *:"(?s, ?t) ∈# ?U" using st unfolding subst_pairs_multiset_def by auto
    have **:"?T = ?U - {#(?s, ?t)#} + {#(?u, ?t)#}" unfolding subst_pairs_multiset_def using T 
      by (auto, metis (no_types, lifting) add_mset_remove_trivial case_prod_conv image_mset_add_mset insert_DiffM st)
    have "(?s, ?u) ∈ rstep R" using su by auto
    with UT have "(subst_pairs_multiset σ U, subst_pairs_multiset σ T) ∈ multiset_pair_reduction_step" 
      using multiset_pair_reduction_step.intros[of ?s ?t ?U ?T ?u] * ** by auto
    then show ?case using gstep by force
  qed
qed

lemma subst_multiset_equiv: "{# (s ⋅ θ, t ⋅ θ). (s, t) ∈# {# (s ⋅ σ, t ⋅ σ). (s, t) ∈# S #} #} = {# (s ⋅ σ ⋅ θ, t ⋅ σ ⋅ θ). (s, t) ∈# S #}" 
proof -
  have "{# (s ⋅ θ, t ⋅ θ). (s, t) ∈# {# (s ⋅ σ, t ⋅ σ). (s, t) ∈# S #} #} = {# (s , t ). (s, t) ∈# {# (s ⋅ σ ⋅ θ, t ⋅ σ ⋅ θ). (s, t) ∈# S #} #}"
    by (auto, smt (verit, ccfv_SIG) comp_apply multiset.map_comp multiset.map_cong0 prod.sel(1) prod.sel(2) split_beta)
  then show ?thesis by auto
qed

lemma subst_closed_compose: "subst_pairs_multiset θ (subst_pairs_multiset σ S) = subst_pairs_multiset (σ ∘⇩s θ) S"
  by (simp add: subst_multiset_equiv subst_pairs_multiset_def)

lemma multiset_reachability_narrowing_reduction: assumes gnd:"multiset_pair_narrowing_derivation G G' σ"
  shows "(subst_pairs_multiset σ G, G') ∈ (multiset_pair_reduction_step)⇧*"
proof -
  note gdefs = subst_pairs_multiset_def trivially_unifiable_def R_solution_of_goal_def 
  from gnd obtain n where "multiset_pair_narrowing_derivation_num G G' σ n" unfolding multiset_pair_narrowing_derivation_num_def
    multiset_pair_narrowing_derivation_def by auto
  then show ?thesis
  proof(induct n arbitrary: G' σ)
    case 0
    hence "G' = G" and "σ = Var" using n0_multiset_pair_narrowing_derivation_num by blast+
    then show ?case unfolding gdefs by auto
  next
    case (Suc n)
    from ‹multiset_pair_narrowing_derivation_num G G' σ (Suc n)›
    have "(∃f τ. f 0 = G ∧ f (Suc n) = G' ∧ 
      (∀i < (Suc n). ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step) ∧ (σ = compose (map (λi. (τ i)) [0 ..< Suc n])))"
      unfolding multiset_pair_narrowing_derivation_num_def by auto
    then obtain f τ where f0:"f 0 = G" and fsucn:"f (Suc n) = G'" and 
      relchain:"∀i < (Suc n). ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step" 
      and σ:"(σ = compose (map (λi. (τ i)) [0 ..< (Suc n)]))" by auto
    let ?τ = "compose (map (λi. (τ i)) [0 ..< n])"
    let ?σ = "if n = 0 then Var else ?τ"
    let ?f = "λi. (if i ≤ n then f i else undefined)"
    from relchain obtain U where U:"f n = U" by simp
    from relchain have nchain:"∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step" by simp
    have "(∃f τ. f 0 = G ∧ f n = U ∧ (∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ multiset_pair_narrowing_step))"
      by (rule exI[of _ "?f"] rule exI[of _ "?σ"], insert U f0 nchain, auto)
    moreover have "multiset_pair_narrowing_derivation_num G U ?σ n" unfolding multiset_pair_narrowing_derivation_num_def
      using U calculation f0 nchain by auto 
    ultimately have IH:"(subst_pairs_multiset ?σ G, U) ∈ multiset_pair_reduction_step⇧*" using Suc(1) by auto
    have "(U, G', τ n) ∈ multiset_pair_narrowing_step" using relchain fsucn U by auto
    then obtain s t u  where st:"(s, t) ∈# U" and G':"G' = subst_pairs_multiset (τ n) (U - {#(s,t)#}) + {#(u, t ⋅ (τ n))#}"
     and su:"(s, u, (τ n)) ∈ narrowing_step" by auto
    let ?s = "s ⋅ (τ n)"
    let ?t = "t ⋅ (τ n)"
    let ?U = "subst_pairs_multiset (τ n) U"
    have *:"(?s, ?t) ∈# ?U" using st subst_pairs_multiset_def by (smt (verit) add_mset_remove_trivial_eq 
      case_prod_conv image_mset_add_mset union_single_eq_member)
    have "G' = {#(u, ?t)#} + (?U - {#(?s, ?t)#})" using G' unfolding gdefs 
      by (simp add: image_mset_Diff st)
    hence G':"G' = ?U - {#(?s, ?t)#} + {#(u, ?t)#}" by auto
    have "(s ⋅ (τ n),  u) ∈ rstep R" using su
    proof -
      from su obtain rl ω p where t':"u = (replace_at s p (snd rl)) ⋅ (τ n)" and ωrl:"ω ∙ rl ∈ R"
        and disj:"(vars_term s ∩ vars_rule rl = {})" and p:"p ∈ fun_poss s" and mgu:"mgu (s |_ p) (fst rl) = Some (τ n)" by auto
      from t' have t:"u = replace_at (s ⋅ (τ n)) p (snd rl ⋅ (τ n))" using p 
        by (simp add: ctxt_of_pos_term_subst fun_poss_imp_poss)
      let ?C = "ctxt_of_pos_term p (s ⋅ (τ n))"
      from mgu have "(s |_ p) ⋅ (τ n) = fst rl ⋅ (τ n)" using subst_apply_term_eq_subst_apply_term_if_mgu by auto
      then show ?thesis using rstep.intros[of "fst rl" "snd rl" R "s ⋅ (τ n)" ?C "(τ n)" u] 
        by (smt (verit) ωrl fun_poss_imp_poss p perm_rstep_conv poss_imp_subst_poss prod.exhaust_sel 
            replace_at_ident rstep_ctxt rstep_subst subset_iff subset_rstep subt_at_subst t)
    qed
    hence **:"(?U, G') ∈ multiset_pair_reduction_step" using multiset_pair_reduction_step.intros * G' by auto
    have σ:"σ = ?σ ∘⇩s (τ n)" using σ by auto
    let ?G = "subst_pairs_multiset ?σ G"
    from subst_closed_compose[of "(τ n)" ?σ G]
    have "subst_pairs_multiset (τ n) ?G = subst_pairs_multiset (?σ ∘⇩s (τ n)) G" by auto
    from subst_closed_multiset_reachability_reduction[of "subst_pairs_multiset ?σ G" U "τ n"]
    have "(subst_pairs_multiset σ G, subst_pairs_multiset (τ n) U) ∈ multiset_pair_reduction_step⇧*" 
      using IH σ subst_closed_compose by metis
    then show ?case using IH ** unfolding gdefs by auto 
  qed
qed

(* A sufficient condition for reachability for multiple goals,  MT2005 *)

proposition reachable_condition_multiple_goals:assumes gnd:"multiset_pair_narrowing_derivation G G' σ"
    and un:"syntactically_unifiable η G'"
  shows "R_solution_of_goal (σ ∘⇩s η) G"
proof -
  note gdefs = subst_pairs_multiset_def syntactically_unifiable_def trivially_unifiable_def R_solution_of_goal_def
  from multiset_reachability_narrowing_reduction[of G G' σ]
  have srstep:"(subst_pairs_multiset σ G, G') ∈ multiset_pair_reduction_step⇧*" using gnd by auto
  from subst_closed_multiset_reachability_reduction[of "subst_pairs_multiset σ G" G' η]
  have subcom:"(subst_pairs_multiset η (subst_pairs_multiset σ G), subst_pairs_multiset η G') ∈ multiset_pair_reduction_step⇧*" using srstep by auto
  from subst_closed_compose[of η σ G]
  have *:"(subst_pairs_multiset (σ ∘⇩s η) G, subst_pairs_multiset η G') ∈ multiset_pair_reduction_step⇧*" using subcom by auto
  have "trivially_unifiable (subst_pairs_multiset η G')" unfolding gdefs using un[unfolded syntactically_unifiable_def] by auto 
  with multiset_reachability_reduction_solution show ?thesis using * by simp
qed

(* Next, consider multiset narrowing on multisets of ordinary terms and multisets of equational terms *)

definition "multiset_narrowing_reachable_from_to S G ⟷  (∃σ η S'. multiset_narrowing_derivation S S' σ ∧ 
    subst_term_multiset η S' = G)"

lemma multiset_narrowing_based_reachable_sufficient_condition_pre:
  assumes nd:"multiset_narrowing_derivation S S' σ"
    and "subst_term_multiset η S' = G"
  shows "(subst_term_multiset (σ ∘⇩s η) S, G) ∈ (multiset_reduction_step)⇧*"
  using assms by (metis multiset_narrowing_reduction subst_closed_multiset_reduction subst_term_multiset_compose)

proposition multiset_narrowing_based_reachable_sufficient_condition:
  assumes "multiset_narrowing_reachable_from_to S G"
  shows "∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*"
  using assms by (metis multiset_narrowing_reduction multiset_narrowing_reachable_from_to_def subst_closed_multiset_reduction subst_term_multiset_compose)
                                                               
proposition multiset_narrowing_based_infeasibility_condition:
  assumes "¬ multiset_narrowing_reachable_from_to S G"
  shows "¬ (∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
proof (rule ccontr)
  assume "¬ ?thesis"
  hence "(∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)" by auto
  then obtain θ where normθ:"normal_subst R θ" and θSG:"(subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*" by auto
  from θSG have *:"(subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*" by blast
  from assms have "(∀σ η S'. multiset_narrowing_derivation S S' σ ⟶ 
    subst_term_multiset η S' ≠ G)" unfolding multiset_narrowing_reachable_from_to_def by auto
  let ?V = "vars_term_set (set_mset S) ∪ subst_domain θ"
  have fnV:"finite ?V" unfolding vars_term_set_def
    by (simp add: finite_subst_domain)
  from lifting_lemma_for_multiset_narrowing
  have "∃τ τ' S'. multiset_narrowing_derivation S S' τ ∧ G = subst_term_multiset τ' S' ∧
     normal_subst R τ' ∧ τ ∘⇩s τ' |s ?V = θ |s ?V" 
    using "*" fnV using normθ by auto
  then show False
    using ‹∀σ η S'. multiset_narrowing_derivation S S' σ ⟶ subst_term_multiset η S' ≠ G› by blast
qed

(* The following lemma is the weak completeness of multiset narrowing
  w.r.t. the multiset reachability problem *)

lemma multiset_narrowing_based_reachability_weak_completeness:
 "multiset_narrowing_reachable_from_to S G ⟶ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
 "¬ multiset_narrowing_reachable_from_to S G ⟶ ¬ (∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
proof -
  show "multiset_narrowing_reachable_from_to S G ⟶ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    by (simp add: multiset_narrowing_based_reachable_sufficient_condition)
  show "¬ multiset_narrowing_reachable_from_to S G ⟶ ¬ (∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    by (simp add: multiset_narrowing_based_infeasibility_condition)
qed

lemma multiset_reduction_single_reachability: assumes wn: "WN (rstep R)"
    and cr: "CR (multiset_reduction_step)"
    and funas_uv:"funas_rule (u, v) ⊆ F"
    and nf:"v ∈ NF (rstep R)"
    and ersteps:"({#Fun ≐ [u, v]#}, {#Fun (⊤) []#}) ∈ (multiset_reduction_step)⇧+"
  shows "((u, v) ∈ (rstep R)⇧*)"
proof -
  from ersteps obtain n where step:"({#Fun ≐ [u, v]#}, {#Fun (⊤) []#}) ∈ (multiset_reduction_step)^^n" and n1:"n ≥ 1" 
    by (metis add_mset_eq_single less_one list.simps(3) relpow_0_E rtrancl_imp_relpow term.inject(2) trancl_into_rtrancl verit_comp_simplify1(3))
  then show ?thesis using funas_uv nf
  proof(induct n arbitrary:u v)
    case 0
    then show ?case by auto
  next
    case (Suc n)
    with step obtain U where uvU:"({#Fun ≐ [u, v]#}, U) ∈ (multiset_reduction_step)" and 
      UG:"(U, {#Fun (⊤) []#}) ∈ (multiset_reduction_step)^^n" by (meson relpow_Suc_D2)
    from uvU obtain s p rl σ where s:"s = Fun ≐ [u, v]" and U:"U = {#replace_at s p ((snd rl) ⋅ σ)#}" and rl:"rl ∈ R" 
      and pfun:"p ∈ poss s" and sp:"s|_ p = (fst rl) ⋅ σ" 
      by (smt (verit) add_mset_eq_single insert_DiffM mul_reduction_correspondence multiset_reduction_step_posE union_is_single)
    have wf_s:"wf_equational_term s" unfolding wf_equational_term_def funas_defs using funas_uv 
      by (metis Suc.prems(3) convert_equation_into_term_sound fst_conv s snd_conv wf_equational_term_def) 
    have nu:"(≐, 2) ∉ funas_term u" and nv:"(≐, 2) ∉ funas_term v" using Suc(4) D D_fresh unfolding funas_defs by auto 
    have poss_us:"∀q ∈ poss u. u |_ q = s |_ (0 # q)" using s by auto  
    have poss_vs:"∀q' ∈ poss v. v |_ q' = s |_ (1 # q')" using s by auto
    let ?srule = "λx. (Fun (≐) [Var x, Var x], Fun (⊤) [])"
    have fr:"∀x. funas_rule (?srule x) = {(≐, 2), (⊤, 0)}" using D 
      unfolding funas_defs by (auto simp add: numeral_2_eq_2)
    have cd:"∀x. funas_rule (?srule x) ⊆ D" using D 
      unfolding funas_defs by (auto simp add: numeral_2_eq_2)
    have rlR':"rl ∈ R' ∨ (∃x. rl = ?srule x)" using R' rl by fastforce
    moreover have rlR'_un:"funas_rule rl ⊆ F ∨ (∃x. rl = ?srule x)" 
      using R_sig by (metis funas_defs(2) le_supI lhs_wf prod.exhaust_sel rhs_wf rlR')
    ultimately have funas_rl:"funas_rule rl ⊆ F ∨ funas_rule rl = {(≐, 2), (⊤, 0)}"
      using fr by metis
    then consider (ordinary) "funas_rule rl ⊆ F" | (special) "funas_rule rl = {(≐, 2), (⊤, 0)}" by auto
    then show ?case
    proof(cases)
      case (special)
      have "funas_rule rl = {(≐, 2), (⊤, 0)}"
      proof(rule ccontr)
        assume asm:"¬ ?thesis"
        have "funas_rule rl = D" using special
          by (simp add: D)
        then show False using asm D by simp
      qed
      hence "(∃x. rl = ?srule x)" using s sp rl funas_rl R' R_sig D D_fresh unfolding funas_defs 
        using rlR'_un special by force
      then obtain x where rlx:"rl = ?srule x" by auto
      with pfun have root_s:"root (s|_ p) = Some (≐, 2)" using sp funas_uv unfolding funas_defs by auto
      have ptop:"p = []"
      proof (rule ccontr)
        assume "¬ ?thesis"
        hence "u ⊵ (s |_ p) ∨ v ⊵ (s |_ p)" using pfun s by auto
            (metis diff_Suc_1 less_Suc0 less_Suc_eq nth_Cons' subt_at_imp_supteq)
        then show False using root_s nu nv unfolding funas_defs
          by (meson root_symbol_in_funas subset_eq supteq_imp_funas_term_subset)
      qed
      hence *:"replace_at s p ((snd rl) ⋅ σ) = Fun (⊤) []" using s rlx by auto
      hence "u = v" using s rlx sp by (simp add: ptop)
      then show ?thesis by auto
    next
      case ordinary
      hence rlR':"rl ∈ R'" using R' D R_sig D_fresh 
        by (metis fr inf.orderE insert_not_empty rlR')
      from fun_root_not_None have nroot:"root (fst rl) ≠ None" 
        by (metis is_Fun_Fun_conv local.wf prod.collapse rl wf_trs_imp_lhs_Fun)
      from root_not_special_symbols[OF rlR' ordinary nroot]
      have *:"root (fst rl) ≠ Some (≐, 2) ∧ root (fst rl) ≠  Some (⊤, 0)" by simp
      hence root_eq:"root (s |_ p) = root(fst rl)" using s
        by (metis (no_types, opaque_lifting) R' empty_pos_in_poss le_sup_iff local.wf poss_is_Fun_fun_poss 
            prod.exhaust_sel rlR' root_subst_inv sp subset_iff subt_at.simps(1) term.disc(2) wf_trs_def)
      hence "root (s |_ p) ≠ Some (≐, 2) ∧ root (s |_ p) ≠ Some (⊤, 0)" using * by force
      hence np:"p ≠ []" using wf_s[unfolded wf_equational_term_def] by force
      hence "∃q r. (q ∈ poss u ∧ p = 0 # q) ∨ (r ∈ poss v ∧ p = 1 # r)" using s ordinary pfun 
      proof(auto, goal_cases)
        case (1 x p)
        then show ?case using fun_poss_imp_poss less_2_cases numeral_2_eq_2 by fastforce
      next
        case (2 x p)
        then show ?case by (metis less_2_cases nth_Cons_0 nth_Cons_Suc numeral_2_eq_2)
      next
        case (3 x p)
        then show ?case by (metis less_2_cases nth_Cons_0 numeral_2_eq_2)
      qed
      then obtain q r where qpos:"(q ∈ poss u ∧ p = 0 # q) ∨ (r ∈ poss v ∧ p = 1 # r)" (is "?A ∨ ?B") by auto
      hence qp:"(q ∈ poss u ∧ p = 0 # q)" using Suc(5) by (metis NF_instance NF_rstep_subterm One_nat_def 
            lhs_notin_NF_rstep nth_Cons_0 nth_Cons_Suc prod.collapse rl s sp subt_at.simps(2) subt_at_imp_supteq')
      hence *:"replace_at s p ((snd rl) ⋅ σ) = Fun ≐ [(replace_at u q ((snd rl) ⋅ σ)), v]"  using s by auto
      hence "U = {#Fun ≐ [(replace_at u q ((snd rl) ⋅ σ)), v]#}" using U by auto
      moreover have "funas_rule (replace_at u q ((snd rl) ⋅ σ), v) ⊆ F" using funas_uv wf_F_subst ordinary unfolding funas_defs
        by auto (metis Suc.prems(3) qp ctxt_supt_id fst_conv funas_rule_def funas_term_ctxt_apply le_sup_iff subsetD,
              metis Suc.prems(3) UnI2 funas_rule_def in_mono snd_eqD)
      moreover have "(replace_at u q ((snd rl) ⋅ σ), v) ∈ (rstep R)⇧*" using Suc(1) UG 
        by (metis One_nat_def Suc.prems(2) Suc.prems(4) calculation(1) calculation(2) le_SucE list.simps(3) 
            nat.inject relpow_0_E single_eq_add_mset term.inject(2))
      ultimately show ?thesis using uvU
        by (metis converse_rtrancl_into_rtrancl ctxt_supt_id poss_us prod.collapse qp rl rstepI sp)
    qed
  qed
qed

lemma infeasibility_using_multiset_narrowing: assumes wf_S:"wf_equational_term_mset S"
  and "¬ (∃σ S'. multiset_narrowing_derivation S S' σ ∧ (∀ s ∈# S'. s = Fun (⊤) []))"
shows "¬ (∃θ T. normal_subst R θ ∧ (subst_term_multiset θ S, T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))"
proof(rule ccontr)
  assume asm:"¬?thesis"
  hence "∃θ T. normal_subst R θ ∧ (subst_term_multiset θ S, T) ∈ multiset_reduction_step⇧* ∧ (∀ t ∈#  T. t = Fun (⊤) [])" by auto
  then obtain θ T  where normθ:"normal_subst R θ" and relsteps:"(subst_term_multiset θ S, T) ∈ multiset_reduction_step⇧*" 
    and eqsat:"∀ t ∈# T. t = Fun (⊤) []"  by auto
  let ?V = "vars_term_set (set_mset S) ∪ subst_domain θ"
  have "finite (vars_term_set (set_mset S))" unfolding vars_term_set_def by simp
  moreover have "finite (subst_domain θ)" using finite_subst_domain by simp
  ultimately have fV:"finite ?V" by simp
  have "wf_equational_term_mset T" unfolding wf_equational_term_mset_def[unfolded wf_equational_term_def] 
    using eqsat by auto
  define U where U:"U = subst_term_multiset θ S" 
  have wf_U:"wf_equational_term_mset U" using wf_eq_mset_subst_inv by (simp add: U wf_S)
  have reltran:"(U, T) ∈ multiset_reduction_step⇧*" using relsteps by (simp add: U)
  from lifting_lemma_for_multiset_narrowing_for_equational_terms [OF normθ wf_S U, of ?V T ]
  have "∃σ θ' S'. multiset_narrowing_derivation S S' σ ∧ subst_term_multiset θ' S' = T ∧ 
    normal_subst R θ' ∧ wf_equational_term_mset S'" using asm fV reltran by blast
  then obtain θ' S' where cond:"∃σ. multiset_narrowing_derivation S S' σ" and sub:"subst_term_multiset θ' S' = T" and "normal_subst R θ'"
    and T:"∀ t ∈# T. t = Fun (⊤) []" and wfS':"wf_equational_term_mset S'" using eqsat by blast
  from T have *:"∀t ∈# T. wf_equational_term t" unfolding wf_equational_term_def by auto
  from sub[unfolded subst_term_multiset_def]
  have *:"{#t ⋅ θ'. t ∈# S'#} = T" by auto
  have "(∀u v θ. Fun ≐ [u, v] ⋅ θ ≠ Fun (⊤) [])" using wf_eq_subst by simp
  from wfS'[unfolded wf_equational_term_mset_def[unfolded wf_equational_term_def]]
  have "∀t∈#S'. t = Fun ⊤ []" using * using wf_eq_subst using eqsat by fastforce
  then show False using assms cond by auto
qed

lemma rstep_single_eq_reduction_step_eq: assumes uv:"(u::('f, 'v)term, v::('f, 'v)term) ∈ (rstep R')⇧*"
  shows "(∃T. ({#Fun ≐ [u, v]#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))"
proof -
  from uv show ?thesis
  proof(induct rule: converse_rtrancl_induct)
    case base
    then show ?case using multiset_red_refl 
      by (meson add_mset_eq_single insert_DiffM r_into_rtrancl)
  next
    case (step P Q)
    from step(3) obtain T' where T:"({#Fun ≐ [Q, v]#}, T') ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T'. t = Fun ⊤ [])" by auto
    from step(1) obtain C rl σ where rl:"rl ∈ R'" and P:"P = C⟨(fst rl) ⋅ σ⟩" and Q:"Q = C⟨(snd rl) ⋅ σ⟩" by auto
    let ?F = "λ q T. ({#Fun ≐ [(fst q)⋅ σ, (snd q)⋅ σ]#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])"
    have "is_Fun (fst rl)" using wf[unfolded wf_trs_def] rl R' by (metis Un_iff eq_fst_iff is_FunI)
    hence hf:"hole_pos C ∈ fun_poss P" using P 
      by (metis hole_pos_poss is_VarE is_VarI poss_is_Fun_fun_poss subst_apply_eq_Var subt_at_hole_pos)
    let ?p = "0 # hole_pos C"
    let ?S = "{# Fun ≐ [P, v] #}"
    let ?s = "Fun ≐ [P, v]"
    let ?rl = rl
    let ?σ = σ
    let ?T = "(?S - {#?s#}) + {#replace_at ?s  ?p ((snd ?rl) ⋅ ?σ)#}"
    have spσ:"?s |_ ?p = (fst ?rl) ⋅ ?σ" using P by auto
    have rlR:"rl ∈ R" using R' rl by fastforce
    have pps:"?p ∈ poss ?s" using hf using fun_poss_imp_poss by auto
    have "(?S, ?T) ∈ multiset_reduction_step_pos" 
    proof - 
      have "?s ∈# {#Fun ≐ [P, v]#}" by simp
      moreover have "?rl ∈ R" using R' rl by fastforce
      moreover have "?p ∈ poss ?s" using hf pps by blast
      moreover have "?s |_ ?p = fst ?rl ⋅ ?σ" using spσ by auto
      ultimately show ?thesis by blast
    qed
    hence st_step:"(?S, ?T) ∈ multiset_reduction_step" using mul_reduction_correspondence 
      by (metis rlR multiset_reduction_stepI pps replace_at_ident rstepI spσ surjective_pairing union_single_eq_member) 
    have "replace_at (Fun ≐ [P, v])  ?p ((snd rl) ⋅ σ) = (Fun ≐ [Q, v])" using Q by (auto simp add: P)
    hence "?T = {#Fun ≐ [Q, v]#}"by fastforce
    hence *:"({#Fun ≐ [P, v]#}, {#Fun ≐ [Q, v]#}) ∈ multiset_reduction_step" 
      using st_step by simp
    then show ?case using T trans_reachable_cond_red_step by blast
  qed
qed

lemma rstep_multiset_reduction_step_eq: fixes C::"(('f, 'v) term × ('f, 'v) term) list"
  assumes funas_A:"funas_trs (set C) ⊆ F"
    and "∀u v. (u, v) ∈ set C ⟶ ((u, v) ∈ (rstep R)⇧* ∧ v ∈ NF(rstep R))"
  shows "(∃T. ({#Fun ≐ [u, v]. (u, v) ∈# mset C#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀t ∈# T. t = Fun (⊤) []))" using assms
proof(induct C)
  case Nil
  then show ?case by auto
next
  case (Cons pair C)
  hence ftC:"funas_trs (set C) ⊆ F" unfolding funas_trs_def funas_rule_def by auto
  from Cons have *:"∀u v. (u, v) ∈ set C ⟶ ((u, v) ∈ (rstep R)⇧* ∧ v ∈ NF(rstep R))" by auto
  obtain u v where pair:"pair = (u, v)" by (meson surj_pair)
  have funas_u:"funas_term u ⊆ F" by (metis Cons.prems(1) lhs_wf list.set_intros(1) pair)
  have funas_v:"funas_term v ⊆ F" by (metis Cons.prems(1) list.set_intros(1) pair rhs_wf)
  have uvR:"(u, v) ∈ (rstep R)⇧*" using Cons by (simp add: pair)
  hence uv:"(u, v) ∈ (rstep R')⇧*" using funas_rstep_R' funas_u funas_v uvR by (simp add: funas_defs(2))
  obtain T where **:"({#Fun ≐ [u, v]. (u, v) ∈# mset C#}, T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" 
    using Cons * ftC by auto
  have "v ∈ NF(rstep R)" using Cons pair by auto
  hence v:"v ∈ NF(rstep R')" using normRR' by (metis NF_iff_no_step R' Un_iff rstep_union)
  hence uvv:"(u, v) ∈ (rstep R')⇧* ∧ v ∈ NF(rstep R')" using uv v by auto
  from rstep_single_eq_reduction_step_eq
  obtain T' where ***:"({#Fun ≐ [u, v]#}, T') ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T'. t = Fun ⊤ [])" using uv v by blast
  have multiset_step_ctxt1:"({#Fun ≐ [u, v]#} + {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset C#},   
    T'+ {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset C#}) ∈ (multiset_reduction_step)⇧*" 
    using *** multiset_red_add by blast
  have multiset_step_ctxt2:"(T'+ {#Fun ≐ [u1, v1]. (u1, v1) ∈# mset C#}, T' + T) ∈ (multiset_reduction_step)⇧*" using multiset_red_add ** add.commute by metis
  show ?case by (auto split:prod.splits, rule exI[of _ "T' + T"], insert ** *** pair addFunT[of T' T] 
        multiset_step_ctxt1 multiset_step_ctxt2, fastforce)
qed                                                                                        

lemma rstep_multiset_reduction_equiv:
  assumes funas_A:"funas_trs (set A) ⊆ F"
    and cstep_A:"(∀u v. ((u, v) ∈ set A) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧*))"
    and wn: "WN (rstep R)"
    and cr: "CR (rstep R)"
    and normτ: "normal_subst R τ"
    and sti:"∀ (u, v) ∈ set A. strongly_irreducible_term R v"
  shows "∃T. (subst_term_multiset τ (convert_equations_into_term_multiset A), T) ∈ (multiset_reduction_step)⇧*
          ∧ (∀ t ∈# T. t = Fun (⊤) [])" using assms
proof(induct A)
  case Nil
  then show ?case by auto
next
  case (Cons pair U)
  obtain u v where pair:"pair = (u, v)" using surjective_pairing by blast
  from funas_A have funU:"funas_trs (set U) ⊆ F" unfolding funas_trs_def funas_rule_def 
    using Cons unfolding funas_defs by auto
  have rstepτ:"∀s t. ((s, t) ∈ set ((u, v) # U)) ⟶ (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*" using Cons pair by fastforce
  have ftτ:"funas_trs (set (subst_equations_list τ ((u, v) # U))) ⊆ F" 
    unfolding subst_equations_list_def[unfolded subst_equation_def] funas_trs_def funas_rule_def 
    using wf_F_subst Cons(2) pair lhs_wf rhs_wf funas_A funU by (auto, smt (verit, ccfv_SIG) insertCI lhs_wf subsetD, 
        smt (verit, best) insertCI rhs_wf subsetD, blast, smt (verit) rhs_wf subsetD)
  have sruv:"∀(s, t) ∈ set ((u, v) # U). strongly_irreducible_term R t" using pair Cons by fastforce
  then show ?case
  proof(auto split:prod.splits, goal_cases)
    case (1 x1 x2)
    have pair:"x1 = u ∧ x2 = v" using 1 pair by force
    have *:"∀s t. (s, t) ∈ set (subst_equations_list τ ((u, v) # U)) ⟶ (s, t) ∈ (rstep R)⇧*" 
      using rstepτ unfolding subst_equations_list_def[unfolded subst_equation_def] by auto 
    have nfcond:"∀(s, t) ∈ set (subst_equations_list τ ((u, v) # U)). t ∈ NF (rstep R)" using rstepτ normτ sruv
      unfolding subst_equations_list_def[unfolded subst_equation_def] strongly_irreducible_term_def by auto
    from rstep_multiset_reduction_step_eq[of "subst_equations_list τ ((u, v) # U)"]
    obtain T where **:"(({#Fun ≐ [u, v]. (u, v) ∈# mset (subst_equations_list τ ((u, v) # U)) #}, T) ∈ 
      (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))" using * ftτ nfcond by (metis case_prodD)
    have equiv:"{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} =
        {#Fun ≐ [u, v]. (u, v) ∈# {#(fst p ⋅ τ, snd p ⋅ τ). p ∈# mset U#}#}"
    proof -
      have "{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#Fun ≐ [u ⋅ τ, v ⋅ τ]. (u, v) ∈# mset U#}"
      proof -
        have "{#t ⋅ τ. t ∈# {#Fun ≐ [u, v]. (u, v) ∈# mset U#}#} = {#(Fun ≐ [u, v]) ⋅ τ. (u, v) ∈# mset U#}" 
          using subst_Fun_mset_equiv by auto
        also have "... = {#(Fun ≐ [u ⋅ τ, v ⋅ τ]). (u, v) ∈# mset U#}" by auto
        ultimately show ?thesis by auto
      qed
      then show ?thesis by (smt (verit, best) case_prod_conv comp_apply multiset.map_comp multiset.map_cong0 prod.collapse)
    qed     
    show ?case  unfolding subst_term_multiset_def subst_equations_list_def using ** pair 
      by (simp add:subst_equations_list_def subst_equation_def, insert equiv, auto) 
  qed
qed

lemma multiset_narrowing_based_infeasibility: assumes wn:"WN (rstep R)"
  and cr:"CR (rstep R)"
  and funas_C:"funas_trs (set C) ⊆ F"
  and sti:"∀ (u, v) ∈ set C. strongly_irreducible_term R v"
shows "¬ multiset_narrowing_reaches_to_success C ⟹ infeasibility C" 
proof -
  assume "¬ multiset_narrowing_reaches_to_success C"
  hence ncs:"(¬ (∃σ S'. multiset_narrowing_derivation (convert_equations_into_term_multiset C) S' σ ∧ 
    (∀ s ∈# S'. s = Fun (⊤) [])))" unfolding multiset_narrowing_reaches_to_success_def by simp
  from convert_equations_into_rule_list_sound[OF funas_C]
  have wfC:"wf_equational_term_mset (convert_equations_into_term_multiset C)" by auto
  from infeasibility_using_multiset_narrowing[OF wfC ncs]
  have *:"¬ (∃θ T V. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))" by auto
  show ?thesis
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "¬ funas_trs (set C) ⊆ F ∨ (∃τ. (∀u v. ((u, v) ∈ set C) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧*)))" 
      unfolding infeasibility_def by auto
    with funas_C have "(∃τ. (∀u v. ((u, v) ∈ set C) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧*)))" by auto
    then obtain τ where τ:"(∀u v. ((u, v) ∈ set C) ⟶ ((u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧*))" by auto
    from WN_obtains_normalizable_subst[OF wn]
    have normτ:"normalizable_subst R τ" by auto
    from obtains_normalized_subst[OF normτ]
    obtain τ' where normτ':"normal_subst R τ'" and subst_r:"(∀x. (τ x, τ' x) ∈ (rstep R)⇧*)" by auto
    have uvC:"(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧*)"
    proof -
      { fix u v
        assume asm:"(u, v) ∈ set C"
        from * have **:"(u ⋅ τ', v ⋅ τ) ∈ (rstep R)⇧↔⇧*" using subst_r 
          by (smt (verit, ccfv_threshold) τ asm meetI meet_imp_conversion subset_iff substs_rsteps)
        from * have uvτ':"(u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧↔⇧*" using subst_r 
          by (meson ** conversionI' conversion_trans term_subst_rsteps transD)
        from sti[unfolded strongly_irreducible_term_def]
        have "v ⋅ τ' ∈ NF (rstep R)" using asm normτ' by auto
        hence "(u ⋅ τ', v ⋅ τ') ∈ (rstep R)⇧*" using cr uvτ'
          by (meson CR_NF_conv normalizability_E)
      } then show ?thesis by auto
    qed
    from rstep_multiset_reduction_equiv[OF funas_C uvC wn cr normτ' sti]
    have **:"∃ T. (subst_term_multiset  τ' (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" by auto
    then obtain T where ***:"(subst_term_multiset  τ' (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀t∈#T. t = Fun ⊤ [])" by auto
    have "(∃θ T V. normal_subst R θ ∧ (subst_term_multiset θ (convert_equations_into_term_multiset C), T) ∈ (multiset_reduction_step)⇧* ∧ (∀ t ∈# T. t = Fun (⊤) []))"
      by (rule exI[of _ τ'], rule exI[of _ T], insert * *** normτ', auto)
    then show False using * by auto
  qed
qed

lemma multiset_narrowing_based_reachable: assumes wn:"WN (rstep R)"
  and cr:"CR (rstep R)"
  and funas_C:"funas_trs (set C) ⊆ F"

  then obtain τ where *:"(∀u v. (u, v) ∈ set C ⟶ (u ⋅ τ, v ⋅ τ) ∈ (rstep R)⇧↔⇧*)" unfolding R_unifiable_def by auto

    { fix u v

      from * have **:"(u ⋅ τ', v ⋅ τ) ∈ (rstep R)⇧↔⇧*" using subst_r

    } then show ?thesis by auto
  qed
  then show ?thesis unfolding reachability_def by auto
qed

lemma complete_rstep_mulrstep_correspondence: assumes comp: "complete (rstep R)"

proof -
  from comp have sn:"SN (rstep R)" and cr:"CR (rstep R)"

  from sn have sn_mul:"SN (multiset_reduction_step)" 
    by (simp add: SN_R_imp_SN_multiset_reduction_step)
  from cr sn_mul have "CR (multiset_reduction_step)"
    by (simp add: CR_correspondence sn)
  then show ?thesis using sn_mul
    by (simp add: complete_on_def)
qed

lemma mulr_closed_addition: assumes A1B1:"(A1, B1) ∈ (multiset_reduction_step)⇧*"
    and A2B2:"(A2, B2) ∈ (multiset_reduction_step)⇧*"
  shows "(A1 + A2, B1 + B2) ∈ (multiset_reduction_step)⇧*" using assms
proof -
  from A1B1 have "(A1 + A2, B1 + A2) ∈ multiset_reduction_step⇧*" 
    by (meson multiset_narrowing.multiset_red_add multiset_narrowing_axioms)
  moreover have "(B1 + A2, B1 + B2) ∈ multiset_reduction_step⇧*"
    by (metis A2B2 add.commute multiset_red_add)
  ultimately show ?thesis by auto
qed

lemma mulr_subst_normal_subst_equiv: assumes comp: "complete (rstep R)"
  and subst: "(∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
  and G:"G ∈ NF (multiset_reduction_step)"
shows  "(∃θ'. normal_subst R θ' ∧ (subst_term_multiset θ' S, G) ∈ (multiset_reduction_step)⇧*)"
proof -
  from comp have sn:"SN (rstep R)" and cr:"CR (rstep R)" by (simp add: complete_on_def, insert comp, auto)
  from comp have comp_mult:"complete (multiset_reduction_step)" using complete_rstep_mulrstep_correspondence by simp
  from comp_mult have cr_mul:"CR (multiset_reduction_step)" and sn_mul:"SN (multiset_reduction_step)"
    by (simp add: complete_on_def, insert comp_mult, auto) 
  from subst obtain θ where *:"(subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*" by auto
  from SN_obtains_normalizable_subst[OF sn]
  have normθ:"normalizable_subst R θ" by auto
  from obtains_normalized_subst[OF normθ]
  obtain θ' where normτ':"normal_subst R θ'" and subst_r:"(∀x. (θ x, θ' x) ∈ (rstep R)⇧*)" by auto
  have tθθ':"∀t. (t ⋅ θ, t ⋅ θ') ∈ (rstep R)⇧*" using subst_r using substs_rsteps by blast
  hence "(subst_term_multiset θ S, subst_term_multiset θ' S) ∈ (multiset_reduction_step)⇧*"
  proof -
    let ?size = "size (subst_term_multiset θ S)"
    have *:"subst_term_multiset θ S = {# (t ⋅ θ). t ∈# S #}" unfolding subst_term_multiset_def by auto
    have **:"subst_term_multiset θ' S = {# (t ⋅ θ'). t ∈# S #}" unfolding subst_term_multiset_def by simp
    have Stθθ':"∀t ∈# S. ({# t ⋅ θ #}, {# t ⋅ θ' #}) ∈ (multiset_reduction_step)⇧*" using tθθ' rm_correspondence
      by (auto, metis add_mset_remove_trivial union_single_eq_member)
    moreover have "({# (t ⋅ θ). t ∈# S #}, {# (t ⋅ θ'). t ∈# S #}) ∈ (multiset_reduction_step)⇧*"
    proof(induct "?size" arbitrary:S)
      case 0
      then show ?case using * 
        by (simp add: subst_term_multiset_def)
    next
      case (Suc x)
      then obtain u where u:"u ∈# S" by fastforce
      let ?S = "S - {#u#}"
      have uθθ':"({# u ⋅ θ #}, {# u ⋅ θ' #}) ∈ (multiset_reduction_step)⇧*" using Stθθ' u 
        by (metis add_mset_add_single add_mset_remove_trivial rm_correspondence tθθ' union_single_eq_member)
      have "size ?S = size S - 1" using u 
        by (simp add: size_Diff_singleton)
      hence "({#t ⋅ θ. t ∈# ?S#}, {#t ⋅ θ'. t ∈# ?S#}) ∈ multiset_reduction_step⇧*"
        using Suc by (simp add: subst_term_multiset_def)
      moreover have "({#t ⋅ θ. t ∈# ?S#} + {# u ⋅ θ #}, {#t ⋅ θ'. t ∈# ?S#} + {# u ⋅ θ' #}) ∈ multiset_reduction_step⇧*"
        using mulr_closed_addition calculation uθθ' by blast
      then show ?case using * ** u 
        by (auto, metis (no_types, lifting) image_mset_add_mset insert_DiffM)
    qed
    then show ?thesis using G comp_mult by (simp add: * **)
  qed
  then show ?thesis using *
    by (metis G NF_not_suc cr_mul normτ' normalizability_E sn_mul the_NF the_NF_steps)
qed

(*  Strong completeness of multiset narrowing w.r.t. ordinary reachability analysis,
    assuming that a semi-complete rewrite system R is given with the strongly irreducibility condition *)

theorem multiset_narrowing_based_reachability: assumes semi_comp:"semi_complete (rstep R)"
  and funas_C:"funas_trs (set C) ⊆ F"
  and sti:"∀ (u, v) ∈ set C. strongly_irreducible_term R v"
shows "multiset_narrowing_reaches_to_success C ⟹ reachability C"
  "¬ multiset_narrowing_reaches_to_success C ⟹ infeasibility C"
proof -
  from semi_comp have wn:"WN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: semi_complete_on_def, insert semi_comp, auto)
  show "multiset_narrowing_reaches_to_success C ⟹ reachability C" 
    using assms cr multiset_narrowing_based_reachable reachability_def wn by blast
  show "¬ multiset_narrowing_reaches_to_success C ⟹ infeasibility C"
    using assms by (simp add: cr multiset_narrowing_based_infeasibility wn)
qed

(* The following theorem is the strong completeness of multiset narrowing
  w.r.t. multiset reachability analysis, assuming that a complete R is given and the goal G
  is in normal form w.r.t. the multiset_reduction_step *)

lemma multiset_narrowing_based_non_reachability_strong_completeness: 
  assumes comp: "complete (rstep R)"
    and G:"G ∈ NF multiset_reduction_step"
  shows "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
proof -
  have "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    by (simp add: multiset_narrowing_based_infeasibility_condition)
  then show "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    using mulr_subst_normal_subst_equiv comp G by blast
qed

theorem multiset_narrowing_based_reachability_strong_completeness: 
  assumes comp: "complete (rstep R)"
    and G:"G ∈ NF multiset_reduction_step"
  shows "multiset_narrowing_reachable_from_to S G ⟹ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
        "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
proof -

    by (simp add: multiset_narrowing_based_reachable_sufficient_condition)
  have "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. normal_subst R θ ∧ (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    by (simp add: multiset_narrowing_based_infeasibility_condition)
  then show "¬ multiset_narrowing_reachable_from_to S G ⟹ ¬ (∃θ. (subst_term_multiset θ S, G) ∈ (multiset_reduction_step)⇧*)"
    using mulr_subst_normal_subst_equiv comp G by blast
qed
