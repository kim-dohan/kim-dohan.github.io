section ‹Formalization of equational narrowing based reachability›

theory Equational_Narrowing_Reachability
  imports
    Equational_Narrowing
begin

locale equational_narrowing_reachability = equational_narrowing R DOTEQ TOP R' F D x
  for R::"('f, 'v:: infinite) trs" 
    and DOTEQ :: 'f ("≐")
    and TOP :: "'f" ("⊤") 
    and R' :: "('f, 'v:: infinite) trs" 
    and F :: "'f sig" 
    and D :: "'f sig"
    and x :: 'v
begin

(* First consider  narrowing based reachability without using equational terms. 
  If a narrowing derivation reaches the target state, then a solution of the reachability problem exists. 
  No CR or SN or WN of R is required. *)

lemma narrowing_based_reachable_single: 
  assumes nar:"(s, t, σ) ∈ narrowing_step"
  shows "(s ⋅ σ, t) ∈ rstep R"
proof -
  from nar obtain rl ω p where t':"t = (replace_at s p (snd rl)) ⋅ σ" and ωrl:"ω ∙ rl ∈ R"
    and disj:"(vars_term s ∩ vars_rule rl = {})" and p:"p ∈ fun_poss s" and mgu:"mgu (s |_ p) (fst rl) = Some σ" by auto
  from t' have t:"t = replace_at (s ⋅ σ) p (snd rl ⋅ σ)" using p 
    by (simp add: ctxt_of_pos_term_subst fun_poss_imp_poss)
  let ?C = "ctxt_of_pos_term p (s ⋅ σ)"
  from mgu have "(s |_ p) ⋅ σ = fst rl ⋅ σ" using subst_apply_term_eq_subst_apply_term_if_mgu by auto
  then show ?thesis using rstep.intros[of "fst rl" "snd rl" R "s ⋅ σ" ?C "σ" t] 
  by (smt (verit) ωrl fun_poss_imp_poss p perm_rstep_conv poss_imp_subst_poss prod.exhaust_sel 
    replace_at_ident rstep_ctxt rstep_subst subset_iff subset_rstep subt_at_subst t)
qed

lemma narrowing_based_reachable': 
  assumes nar:"narrowing_derivation s t σ"
  shows "(s ⋅ σ, t) ∈ (rstep R)⇧*"
proof -
  from nar obtain n where "narrowing_derivation_num s t σ n" unfolding narrowing_derivation_def 
      narrowing_derivation_num_def by auto
  then show ?thesis
  proof(induct n arbitrary: t σ)
    case 0
    then show ?case by (metis n0_narrowing_derivation_num rtrancl.simps subst.cop_nil)
  next
    case (Suc n)
    from ‹narrowing_derivation_num s t σ (Suc n)›
    have "(∃f τ. f 0 = s ∧ f (Suc n) = t ∧ 
      (∀i < (Suc n). ((f i), (f (Suc i)), (τ i)) ∈ narrowing_step) ∧ (σ = compose (map (λi. (τ i)) [0 ..< Suc n])))"
      unfolding narrowing_derivation_num_def by auto
    then obtain f τ where f0:"f 0 = s" and fsucn:"f (Suc n) = t" and 
      relchain:"∀i < (Suc n). ((f i), (f (Suc i)), (τ i)) ∈ narrowing_step" 
      and σ:"(σ = compose (map (λi. (τ i)) [0 ..< (Suc n)]))" by auto
    let ?τ = "compose (map (λi. (τ i)) [0 ..< n])"
    let ?σ = "if n = 0 then Var else ?τ"
    let ?f = "λi. (if i ≤ n then f i else undefined)"
    from relchain obtain u where u:"f n = u" by simp
    from relchain have nchain:"∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ narrowing_step" by simp
    have "(∃f τ. f 0 = s ∧ f n = u ∧ (∀i < n. ((f i), (f (Suc i)), (τ i)) ∈ narrowing_step))"
      by (rule exI[of _ "?f"] rule exI[of _ "?σ"], insert u f0 nchain, auto)
    moreover have "narrowing_derivation_num s u ?σ n" unfolding narrowing_derivation_num_def
      using u calculation f0 nchain by auto 
    ultimately have IH:"(s ⋅ ?σ, u) ∈ (rstep R)⇧*" using Suc(1) by auto
    have "(u, t, τ n) ∈ narrowing_step" using relchain fsucn u by auto
    from narrowing_based_reachable_single[OF this] 
    have *:"(u ⋅ (τ n), t) ∈ (rstep R)" by auto
    have "σ = ?σ ∘⇩s (τ n)" using σ by auto  
    then show ?case using IH by (smt (verit, del_insts) * rstep_rtrancl_idemp rstep_subst 
      rtrancl.rtrancl_into_rtrancl subst_subst_compose)
  qed
qed

lemma narrowing_based_reachable: 
  assumes nar:"∃σ. narrowing_derivation s (t ⋅ σ) σ"
  shows "∃θ. (s ⋅ θ , t ⋅ θ) ∈ (rstep R)⇧*"
proof -
  from nar obtain σ where nd:"narrowing_derivation s (t ⋅ σ) σ" by auto
  from narrowing_based_reachable'[OF nd]
  have *:"(s ⋅ σ , t ⋅ σ) ∈ (rstep R)⇧*" by auto
  then show ?thesis by (rule exI[of _ σ])
qed

(* A sufficient condition of satisfying the reachability problem *)

lemma narrowing_based_reachable_with_unification: 
  assumes nar:"∃σ. narrowing_derivation s (t' ⋅ σ) σ ∧ mgu (t' ⋅ σ) (t ⋅ σ) ≠ None"
  shows "∃θ. (s ⋅ θ , t ⋅ θ) ∈ (rstep R)⇧*"
proof -
  from nar obtain σ δ where nd:"narrowing_derivation s (t' ⋅ σ) σ" and mgu:"mgu (t' ⋅ σ) (t ⋅ σ) = Some δ" by auto
  from narrowing_based_reachable'[OF nd]
  have "(s ⋅ σ , t' ⋅ σ) ∈ (rstep R)⇧*" by auto
  moreover from mgu have "(t' ⋅ σ) ⋅ δ = (t ⋅ σ) ⋅ δ" 
    using subst_apply_term_eq_subst_apply_term_if_mgu by auto
  ultimately have *: "(s ⋅ (σ ∘⇩s δ) , t ⋅ (σ ∘⇩s δ)) ∈ (rstep R)⇧*" 
    by (metis (mono_tags, lifting) rsteps_closed_subst subst_subst_compose)
  show ?thesis by (rule exI[of _ "(σ ∘⇩s δ)"], insert *, auto)
qed

(* Narrowing based reachability using equational terms. If a narrowing derivation reaches the target state, 
  then a solution of the reachability problem exists. *)

lemma equational_narrowing_based_reachability_pre: 
  assumes wf_S:"wf_equational_term S"
    and nd:"narrowing_derivation S (Fun ⊤ []) σ"
  shows "(S ⋅ σ, Fun ⊤ []) ∈ (rstep R)⇧*" using narrowing_based_reachable' nd wf_S by auto

lemma equational_narrowing_based_reachability': 
  assumes wf_S:"wf_equational_term S"
    and nd:"∃σ. narrowing_derivation S (Fun ⊤ []) σ"
  shows "∃θ. (S ⋅ θ, Fun ⊤ []) ∈ (rstep R)⇧*" using narrowing_based_reachable' nd wf_S by auto

lemma infeasibility_using_narrowing: assumes wf_S:"wf_equational_term S"
  and "¬ (∃σ. narrowing_derivation S (Fun ⊤ []) σ)"
shows "¬ (∃θ. normal_subst R θ ∧ (S ⋅ θ, Fun ⊤ []) ∈ (rstep R)⇧*)"
proof(rule ccontr)
  assume asm:"¬?thesis"
  then obtain θ T  where normθ:"normal_subst R θ" and relsteps:"(S ⋅ θ, T) ∈ (rstep R)⇧*" and goal:"(T = Fun (⊤) [])" by auto
  let ?V = "vars_term S ∪ subst_domain θ"
  have "finite (vars_term S)" unfolding vars_term_set_def by simp
  moreover have "finite (subst_domain θ)" using finite_subst_domain by auto 
  ultimately have fV:"finite ?V" by simp
  define U where U:"U = S ⋅ θ" 
  have wf_U:"wf_equational_term U" using U wf_S wf_eq_subst by blast
  have reltran:"(U, T) ∈ (rstep R)⇧*" using relsteps by (simp add: U)
  from lifting_lemma_equational_terms [OF normθ wf_S U, of ?V T ]
  have "∃σ θ' S'. narrowing_derivation S S' σ ∧ T =  S' ⋅ θ' ∧ normal_subst R θ' ∧ wf_equational_term S'" using asm fV reltran by blast
  then obtain σ θ' S' where nar:"narrowing_derivation S S' σ" and T:"T =  S' ⋅ θ'" and normθ':"normal_subst R θ'" 
    and wf_S':"wf_equational_term S'" by auto
  have "(∀u v θ. Fun ≐ [u, v] ⋅ θ ≠ Fun (⊤) [])" using wf_eq_subst by simp
  from wf_S'[unfolded wf_equational_term_mset_def[unfolded wf_equational_term_def]]
  have "S'= Fun ⊤ []" using wf_eq_subst using T goal wf_equational_term_def by fastforce
  then show False using asm goal using assms(2) nar by auto
qed

lemma equational_narrowing_based_infeasibility_using_normalized_subst_reachability_property: 
  assumes nsrp:"normalized_subst_reachability_property s t"
    and funas_C:"funas_rule (s, t) ⊆ F"
  shows "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ infeasible (s, t)" 
proof -
  assume "¬ narrowing_derivation_reaches_to_success (s, t)"
  hence ncs:"¬ (∃σ. narrowing_derivation (Fun (≐) [s, t]) ( Fun (⊤) []) σ) " unfolding narrowing_derivation_reaches_to_success_def by auto
  from convert_equation_into_term_sound[OF funas_C]
  have wfC:"wf_equational_term (Fun (≐) [s, t])" by auto
  from infeasibility_using_narrowing[OF wfC ncs]
  have *:"¬ (∃θ. normal_subst R θ ∧ (Fun (≐) [s, t] ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*)" by auto
  show ?thesis
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "(¬ funas_rule (s, t) ⊆ F ∨ (∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*))"
      unfolding infeasible_def by auto
    with funas_C have reach:"∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*" by auto
    hence "∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R')⇧*" using R' R_sig unfolding funas_defs 
      by (metis (no_types, lifting) Un_subset_iff fst_eqD funas_C funas_rstep_R' funas_rule_def snd_eqD wf_F_subst)
    then obtain τ where "(s ⋅ τ, t ⋅ τ) ∈ (rstep R')⇧*" by auto
    with nsrp[unfolded normalized_subst_reachability_property_def] obtain τ' where 
      τ':"(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧*" and normτ':"normal_subst R τ'" and "∀x. (τ x, τ' x) ∈ (rstep R)⇧*" 
      by (metis (no_types, opaque_lifting) Int_iff R' Un_upper1 inf.orderE rstep_union rtrancl_mono)
    from rstep_reduction_step_eq
    have "∃θ. normal_subst R θ ∧ (Fun ≐ [s ⋅ θ, t ⋅ θ], Fun (⊤) []) ∈ (rstep R)⇧*" using τ' normτ' funas_C by blast
    then show False using * by auto
  qed
qed

(* Sufficient condition for normalized_subst_reachability property *)

lemma normalized_subst_reachability_cond: assumes wn:"WN (rstep R)"
  and cr: "CR (rstep R)"
  and funas_C:"funas_rule (s, t) ⊆ F"
  and sti:"strongly_irreducible_term R t"
shows "normalized_subst_reachability_property s t" 
proof -
  { fix τ
    assume asm:"(s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*"
    from WN_obtains_normalizable_subst[OF wn]
    have normτ:"normalizable_subst R τ" by auto
    from obtains_normalized_subst[OF normτ]
    obtain τ' where normτ':"normal_subst R τ'" and ττ':"∀x. (τ x, τ' x) ∈ (rstep R)⇧*" by auto
    have rstepτ':"(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧*"
    proof -
      have "(s ⋅ τ, s ⋅ τ') ∈ (rstep R)⇧*" using ττ' 
        by (simp add: substs_rsteps)
      moreover have "(t ⋅ τ, t ⋅ τ') ∈ (rstep R)⇧*" using ττ'  
        by (simp add: substs_rsteps)
      moreover have "t ⋅ τ' ∈ NF (rstep R)" using sti[unfolded strongly_irreducible_term_def] normτ' by auto
      moreover have "(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧↔⇧*" 
        by (meson asm calculation(1) calculation(2) meetI meet_imp_conversion subset_eq transD trans_rtrancl)
      ultimately show "(s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧*" using cr[unfolded CR_on_def]
        by (meson CR_NF_conv cr normalizability_E)
    qed
    hence "(∃ τ'. normal_subst R τ' ∧ (s ⋅ τ', t ⋅ τ') ∈ (rstep R)⇧* ∧ (∀x. (τ x, τ' x) ∈ (rstep R)⇧*))" 
      using normτ' ττ' rstepτ' by auto
  } then show ?thesis unfolding normalized_subst_reachability_property_def by auto
qed

(* Equational narrowing-based infeasibility *)

lemma narrowing_based_infeasibility_implies_infeasibility: assumes wn:"WN (rstep R)"
  and cr: "CR (rstep R)"
  and funas_C:"funas_rule (s, t) ⊆ F"
  and sti:"strongly_irreducible_term R t"
shows "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ (¬ (∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*))" 
proof -
  assume "¬ narrowing_derivation_reaches_to_success (s, t)"
  hence ncs:"¬ (∃σ. narrowing_derivation (Fun (≐) [s, t]) ( Fun (⊤) []) σ) " unfolding narrowing_derivation_reaches_to_success_def by auto
  from convert_equation_into_term_sound[OF funas_C]
  have wfC:"wf_equational_term (Fun (≐) [s, t])" by auto
  from infeasibility_using_narrowing[OF wfC ncs]
  have *:"¬ (∃θ. normal_subst R θ ∧ (Fun (≐) [s, t] ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*)" by auto
  show ?thesis
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "(¬ funas_rule (s, t) ⊆ F ∨ (∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*))"
      unfolding infeasible_def by auto
    with funas_C have reach:"∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧*" by auto
    from rstep_reduction_equiv[OF wn cr sti funas_C reach]
    have "∃θ. normal_subst R θ ∧ (Fun ≐ [s ⋅ θ, t ⋅ θ], Fun (⊤) []) ∈ (rstep R)⇧*" by auto
    then show False using * by auto
  qed
qed

(* No SN or WN required for the following theorem *)
theorem narrowing_based_infeasibility_implies_normalizable_infeasibility: assumes cr: "CR (rstep R)"
  and funas_C:"funas_rule (s, t) ⊆ F"
shows "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ normalizable_infeasibility (s, t)" 
proof -
  assume "¬ narrowing_derivation_reaches_to_success (s, t)"
  hence ncs:"¬ (∃σ. narrowing_derivation (Fun (≐) [s, t]) ( Fun (⊤) []) σ) " unfolding narrowing_derivation_reaches_to_success_def by auto
  from convert_equation_into_term_sound[OF funas_C]
  have wfC:"wf_equational_term (Fun (≐) [s, t])" by auto
  from infeasibility_using_narrowing[OF wfC ncs]
  have *:"¬ (∃θ. normal_subst R θ ∧ (Fun (≐) [s, t] ⋅ θ, Fun (⊤) []) ∈ (rstep R)⇧*)" by auto
  show ?thesis
  proof(rule ccontr)
    assume "¬ ?thesis"
    hence "(¬ funas_rule (s, t) ⊆ F ∨ (∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧* ∧ normalizable_subst R τ) )"
      unfolding normalizable_infeasibility_def by auto
    with funas_C have reach:"∃τ. (s ⋅ τ, t ⋅ τ) ∈ (rstep R)⇧* ∧ normalizable_subst R τ" by auto
    from rstep_reduction_normalizable_equiv[OF cr reach]
    have "∃θ. normal_subst R θ ∧ (Fun ≐ [s ⋅ θ, t ⋅ θ], Fun (⊤) []) ∈ (rstep R)⇧*" by auto
    then show False using * by auto
  qed
qed

lemma narrowing_based_infeasibility_implies_infeasible: assumes "semi_complete (rstep R)" 
  and funas_C:"funas_rule (s, t) ⊆ F"
  and nd: "¬ narrowing_derivation_reaches_to_success (s, t)"
shows "infeasible (s, t)" using assms narrowing_based_infeasibility_implies_normalizable_infeasibility 
  by (meson equational_narrowing.WN_obtains_normalizable_subst equational_narrowing_axioms infeasible_def normalizable_infeasibility_def semi_complete_on_def)

(* Equational narrowing-based reachable *)

lemma equational_narrowing_based_reachable:
  assumes wn: "WN (rstep R)"
    and cr: "CR (rstep R)"
    and nd:"∃σ. narrowing_derivation (Fun ≐ [s, t]) (Fun ⊤ []) σ"
    and funas_st:"funas_rule (s, t) ⊆ F"
    and sti:"strongly_irreducible_term R t"
  shows "∃θ. (s ⋅ θ, t ⋅ θ) ∈ (rstep R)⇧*"
proof -
  have wf:"wf_equational_term (Fun ≐ [s, t])" unfolding wf_equational_term_def using funas_st D D_fresh 
      funas_defs(2) by fastforce+
  from equational_narrowing_based_reachability'[OF wf nd]
  obtain θ where *: "(Fun ≐ [s ⋅ θ, t ⋅ θ], Fun ⊤ []) ∈ (rstep R)⇧+"
    by (auto, simp add: rtrancl_eq_or_trancl)
  hence fr':"funas_rule (s ⋅ θ, t ⋅ θ) ⊆ F" using funas_st unfolding funas_defs wf_F_subst 
    by (auto, (meson subsetD wf_F_subst)+) 
  from wn have normθ:"normalizable_subst R θ"
    using WN_obtains_normalizable_subst by auto
  from obtains_normalized_subst[OF normθ]
  obtain θ' where normθ':"normal_subst R θ'" and θθ':"(∀x. (θ x, θ' x) ∈ (rstep R)⇧*)" by auto
  have fr:"funas_rule (s ⋅ θ', t ⋅ θ') ⊆ F" using funas_st unfolding funas_defs wf_F_subst 
    by (auto, (meson subsetD wf_F_subst)+)
  from norm_subst_rstep_preserve[OF normθ' θθ']
  have "(Fun ≐ [s ⋅ θ, t ⋅ θ], Fun ≐ [s ⋅ θ', t ⋅ θ']) ∈ (rstep R)⇧*" using fr unfolding funas_defs by auto
  hence **:"(Fun ≐ [s ⋅ θ', t ⋅ θ'], Fun ⊤ []) ∈ (rstep R)⇧*" using * cr[unfolded CR_on_def]
    by (metis CR_NF_conv CR_imp_conversionIff_join NF_I cr iso_tuple_UNIV_I normalizability_E not_reducible_T 
        r_into_rtrancl trancl_rtrancl_absorb)
  have "t ⋅ θ' ∈ NF (rstep R)" using sti[unfolded strongly_irreducible_term_def] 
    using normθ' by auto 
  hence nt:"t ⋅ θ' ∈ NF (rstep R')" using R' NF_trs_mono[of R' R] by auto 
  from rstep_red_equiv[OF fr ** nt] 
  have "(s ⋅ θ', t ⋅ θ') ∈ (rstep R')⇧*" using ** by auto
  hence "(s ⋅ θ', t ⋅ θ') ∈ (rstep R)⇧*" using R' by (metis R' in_rtrancl_UnI rstep_union)
  then show ?thesis by (rule exI[of _ θ'])
qed

(* Equational narrowing based reachability, providing a decision procedure  
   for strongly irreducible terms t *)

theorem narrowing_based_reachability: 
  assumes semi_comp:"semi_complete (rstep R)"
    and funas_C:"funas_rule (s, t) ⊆ F"
    and sti:"strongly_irreducible_term R t"
  shows "narrowing_derivation_reaches_to_success (s, t) ⟹ reachable (s, t)"
    "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ infeasible (s, t)"
proof -
  from semi_comp have wn:"WN (rstep R)" and cr:"CR (rstep R)" 
    by (simp add: semi_complete_on_def, insert semi_comp, auto)
  show "narrowing_derivation_reaches_to_success (s, t) ⟹ reachable (s, t)"
    by (simp add: cr funas_C equational_narrowing_based_reachable narrowing_derivation_reaches_to_success_def reachable_def sti wn)
  show "¬ narrowing_derivation_reaches_to_success (s, t) ⟹ infeasible (s, t)"
    using narrowing_based_infeasibility_implies_infeasible[OF semi_comp funas_C] .
qed
